    93→    93→function eventStatus(item: LxScheduleItem, startIso: string, endIso: string): string {
    94→    94→  const tag = String(item.tag || "").toLowerCase();
    95→    95→  if (tag === "live" || tag === "live now") return "LIVE";
    96→    96→  if (tag === "upcoming") return "UPCOMING";
    97→    97→  if (tag === "ended" || tag === "cancelled") return tag === "cancelled" ? "CANCELLED" : "ENDED";
    98→    98→  const now = Date.now();
    99→    99→  const s = new Date(startIso || "").getTime();
   100→   100→  const e = new Date(endIso || "").getTime();
   101→   101→  if (Number.isFinite(s) && Number.isFinite(e)) {
   102→   102→    if (now >= s && now <= e) return "LIVE";
   103→   103→    if (now < s) return "UPCOMING";
   104→   104→  }
   105→   105→  return "ENDED";
   106→   106→}

   136→   136→function contentNodesFromItems(
   137→   137→  batchId: string,
   138→   138→  subjectId: string,
   139→   139→  items: LxScheduleItem[],
   140→   140→  kind: "LECTURE" | "DPP_VIDEOS" | "NOTES",
   141→   141→): Array<{ _id: string; type: string; data: Record<string, unknown> }> {
   142→   142→  const nodes: Array<{ _id: string; type: string; data: Record<string, unknown> }> = [];
   143→   143→  for (const item of items) {
   144→   144→    const pdfId = item._id || "";
   145→   145→
   146→   146→    if (kind === "LECTURE" || kind === "DPP_VIDEOS") {
   147→   147→      const vd = item.videoDetails || {};
   148→   148→      nodes.push({
   149→   149→        _id: item._id || "",
   150→   150→        type: kind,
   151→   151→        data: {
   152→   152→          _id: item._id || "",
   153→   153→          topic: item.topic || vd.name || item.name || "Lecture",
   154→   154→          date: item.date || item.startTime || "",
   155→   155→          urlType: item.urlType || "",
   156→   156→          url: item.url || "",
   157→   157→          videoDetails: {
   158→   158→            _id: item._id || "",
   159→   159→            image: vd.image || item.previewImageUrl || "",
   160→   160→            videoUrl: "",
   161→   161→            duration: vd.duration || "",
   162→   162→          },
   163→   163→          homeworkIds: asArray(item.homeworkIds).flatMap((hw) => {
   164→   164→            const h = hw as { _id?: string; topic?: string; attachmentIds?: unknown[] };
   165→   165→            return asArray(h.attachmentIds).map((att) => {
   166→   166→              const a = att as { _id?: string };
   167→   167→              return {
   168→   168→                _id: h._id || "",
   169→   169→                topic: h.topic || "Class Note",
   170→   170→                attachmentIds: [pdfAttachmentRef(batchId, subjectId, pdfId, a._id || "")],
   171→   171→              };
   172→   172→            });
   173→   173→          }),
   174→   174→          exerciseIds: [],
   175→   175→        },
   176→   176→      });
   177→   177→      continue;
   178→   178→    }
   179→   179→
   180→   180→    // NOTES / DppNotes → PDF cards
   181→   181→    const atts: Array<Record<string, unknown>> = [];
   182→   182→    let title = item.topic || item.name || "";
   183→   183→    const homeworkSources = [
   184→   184→      ...asArray(item.homeworkIds),
   185→   185→      ...asArray(item.dpp?.homeworkIds),
   186→   186→    ];
   187→   187→    for (const hw of homeworkSources) {
   188→   188→      const h = hw as { _id?: string; topic?: string; attachmentIds?: unknown[] };
   189→   189→      if (!title && h.topic) title = h.topic;
   190→   190→      for (const att of asArray(h.attachmentIds)) {
   191→   191→        const a = att as { _id?: string; key?: string; baseUrl?: string; name?: string };
   192→   192→        if (!a?._id) continue;
   193→   193→        if (!title) title = a.name || "";
   194→   194→        // some attachments already carry a real static.pw.live key
   195→   195→        const direct = a.baseUrl && a.key ? `${a.baseUrl}${a.key}` : "";
   196→   196→        const ref = direct
   197→   197→          ? { baseUrl: direct, key: "", _id: a._id }
   198→   198→          : pdfAttachmentRef(batchId, subjectId, pdfId, a._id);
   199→   199→        atts.push({
   200→   200→          _id: a._id,
   201→   201→          topic: h.topic || a.name || item.topic || item.name || "Study Material",
   202→   202→          attachmentIds: [ref],
   203→   203→        });
   204→   204→      }
   205→   205→    }
   206→   206→    if (atts.length === 0) continue;
   207→   207→    nodes.push({
   208→   208→      _id: item._id || "",
   209→   209→      type: "NOTES",
   210→   210→      data: {
   211→   211→        _id: item._id || "",
   212→   212→        topic: title || "Study Material",
   213→   213→        date: item.date || item.startTime || "",
   214→   214→        homeworkIds: atts,
   215→   215→      },
   216→   216→    });
   217→   217→  }
   218→   218→  return nodes;
   219→   219→}

   256→   256→// ---------------------------------------------------------------------------
   257→   257→// Lecture stream via the StudyParcham media proxy (preferred lecture format):
   258→   258→//
   259→   259→//   https://proxy.studyparcham.in/{cf_host}/{folder}/hls/720/main.m3u8{signature}
   260→   260→//
   261→   261→// {cf_host} is the PW CloudFront distribution the video-url API pointed at,
   262→   262→// {folder} the per-video uuid folder and {signature} the folder-wide signed
   263→   263→// query (?Signature=…&Key-Pair-Id=…&Policy=…) — the embedded Policy covers
   264→   264→// …/<folder>/*, so it authorizes the hls/720 rendition as well. The proxy
   265→   265→// simply forwards to the CloudFront host named in the path.
   266→   266→//
   267→   267→// We probe the rendition once per folder (cached) and hand the player a
   268→   268→// same-origin relay URL (/api/pw/x/…). If the HLS rendition is unavailable we
   269→   269→// try the same asset directly on CloudFront and finally fall back to the
   270→   270→// legacy DASH+ClearKey flow below.
   271→   271→// ---------------------------------------------------------------------------
   272→   272→const PROXY_HOST = "proxy.studyparcham.in";
   273→   273→const PROXY_RENDITION = "720/main.m3u8";
   274→   274→const UUID_FOLDER_RE = /^[a-f0-9-]{16,}$/i;
   275→   275→
   276→   276→type HlsChoice = { via: "proxy" | "direct"; rendition: string } | null;
   277→   277→
   278→   278→/** Same shape resolveVideoPayload hands to the player. */
   279→   279→interface ResolvedVideoPayload {
   280→   280→  kind: "dash" | "hls" | "youtube" | "mp4";
   281→   281→  url: string;
   282→   282→  signedUrl: string;
   283→   283→  clearKeys: Record<string, string> | null;
   284→   284→  poster: string;
   285→   285→  drmDetails: unknown;

   287→   287→
   288→   288→const probeCache = new Map<string, { choice: HlsChoice; at: number }>();
   289→   289→const PROBE_OK_TTL = 30 * 60 * 1000;
   290→   290→const PROBE_FAIL_TTL = 3 * 60 * 1000;
   291→   291→
   292→   292→async function manifestOk(upstreamUrl: string): Promise<boolean> {
   293→   293→  try {
   294→   294→    const controller = new AbortController();
   295→   295→    const timer = setTimeout(() => controller.abort(), 9000);
   296→   296→    const res = await fetch(upstreamUrl, {
   297→   297→      cache: "no-store",
   298→   298→      signal: controller.signal,
   299→   299→      headers: {
   300→   300→        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 Chrome/126.0 Safari/537.36",
   301→   301→      },
   302→   302→    });
   303→   303→    if (!res.ok) {
   304→   304→      clearTimeout(timer);
   305→   305→      return false;
   306→   306→    }
   307→   307→    const text = await res.text();
   308→   308→    clearTimeout(timer);
   309→   309→    return text.trimStart().startsWith("#EXTM3U");
   310→   310→  } catch {
   311→   311→    return false;
   312→   312→  }
   313→   313→}

   315→   315→/** Pick the working HLS-720 upstream (proxy first, direct CloudFront second). */
   316→   316→async function chooseHlsUpstream(cfHost: string, folder: string, signedQuery: string): Promise<HlsChoice> {
   317→   317→  const cached = probeCache.get(folder);
   318→   318→  if (cached) {
   319→   319→    const ttl = cached.choice ? PROBE_OK_TTL : PROBE_FAIL_TTL;
   320→   320→    if (Date.now() - cached.at < ttl) return cached.choice;
   321→   321→    probeCache.delete(folder);
   322→   322→  }
   323→   323→  let choice: HlsChoice = null;
   324→   324→  const qs = signedQuery.startsWith("?") ? signedQuery : `?${signedQuery}`;
   325→   325→  if (await manifestOk(`https://${PROXY_HOST}/${cfHost}/${folder}/hls/${PROXY_RENDITION}${qs}`)) {
   326→   326→    choice = { via: "proxy", rendition: PROXY_RENDITION };
   327→   327→  } else if (await manifestOk(`https://${cfHost}/${folder}/hls/${PROXY_RENDITION}${qs}`)) {
   328→   328→    choice = { via: "direct", rendition: PROXY_RENDITION };
   329→   329→  }
   330→   330→  probeCache.set(folder, { choice, at: Date.now() });
   331→   331→  return choice;
   332→   332→}

   334→   334→/** Same-origin relay URL for the chosen HLS rendition (stateless signature). */
   335→   335→function hlsRelayUrl(choice: NonNullable<HlsChoice>, cfHost: string, folder: string, signedQuery: string): string {
   336→   336→  const sigSeg = sigPathSegment(signedQuery);
   337→   337→  if (choice.via === "proxy") {
   338→   338→    return `/api/pw/x/${cfHost}/${folder}/s/${sigSeg}/hls/${choice.rendition}`;
   339→   339→  }
   340→   340→  return `/api/pw/${folder}/s/${sigSeg}/hls/${choice.rendition}`;
   341→   341→}
   342→   342→
   343→   343→/**
   344→   344→ * Prefer the proxy HLS lecture stream for signed CloudFront VOD. Returns null
   345→   345→ * when the stream is not eligible (no signature / non-uuid folder / no
   346→   346→ * rendition upstream) so the caller can fall back to its legacy flow.
   347→   347→ */
   348→   348→async function preferProxyHls(url: string, signedUrl: string): Promise<ResolvedVideoPayload | null> {
   349→   349→  if (!signedUrl) return null;
   350→   350→  let cfHost = "";
   351→   351→  let folder = "";
   352→   352→  try {
   353→   353→    const u = new URL(url);
   354→   354→    if (u.host === PROXY_HOST) {
   355→   355→      // Upstream already hands out proxy-form urls — unwrap cf_host/folder.
   356→   356→      const parts = u.pathname.split("/").filter(Boolean);
   357→   357→      cfHost = parts[0] || "";
   358→   358→      folder = parts[1] || "";
   359→   359→    } else {
   360→   360→      cfHost = u.host;
   361→   361→      folder = folderFromUrl(url);
   362→   362→    }
   363→   363→  } catch {
   364→   364→    return null;
   365→   365→  }
   366→   366→  if (!cfHost || !isAllowedMediaHost(cfHost) || !UUID_FOLDER_RE.test(folder)) return null;
   367→   367→  registerSignedFolder(folder, signedUrl);
   368→   368→  const choice = await chooseHlsUpstream(cfHost, folder, signedUrl);
   369→   369→  if (!choice) return null;
   370→   370→  return {
   371→   371→    kind: "hls",
   372→   372→    url: hlsRelayUrl(choice, cfHost, folder, signedUrl),
   373→   373→    signedUrl: "",
   374→   374→    clearKeys: null,
   375→   375→    poster: "",
   376→   376→    drmDetails: null,
   377→   377→  };
   378→   378→}

   380→   380→/** Resolve a video → same-origin playable payload (dash/hls/youtube/mp4). */
   381→   381→async function resolveVideoPayload(childId: string, batchId: string, subjectId: string) {
   382→   382→  const vid = await lxVideoUrl(batchId, subjectId, childId);
   383→   383→  if (vid?.url) {
   384→   384→    const url = vid.url.trim();
   385→   385→    if (/\.mpd(\?|$)/i.test(url)) {
   386→   386→      // Preferred: proxy.studyparcham.in HLS-720 lecture stream.
   387→   387→      const hls = await preferProxyHls(url, vid.signedUrl || "");
   388→   388→      if (hls) return hls;
   389→   389→      const folder = folderFromUrl(url);
   390→   390→      const sig = sigPathSegment(vid.signedUrl || "");
   391→   391→      if (folder && sig) registerSignedFolder(folder, vid.signedUrl || "");
   392→   392→      let clearKeys: Record<string, string> | null = null;
   393→   393→      try {
   394→   394→        const mpdUrl = `${url}${vid.signedUrl || ""}`;
   395→   395→        const controller = new AbortController();
   396→   396→        const timer = setTimeout(() => controller.abort(), 30000);
   397→   397→        const mpdRes = await fetch(mpdUrl, {
   398→   398→          cache: "no-store",
   399→   399→          signal: controller.signal,
   400→   400→          headers: { "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64)" },
   401→   401→        });
   402→   402→        clearTimeout(timer);
   403→   403→        if (mpdRes.ok) {
   404→   404→          const text = await mpdRes.text();
   405→   405→          const kidMatch = text.match(/default_KID="([0-9a-fA-F-]+)"/);
   406→   406→          if (kidMatch) {
   407→   407→            clearKeys = await lxOtp(kidMatch[1]);
   408→   408→          }
   409→   409→        }
   410→   410→      } catch {
   411→   411→        /* keys unavailable — player will surface a decrypt error */
   412→   412→      }
   413→   413→      // Stateless relay URL: relative DASH segments inherit the signature.
   414→   414→      const manifestUrl = folder
   415→   415→        ? `/api/pw/${folder}${sig ? `/s/${sig}` : ""}/master.mpd`
   416→   416→        : "";
   417→   417→      if (!manifestUrl) return null;
   418→   418→      return {
   419→   419→        kind: "dash",
   420→   420→        url: manifestUrl,
   421→   421→        signedUrl: "",
   422→   422→        clearKeys,
   423→   423→        poster: "",
   424→   424→        drmDetails: null,
   425→   425→      };
   426→   426→    }
   427→   427→    if (/\.m3u8(\?|$)/i.test(url)) {
   428→   428→      const folder = folderFromUrl(url);
   429→   429→      const sig = sigPathSegment(vid.signedUrl || "");
   430→   430→      if (!folder) return null;
   431→   431→      if (sig) registerSignedFolder(folder, vid.signedUrl || "");
   432→   432→      // Root-level LIVE manifest (…/index.m3u8?Signature=… — no uuid folder):
   433→   433→      // each live class gets its own channel like this, on its own live host.
   434→   434→      // The host is embedded in the relay path (validated against an
   435→   435→      // allowlist in pw-cdn); the manifest itself is the relayed asset and
   436→   436→      // segments resolve as its siblings through the same /s/<sig>/ path.
   437→   437→      const isUuidFolder = /^[a-f0-9-]{16,}$/i.test(folder);
   438→   438→      if (!isUuidFolder && /^[A-Za-z0-9._-]+\.m3u8$/i.test(folder)) {
   439→   439→        const liveHost = hostFromUrl(url);
   440→   440→        return {
   441→   441→          kind: "hls",
   442→   442→          url: `/api/pw/h/${liveHost}/${folder}${sig ? `/s/${sig}` : ""}`,
   443→   443→          signedUrl: "",
   444→   444→          clearKeys: null,
   445→   445→          poster: "",
   446→   446→          drmDetails: null,
   447→   447→        };
   448→   448→      }
   449→   449→      // VOD m3u8 inside a uuid folder — prefer the proxy HLS-720 stream too.
   450→   450→      const hls = await preferProxyHls(url, vid.signedUrl || "");
   451→   451→      if (hls) return hls;
   452→   452→      return {
   453→   453→        kind: "hls",
   454→   454→        url: `/api/pw/${folder}${sig ? `/s/${sig}` : ""}/master.m3u8`,
   455→   455→        signedUrl: "",
   456→   456→        clearKeys: null,
   457→   457→        poster: "",
   458→   458→        drmDetails: null,
   459→   459→      };
   460→   460→    }
   461→   461→    if (/youtube\.com|youtu\.be/i.test(url)) {
   462→   462→      return { kind: "youtube", url, signedUrl: "", clearKeys: null, poster: "", drmDetails: null };
   463→   463→    }
   464→   464→    // plain file (mp4 etc.) inside the signed folder
   465→   465→    const folder = folderFromUrl(url);
   466→   466→    const sig = sigPathSegment(vid.signedUrl || "");
   467→   467→    if (folder) {
   468→   468→      if (sig) registerSignedFolder(folder, vid.signedUrl || "");
   469→   469→      const fileName = url.split("/").pop() || "video.mp4";
   470→   470→      return {
   471→   471→        kind: "mp4",
   472→   472→        url: `/api/pw/${folder}${sig ? `/s/${sig}` : ""}/${fileName}`,
   473→   473→        signedUrl: "",
   474→   474→        clearKeys: null,
   475→   475→        poster: "",
   476→   476→        drmDetails: null,
   477→   477→      };
   478→   478→    }
   479→   479→    return { kind: "mp4", url, signedUrl: "", clearKeys: null, poster: "", drmDetails: null };
   480→   480→  }
   481→   481→
   482→   482→  // No signed stream — maybe a youtube lecture.
   483→   483→  if (subjectId) {
   484→   484→    const detail = await lxScheduleDetail(batchId, subjectId, childId);
   485→   485→    const url = detail?.url || "";
   486→   486→    if (url && /youtube\.com|youtu\.be/i.test(url)) {
   487→   487→      return { kind: "youtube", url, signedUrl: "", clearKeys: null, poster: "", drmDetails: null };
   488→   488→    }
   489→   489→  }
   490→   490→  return null;
   491→   491→}

   880→   880→      { success: false, error: error instanceof Error ? error.message : String(error) },
   881→   881→      { status: 500, headers: CORS_HEADERS },
   882→   882→    );
   883→   883→  }

   886→   886→/** Flatten test ids from scalars / arrays / objects ({testId}|{_id}|{id}). */
   887→   887→function collectTestIds(source: unknown, out: string[], seen: Set<string>): void {
   888→   888→  if (source === undefined || source === null) return;
   889→   889→  if (Array.isArray(source)) {
   890→   890→    source.forEach((item) => collectTestIds(item, out, seen));
   891→   891→    return;
   892→   892→  }
   893→   893→  if (typeof source === "object") {
   894→   894→    const o = source as Record<string, unknown>;
   895→   895→    collectTestIds(o.testId ?? o._id ?? o.id, out, seen);
   896→   896→    return;
   897→   897→  }
   898→   898→  const s = String(source).trim();
   899→   899→  if (s && !seen.has(s)) {
   900→   900→    seen.add(s);
   901→   901→    out.push(s);
   902→   902→  }
   903→   903→}
