     1→'use client';
     2→
     3→// ============================================================================
     4→// Study Bison — main SPA (rebranded from the studybee pro clone)
     5→// Views: All Courses (home) / Study Dashboard / All Classes / Chapters /
     6→// Content / All Tests. Every content action is backed by the REAL PW data
     7→// relayed from the LearnxPW upstream (batches, subjects, chapters, lectures,
     8→// notes & DPP PDFs, tests, live schedule).
     9→// ============================================================================
    10→
    11→import { useCallback, useEffect, useRef, useState } from 'react';
    12→import {
    13→  api,
    14→  fetchCatalog,
    15→  imageUrl,
    16→  BatchCatalogItem,
    17→  BatchDetailPublic,
    18→  BatchSubjectPublic,
    19→  ChapterItemPublic,
    20→  ContentNodePublic,
    21→  ScheduleEventPublic,
    22→  TestItemPublic,
    23→} from '@/lib/api-client';
    24→import { addXP, useEnrolledBatches, useXP, useWatchedVideos } from '@/lib/local-state';
    25→
    26→type ViewId = 'home-view' | 'study-view' | 'all-classes-view' | 'chapters-view' | 'content-view' | 'all-tests-view';
    27→
    28→interface SubjectRef { id: string; internalId: string; name: string; masterId: string; batchSubjectId: string }
    29→interface TopicRef { id: string; name: string }
    30→interface AttachmentRow { title: string; url: string }
    31→interface DppMeta {
    32→  testId: string;
    33→  scheduleId: string;
    34→  subjectId: string;
    35→  name: string;
    36→}
    37→
    38→const DEFAULT_THUMB = 'https://store.pw.live/footer-logo.svg';
    39→
    40→export default function StudyBisonApp() {
    41→  // ---------------- global nav ----------------
    42→  const [view, setView] = useState<ViewId>('home-view');
    43→  const [sidebarOpen, setSidebarOpen] = useState(false);
    44→
    45→  // ---------------- home / catalog ----------------
    46→  const [batches, setBatches] = useState<BatchCatalogItem[]>([]);
    47→  const [page, setPage] = useState(1);
    48→  const [hasMore, setHasMore] = useState(false);
    49→  const [searchTerm, setSearchTerm] = useState('');
    50→  const [searching, setSearching] = useState(false);
    51→  const [catalogError, setCatalogError] = useState('');
    52→  const searchTimeout = useRef<ReturnType<typeof setTimeout> | null>(null);
    53→  const debounceTerm = useRef('');
    54→
    55→  // ---------------- enrollment ----------------
    56→  const { enrolled, enroll, unenroll } = useEnrolledBatches();
    57→  const { watched, toggleWatched } = useWatchedVideos();
    58→  const xp = useXP();
    59→  const batchDetailCache = useRef<Map<string, BatchDetailPublic>>(new Map());
    60→
    61→  // ---------------- study ----------------
    62→  const [currentBatchId, setCurrentBatchId] = useState<string | null>(null);
    63→  const [enrolledDetails, setEnrolledDetails] = useState<BatchDetailPublic[]>([]);
    64→  const [events, setEvents] = useState<ScheduleEventPublic[]>([]);
    65→  const [eventsLoading, setEventsLoading] = useState(false);
    66→
    67→  // ---------------- classes / chapters / content ----------------
    68→  const [subjects, setSubjects] = useState<BatchSubjectPublic[]>([]);
    69→  const [subjectsLoading, setSubjectsLoading] = useState(false);
    70→  const [currentSubject, setCurrentSubject] = useState<SubjectRef | null>(null);
    71→  const [chaptersTab, setChaptersTab] = useState<'UNITS' | 'STUDY_MATERIAL'>('UNITS');
    72→  const [chapters, setChapters] = useState<ChapterItemPublic[]>([]);
    73→  const [chaptersLoading, setChaptersLoading] = useState(false);
    74→  const [currentTopic, setCurrentTopic] = useState<TopicRef | null>(null);
    75→  const [contentTab, setContentTab] = useState('ALL');
    76→  const [contentNodes, setContentNodes] = useState<ContentNodePublic[]>([]);
    77→  const [dppItems, setDppItems] = useState<TestItemPublic[]>([]);
    78→  const [contentLoading, setContentLoading] = useState(false);
    79→
    80→  // ---------------- tests ----------------
    81→  const [tests, setTests] = useState<TestItemPublic[]>([]);
    82→  const [testsLoading, setTestsLoading] = useState(false);
    83→
    84→  // ---------------- resources tab ----------------
    85→  const [resourcesTab, setResourcesTab] = useState<'subjects' | 'resources'>('subjects');
    86→  const [resourceTopics, setResourceTopics] = useState<ChapterItemPublic[]>([]);
    87→  const [resourcesLoading, setResourcesLoading] = useState(false);
    88→
    89→  // ---------------- modals ----------------
    90→  const [batchInfoOpen, setBatchInfoOpen] = useState(false);
    91→  const [batchInfo, setBatchInfo] = useState<BatchDetailPublic | null>(null);
    92→  const [batchInfoLoading, setBatchInfoLoading] = useState(false);
    93→
    94→  const [notesOpen, setNotesOpen] = useState(false);
    95→  const [notesLoading, setNotesLoading] = useState(false);
    96→  const [noteRows, setNoteRows] = useState<AttachmentRow[]>([]);
    97→
    98→  const [dppOpen, setDppOpen] = useState(false);
    99→  const [dppMeta, setDppMeta] = useState<DppMeta | null>(null);
   100→
   101→  const [testFrame, setTestFrame] = useState<{ open: boolean; url: string; title: string }>({ open: false, url: '', title: 'Test Attempt' });
   102→
   103→  const [instructions, setInstructions] = useState<{ open: boolean; loading: boolean; title: string; html: string }>({ open: false, loading: false, title: '', html: '' });
   104→
   105→  // ==================================================================
   106→  // NAV
   107→  // ==================================================================
   108→  const nav = useCallback((v: ViewId) => {
   109→    setView(v);
   110→    setSidebarOpen(false);
   111→    window.scrollTo(0, 0);
   112→  }, []);
   113→
   114→  // ==================================================================
   115→  // CATALOG (home view)
   116→  // ==================================================================
   117→  const loadCatalog = useCallback(async (q: string, p: number, isNew: boolean) => {
   118→    try {
   119→      if (isNew) {
   120→        setSearching(true);
   121→        setBatches([]);
   122→      }
   123→      setCatalogError('');
   124→      const res = await fetchCatalog(q, p, 20);
   125→      setBatches((prev) => (isNew ? res.data : [...prev, ...res.data]));
   126→      setHasMore(res.hasMore);
   127→      setPage(p);
   128→    } catch {
   129→      setCatalogError('Failed to load batches.');
   130→    } finally {
   131→      setSearching(false);
   132→    }
   133→  }, []);
   134→
   135→  useEffect(() => {
   136→    loadCatalog('', 1, true);
   137→  }, [loadCatalog]);
   138→
   139→  const handleSearchInput = (value: string) => {
   140→    setSearchTerm(value);
   141→    debounceTerm.current = value;
   142→    if (searchTimeout.current) clearTimeout(searchTimeout.current);
   143→    searchTimeout.current = setTimeout(() => {
   144→      loadCatalog(debounceTerm.current, 1, true);
   145→    }, 500);
   146→  };
   147→
   148→  const loadNextPage = () => {
   149→    loadCatalog(searchTerm, page + 1, false);
   150→  };
   151→
   152→  /** Catalog card click — open the real-data batch detail sheet. */
   153→  const openCatalogInfo = async (b: BatchCatalogItem) => {
   154→    setBatchInfoOpen(true);
   155→    setBatchInfoLoading(true);
   156→    setBatchInfo(b);
   157→    const res = await api.batchDetail(b._id);
   158→    if (res.success && res.data) setBatchInfo(res.data);
   159→    setBatchInfoLoading(false);
   160→  };
   161→
   162→  const enrollInBatch = (batchId: string) => {
   163→    enroll(batchId);
   164→    setCurrentBatchId(batchId);
   165→    setBatchInfoOpen(false);
   166→    nav('study-view');
   167→  };
   168→
   169→  const studyBatch = (batchId: string) => {
   170→    setBatchInfoOpen(false);
   171→    setCurrentBatchId(batchId);
   172→    nav('study-view');
   173→  };
   174→
   175→  // ==================================================================
   176→  // STUDY DASHBOARD
   177→  // ==================================================================
   178→  const refreshEnrolledDetails = useCallback(async () => {
   179→    const results: BatchDetailPublic[] = [];
   180→    for (const id of enrolled) {
   181→      const cached = batchDetailCache.current.get(id);
   182→      if (cached) {
   183→        results.push(cached);
   184→        continue;
   185→      }
   186→      const res = await api.batchDetail(id);
   187→      if (res.success && res.data) {
   188→        batchDetailCache.current.set(id, res.data);
   189→        results.push(res.data);
   190→      }
   191→    }
   192→    setEnrolledDetails(results);
   193→    setCurrentBatchId((cur) => {
   194→      if (cur && results.some((r) => r._id === cur)) return cur;
   195→      return results.length > 0 ? results[0]._id : null;
   196→    });
   197→  }, [enrolled]);
   198→
   199→  useEffect(() => {
   200→    refreshEnrolledDetails();
   201→  }, [refreshEnrolledDetails]);
   202→
   203→  const loadEvents = useCallback(async (batchId: string) => {
   204→    setEventsLoading(true);
   205→    const res = await api.todaySchedule(batchId);
   206→    setEvents(res.success && res.data ? res.data : []);
   207→    setEventsLoading(false);
   208→  }, []);
   209→
   210→  useEffect(() => {
   211→    if (currentBatchId) loadEvents(currentBatchId);
   212→    else setEvents([]);
   213→  }, [currentBatchId, loadEvents]);
   214→
   215→  const currentBatch = enrolledDetails.find((b) => b._id === currentBatchId) || null;
   216→
   217→  // ==================================================================
   218→  // ALL CLASSES — SUBJECTS & RESOURCES
   219→  // ==================================================================
   220→  const loadSubjects = useCallback(async (batchId: string) => {
   221→    setSubjectsLoading(true);
   222→    // Batch detail carries the real subject mapping (slug-addressed).
   223→    const res = await api.batchDetail(batchId);
   224→    if (res.success && res.data) {
   225→      batchDetailCache.current.set(batchId, res.data);
   226→      setSubjects(res.data.subjects || []);
   227→    } else {
   228→      setSubjects([]);
   229→    }
   230→    setSubjectsLoading(false);
   231→  }, []);
   232→
   233→  const loadResources = useCallback(async (batchId: string) => {
   234→    setResourcesLoading(true);
   235→    const res = await api.resourceTopics(batchId, '');
   236→    const topics = res.success && res.data ? (res.data as unknown as { data?: ChapterItemPublic[] }).data || [] : [];
   237→    setResourceTopics(topics);
   238→    setResourcesLoading(false);
   239→  }, []);
   240→
   241→  const openAllClasses = () => {
   242→    setResourcesTab('subjects');
   243→    nav('all-classes-view');
   244→    if (currentBatchId) loadSubjects(currentBatchId);
   245→  };
   246→
   247→  const switchClassesTab = (tab: 'subjects' | 'resources') => {
   248→    setResourcesTab(tab);
   249→    if (tab === 'subjects' && currentBatchId) loadSubjects(currentBatchId);
   250→    if (tab === 'resources' && currentBatchId) loadResources(currentBatchId);
   251→  };
   252→
   253→  // ==================================================================
   254→  // CHAPTERS
   255→  // ==================================================================
   256→  const loadChaptersList = useCallback(async (subjectId: string, tagType: 'UNITS' | 'STUDY_MATERIAL') => {
   257→    if (!currentBatchId) return;
   258→    setChaptersLoading(true);
   259→    const res = await api.subjectTopics(currentBatchId, subjectId, tagType);
   260→    const list = res.success && res.data ? (res.data as unknown as { data?: ChapterItemPublic[] }).data || [] : [];
   261→    setChapters(list);
   262→    setChaptersLoading(false);
   263→  }, [currentBatchId]);
   264→
   265→  const openSubject = (sub: BatchSubjectPublic) => {
   266→    setCurrentSubject({
   267→      id: sub.subjectId || sub._id,
   268→      internalId: sub.subjectId || sub._id,
   269→      name: sub.subject,
   270→      masterId: sub.masterId || '',
   271→      batchSubjectId: sub.batchSubjectId || '',
   272→    });
   273→    setChaptersTab('UNITS');
   274→    nav('chapters-view');
   275→    loadChaptersList(sub.subjectId || sub._id, 'UNITS');
   276→  };
   277→
   278→  const switchChaptersTab = (tagType: 'UNITS' | 'STUDY_MATERIAL') => {
   279→    setChaptersTab(tagType);
   280→    if (currentSubject) loadChaptersList(currentSubject.id, tagType);
   281→  };
   282→
   283→  // ==================================================================
   284→  // CONTENT
   285→  // ==================================================================
   286→  const loadContent = useCallback(async (subjectId: string, tagId: string, filter: string) => {
   287→    if (!currentBatchId) return;
   288→    setContentLoading(true);
   289→    if (filter === 'DPP_TEST') {
   290→      const subject = currentSubject;
   291→      const res = await api.dppList(currentBatchId, subject?.batchSubjectId || '', tagId);
   292→      const list = res.success && res.data ? (res.data as unknown as { data?: TestItemPublic[] }).data || [] : [];
   293→      setDppItems(list);
   294→      setContentNodes([]);
   295→    } else {
   296→      const res = await api.scheduleContent(currentBatchId, subjectId, tagId, filter);
   297→      const nodes = res.success && res.data ? (res.data as ContentNodePublic[]) : [];
   298→      setContentNodes(nodes);
   299→      setDppItems([]);
   300→    }
   301→    setContentLoading(false);
   302→  }, [currentBatchId, currentSubject]);
   303→
   304→  const openTopic = (t: ChapterItemPublic) => {
   305→    setCurrentTopic({ id: t._id, name: t.name });
   306→    setContentTab('ALL');
   307→    nav('content-view');
   308→    if (currentSubject) loadContent(currentSubject.id, t._id, 'ALL');
   309→  };
   310→
   311→  const openResourceTopic = (t: ChapterItemPublic) => {
   312→    if (!currentSubject) return;
   313→    setCurrentTopic({ id: t._id, name: t.name });
   314→    setContentTab('ALL');
   315→    nav('content-view');
   316→    loadContent(currentSubject.id, t._id, 'ALL');
   317→  };
   318→
   319→  const filterContentTab = (filter: string) => {
   320→    setContentTab(filter);
   321→    if (currentSubject && currentTopic) loadContent(currentSubject.id, currentTopic.id, filter);
   322→  };
   323→
   324→  // ==================================================================
   325→  // TESTS
   326→  // ==================================================================
   327→  const openAllTests = () => {
   328→    nav('all-tests-view');
   329→    if (!currentBatchId) return;
   330→    setTestsLoading(true);
   331→    api.testList(currentBatchId).then((res) => {
   332→      setTests(res.success && res.data ? res.data : []);
   333→      setTestsLoading(false);
   334→    }).catch(() => {
   335→      setTests([]);
   336→      setTestsLoading(false);
   337→    });
   338→  };
   339→
   340→  const openInstructions = async (t: TestItemPublic) => {
   341→    if (!currentBatchId) return;
   342→    setInstructions({ open: true, loading: true, title: t.name || 'Test', html: '' });
   343→    const res = await api.testDetail(t._id, currentBatchId);
   344→    const d = (res.data || {}) as Record<string, unknown>;
   345→    // upstream instructions arrive as {en: "<html>"} or a plain html string
   346→    const pick = (v: unknown): string => {
   347→      if (typeof v === 'string') return v;
   348→      if (v && typeof v === 'object') return String((v as { en?: string }).en || '');
   349→      return '';
   350→    };
   351→    const raw = pick(d.multiGeneralInstructions) || pick(d.generalInstructions);
   352→    const html = raw.replace(/<script[\s\S]*?<\/script>/gi, '');
   353→    setInstructions({ open: true, loading: false, title: t.name || 'Test', html: html || '<p>No instructions available.</p>' });
   354→  };
   355→
   356→  const launchQuiz = (p: { testId: string; scheduleId?: string; subjectId?: string; title: string }) => {
   357→    if (!currentBatchId) return;
   358→    const q = new URLSearchParams({
   359→      testId: p.testId,
   360→      batchId: currentBatchId,
   361→      scheduleId: p.scheduleId || '',
   362→      subjectId: p.subjectId || currentSubject?.masterId || '',
   363→      type: 'Start',
   364→    });
   365→    addXP(10, 'Attempted Test/DPP');
   366→    setDppOpen(false);
   367→    setTestFrame({ open: true, url: `/quiz?${q.toString()}`, title: p.title || 'Test Attempt' });
   368→  };
   369→
   370→  // ==================================================================
   371→  // MODALS
   372→  // ==================================================================
   373→  const openBatchInfo = async () => {
   374→    if (!currentBatchId) return;
   375→    setBatchInfoOpen(true);
   376→    setBatchInfoLoading(true);
   377→    const res = await api.batchDetail(currentBatchId);
   378→    setBatchInfo(res.success && res.data ? res.data : null);
   379→    setBatchInfoLoading(false);
   380→  };
   381→
   382→  const openNotesModal = async (scheduleId: string, subjectMasterId: string) => {
   383→    if (!currentBatchId) return;
   384→    setNotesOpen(true);
   385→    setNotesLoading(true);
   386→    setNoteRows([]);
   387→    const res = await api.scheduleDetail(currentBatchId, scheduleId, subjectMasterId);
   388→    const item = res.data as {
   389→      topic?: string;
   390→      name?: string;
   391→      homeworkIds?: Array<{ topic?: string; attachmentIds?: Array<{ baseUrl?: string; key?: string }> }>;
   392→    } | undefined;
   393→
   394→    const rows: AttachmentRow[] = [];
   395→    if (item) {
   396→      (item.homeworkIds || []).forEach((hw) => {
   397→        (hw.attachmentIds || []).forEach((att) => {
   398→          rows.push({ title: hw.topic || 'Class Note', url: imageUrl(att) });
   399→        });
   400→      });
   401→    }
   402→    setNoteRows(rows);
   403→    setNotesLoading(false);
   404→  };
   405→
   406→  const handleUnenroll = (batchId: string) => {
   407→    if (!window.confirm('Are you sure you want to unenroll from this batch?')) return;
   408→    unenroll(batchId);
   409→    setBatchInfoOpen(false);
   410→    if (currentBatchId === batchId) {
   411→      const remaining = enrolled.filter((id) => id !== batchId);
   412→      if (remaining.length > 0) setCurrentBatchId(remaining[0]);
   413→      else {
   414→        setCurrentBatchId(null);
   415→        nav('home-view');
   416→      }
   417→    }
   418→  };
   419→
   420→  // ==================================================================
   421→  // RENDER HELPERS
   422→  // ==================================================================
   423→  const renderLectureCard = (node: ContentNodePublic) => {
   424→    const d = node.data;
   425→    const thumb = d.videoDetails?.image || DEFAULT_THUMB;
   426→    const title = d.topic || d.name || 'Lecture';
   427→    const date = d.date ? new Date(d.date).toLocaleDateString('en-GB', { day: 'numeric', month: 'short', year: 'numeric' }) : '';
   428→    const vId = d._id || d.videoDetails?._id || node._id;
   429→    const master = currentSubject?.masterId || currentSubject?.id || '';
   430→    const isWatched = watched.includes(vId);
   431→    const bName = currentBatch?.name || 'PW Batch';
   432→    const base = node.type === 'DPP_VIDEOS' ? '/player' : '/player';
   433→    const playerUrl = `${base}?videoId=${encodeURIComponent(vId)}&batchId=${encodeURIComponent(currentBatchId || '')}&subjectId=${encodeURIComponent(master)}&schId=${encodeURIComponent(node._id)}&title=${encodeURIComponent(title)}&batchName=${encodeURIComponent(bName)}`;
   434→    const hasAttachments = d.homeworkIds && d.homeworkIds.length > 0;
   435→
   436→    return (
   437→      <div className="content-card" key={node._id} style={{ flexDirection: 'column', alignItems: 'stretch' }}>
   438→        <div className="d-flex gap-3 mb-3">
   439→          <img src={thumb} className="lecture-thumb" alt={title} />
   440→          <div className="flex-grow-1">
   441→            <span style={{ background: '#eff6ff', color: '#2563eb', fontSize: '0.7rem', fontWeight: 800, padding: '2px 6px', borderRadius: 4, display: 'inline-block', marginBottom: 5 }}>
   442→              {node.type === 'DPP_VIDEOS' ? 'DPP VIDEO' : 'VIDEO'}
   443→            </span>
   444→            <div style={{ fontWeight: 700, fontSize: '1rem', color: 'var(--sb-text-dark)' }}>{title}</div>
   445→            <div style={{ color: 'var(--sb-text-muted)', fontSize: '0.8rem', marginTop: 5 }}>
   446→              <i className="fas fa-clock"></i> {date}
   447→            </div>
   448→          </div>
   449→        </div>
   450→        <div className="d-flex gap-2">
   451→          <button
   452→            className={`btn-check-done ${isWatched ? 'active' : ''}`}
   453→            title="Mark as Watched"
   454→            onClick={() => {
   455→              const added = toggleWatched(vId);
   456→              if (added) addXP(10, 'Watched Video');
   457→            }}
   458→          >
   459→            <i className="fas fa-check"></i>
   460→          </button>
   461→          <button className="btn-play" style={{ flexGrow: 1 }} onClick={() => window.open(playerUrl, '_blank')}>
   462→            <i className="fas fa-play"></i> Watch
   463→          </button>
   464→          {hasAttachments ? (
   465→            <button className="btn-notes" onClick={() => openNotesModal(node._id, master)}>
   466→              <i className="fas fa-file-alt"></i>
   467→            </button>
   468→          ) : null}
   469→        </div>
   470→      </div>
   471→    );
   472→  };
   473→
   474→  const renderPdfCard = (node: ContentNodePublic) => {
   475→    const d = node.data;
   476→    const title = d.topic || 'Document';
   477→    const att = d.homeworkIds?.[0]?.attachmentIds?.[0];
   478→    const url = imageUrl(att);
   479→    const isDpp = node.type === 'DPP_PDF';
   480→    const tagColor = isDpp ? '#d97706' : '#e11d48';
   481→    const tagBg = isDpp ? '#fef3c7' : '#fee2e2';
   482→    return (
   483→      <div className="content-card d-flex align-items-center justify-content-between" key={node._id} style={{ padding: 20 }}>
   484→        <div className="d-flex align-items-center gap-3" style={{ maxWidth: '75%' }}>
   485→          <div style={{ background: tagBg, color: tagColor, width: 40, height: 40, borderRadius: 8, display: 'flex', alignItems: 'center', justifyContent: 'center', fontSize: '1.2rem', flexShrink: 0 }}>
   486→            <i className="fas fa-file-pdf"></i>
   487→          </div>
   488→          <div style={{ fontWeight: 600, fontSize: '0.95rem', wordBreak: 'break-word' }}>{title}</div>
   489→        </div>
   490→        <button
   491→          className="btn btn-sb-dark"
   492→          style={{ padding: '8px 16px', borderRadius: 8, fontSize: '0.9rem', width: 'auto' }}
   493→          onClick={() => {
   494→            if (!url) return;
   495→            window.open(url, '_blank');
   496→            addXP(5, 'Viewed Document');
   497→          }}
   498→        >
   499→          <i className="fas fa-eye"></i> View
   500→        </button>
   501→      </div>
   502→    );
   503→  };
   504→
   505→  const renderDppItem = (t: TestItemPublic) => {
   506→    return (
   507→      <div className="content-card" key={t._id} style={{ flexDirection: 'column', alignItems: 'stretch' }}>
   508→        <div className="d-flex align-items-center gap-3 mb-3">
   509→          <div style={{ background: '#fef3c7', color: '#d97706', width: 45, height: 45, borderRadius: 8, display: 'flex', alignItems: 'center', justifyContent: 'center', fontSize: '1.2rem' }}>
   510→            <i className="fas fa-tasks"></i>
   511→          </div>
