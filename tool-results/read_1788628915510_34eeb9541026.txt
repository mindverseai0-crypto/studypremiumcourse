   124→  const loadCatalog = useCallback(async (q: string, p: number, isNew: boolean) => {
   125→    try {
   126→      if (isNew) {
   127→        setSearching(true);
   128→        setBatches([]);
   129→      }
   130→      setCatalogError('');
   131→      const res = await fetchCatalog(q, p, 20);
   132→      setBatches((prev) => (isNew ? res.data : [...prev, ...res.data]));
   133→      setHasMore(res.hasMore);
   134→      setPage(p);
   135→    } catch {
   136→      setCatalogError('Failed to load batches.');
   137→    } finally {
   138→      setSearching(false);
   139→    }
   140→  }, []);
   141→
   142→  useEffect(() => {
   143→    loadCatalog('', 1, true);
   144→  }, [loadCatalog]);
   145→
   146→  const handleSearchInput = (value: string) => {
   147→    setSearchTerm(value);
   148→    debounceTerm.current = value;
   149→    if (searchTimeout.current) clearTimeout(searchTimeout.current);
   150→    searchTimeout.current = setTimeout(() => {
   151→      loadCatalog(debounceTerm.current, 1, true);

   155→  const loadNextPage = () => {
   156→    loadCatalog(searchTerm, page + 1, false);

   159→  /** Catalog card click — open the real-data batch detail sheet. */
   160→  const openCatalogInfo = async (b: BatchCatalogItem) => {
   161→    setBatchInfoOpen(true);
   162→    setBatchInfoLoading(true);
   163→    setBatchInfo(b);
   164→    const res = await api.batchDetail(b._id);
   165→    if (res.success && res.data) setBatchInfo(res.data);
   166→    setBatchInfoLoading(false);
   167→  };
   168→
   169→  const enrollInBatch = (batchId: string) => {
   170→    enroll(batchId);
   171→    setCurrentBatchId(batchId);
   172→    setBatchInfoOpen(false);

   183→  // STUDY DASHBOARD
   184→  // ==================================================================
   185→  const refreshEnrolledDetails = useCallback(async () => {
   186→    const results: BatchDetailPublic[] = [];
   187→    for (const id of enrolled) {
   188→      const cached = batchDetailCache.current.get(id);
   189→      if (cached) {
   190→        results.push(cached);
   191→        continue;
   192→      }
   193→      const res = await api.batchDetail(id);
   194→      if (res.success && res.data) {
   195→        batchDetailCache.current.set(id, res.data);
   196→        results.push(res.data);
   197→      }
   198→    }
   199→    setEnrolledDetails(results);
   200→    setCurrentBatchId((cur) => {
   201→      if (cur && results.some((r) => r._id === cur)) return cur;
   202→      return results.length > 0 ? results[0]._id : null;
   203→    });
   204→  }, [enrolled]);
   205→
   206→  useEffect(() => {
   207→    refreshEnrolledDetails();
   208→  }, [refreshEnrolledDetails]);
   209→
   210→  const loadEvents = useCallback(async (batchId: string) => {
   211→    setEventsLoading(true);
   212→    const res = await api.todaySchedule(batchId);
   213→    setEvents(res.success && res.data ? res.data : []);
   214→    setEventsLoading(false);
   215→  }, []);
   216→
   217→  useEffect(() => {
   218→    if (currentBatchId) loadEvents(currentBatchId);
   219→    else setEvents([]);
   220→  }, [currentBatchId, loadEvents]);
   221→
   222→  const currentBatch = enrolledDetails.find((b) => b._id === currentBatchId) || null;
   223→

   225→  // ALL CLASSES — SUBJECTS & RESOURCES
   226→  // ==================================================================
   227→  const loadSubjects = useCallback(async (batchId: string) => {
   228→    setSubjectsLoading(true);
   229→    // Real catalog has no subject mapping — the API returns an empty list.
   230→    const res = await api.batchDetail(batchId);
   231→    if (res.success && res.data) {
   232→      batchDetailCache.current.set(batchId, res.data);
   233→    }
   234→    setSubjects([]);
   235→    setSubjectsLoading(false);
   236→  }, []);
   237→
   238→  const loadResources = useCallback(async (batchId: string) => {
   239→    setResourcesLoading(true);
   240→    const res = await api.resourceTopics(batchId, '');
   241→    const topics = res.success && res.data ? (res.data as unknown as { data?: ChapterItemPublic[] }).data || [] : [];
   242→    setResourceTopics(topics);
   243→    setResourcesLoading(false);
   244→  }, []);
   245→
   246→  const openAllClasses = () => {
   247→    setResourcesTab('subjects');
   248→    nav('all-classes-view');
   249→    if (currentBatchId) loadSubjects(currentBatchId);
   250→  };
   251→
   252→  const switchClassesTab = (tab: 'subjects' | 'resources') => {
   253→    setResourcesTab(tab);
   254→    if (tab === 'subjects' && currentBatchId) loadSubjects(currentBatchId);
   255→    if (tab === 'resources' && currentBatchId) loadResources(currentBatchId);

   259→  // CHAPTERS

   268→  }, [currentBatchId]);
   269→
   270→  const openSubject = (sub: BatchSubjectPublic) => {

   272→    setChaptersTab('UNITS');
   273→    nav('chapters-view');
   274→    loadChaptersList(sub._id, 'UNITS');
   275→  };
   276→
   277→  const switchChaptersTab = (tagType: 'UNITS' | 'STUDY_MATERIAL') => {
   278→    setChaptersTab(tagType);
   279→    if (currentSubject) loadChaptersList(currentSubject.id, tagType);
   280→  };
   281→
   282→  // ==================================================================
   283→  // CONTENT
   284→  // ==================================================================
   285→  const loadContent = useCallback(async (subjectId: string, tagId: string, filter: string) => {
   286→    if (!currentBatchId) return;
   287→    setContentLoading(true);
   288→    if (filter === 'DPP_TEST') {
   289→      const res = await api.dppList(currentBatchId, subjectId || '', tagId);
   290→      const list = res.success && res.data ? (res.data as unknown as { data?: TestItemPublic[] }).data || [] : [];

   292→      setContentNodes([]);
   293→    } else {
   294→      const res = await api.scheduleContent(currentBatchId, subjectId || '', tagId, filter);
   295→      const nodes = res.success && res.data ? (res.data as ContentNodePublic[]) : [];
   296→      setContentNodes(nodes);
   297→      setDppItems([]);
   298→    }
   299→    setContentLoading(false);
   300→  }, [currentBatchId]);
   301→
   302→  const openTopic = (t: ChapterItemPublic) => {
   303→    setCurrentTopic({ id: t._id, name: t.name });
   304→    setContentTab('ALL');
   305→    nav('content-view');
   306→    if (currentSubject) loadContent(currentSubject.id, t._id, 'ALL');
   307→  };
   308→
   309→  const openResourceTopic = (t: ChapterItemPublic) => {
   310→    setCurrentSubject({ id: t.typeId, internalId: t.typeId, name: 'Resources' });
   311→    setCurrentTopic({ id: t._id, name: t.name });
   312→    setContentTab('ALL');
   313→    nav('content-view');
   314→    loadContent(t.typeId, t._id, 'ALL');

   316→

   338→  const launchTestGateway = async (testId: string) => {
   339→    setTestPreparing(testId);
   340→    const res = await api.testMap(testId);
   341→    let actionType = 'Start';
   342→    let testMapId = '';
   343→    if (res.success && res.data && res.data.length > 0) {
   344→      actionType = res.data[0].tag2 || 'Start';
   345→      testMapId = res.data[0].testStudentMapping?._id || '';
   346→    }
   347→    const url = `/quiz?testId=${testId}&batchId=${currentBatchId || ''}&mode=TEST&type=${actionType}&mapId=${testMapId}`;

   364→  const openNotesModal = async (scheduleId: string) => {
   365→    if (!currentBatchId) return;
   366→    setNotesOpen(true);
   367→    setNotesLoading(true);
   368→    setNoteRows([]);
   369→    setNoteDppRows([]);
   370→    const res = await api.scheduleDetail(currentBatchId, scheduleId);
   371→    const item = res.data as {
   372→      topic?: string;
   373→      name?: string;
   374→      attachmentIds?: Array<{ baseUrl?: string; key?: string }>;
   375→      homeworkIds?: Array<{ topic?: string; attachmentIds?: Array<{ baseUrl?: string; key?: string }> }>;
   376→      exerciseIds?: Array<Record<string, unknown>>;
   377→    } | undefined;
   378→
   379→    const rows: AttachmentRow[] = [];
   380→    if (item) {
   381→      (item.homeworkIds || []).forEach((hw) => {
   382→        (hw.attachmentIds || []).forEach((att) => {
   383→          rows.push({ title: hw.topic || 'Class Note', url: imageUrl(att) });
   384→        });
   385→      });
   386→      (item.attachmentIds || []).forEach((att) => {
   387→        rows.push({ title: item.topic || item.name || 'Document', url: imageUrl(att) });
   388→      });
   389→    }
   390→    setNoteRows(rows);
   391→
   392→    // DPP matches from exercises
   393→    const testRes = await api.testList(currentBatchId);
   394→    const testList = testRes.success && testRes.data ? testRes.data : [];
   395→    const dppRows: DppRow[] = [];
   396→    (item?.exerciseIds || []).forEach((ex) => {
   397→      const exDppId = String((ex as { dppId?: string }).dppId || (ex as { _id?: string })._id || '');
   398→      const match = testList.find((t) => t.dppQuizDetails && t.dppQuizDetails.contentId === exDppId);
   399→      if (match && match.dppQuizDetails) {
   400→        const status = String((match.dppQuizDetails.testStudentMapping as { testActivityStatus?: string })?.testActivityStatus || '');
   401→        dppRows.push({
   402→          testId: match.dppQuizDetails.test._id,
   403→          scheduleId: match.dppQuizDetails.scheduleId,
   404→          name: (ex as { title?: string }).title || match.dppQuizDetails.test.name,
   405→          isAttempted: status === 'Finished' || status === 'COMPLETED',
   406→          actionType: status === 'Finished' ? 'Reattempt' : status === 'Started' ? 'Resume' : 'Start',
   407→          testMapId: '',
   408→        });
   409→      }
   410→    });
   411→    setNoteDppRows(dppRows);
   412→    setNotesLoading(false);
   413→  };
   414→
   415→  const openDirectPdf = async (scheduleId: string) => {
   416→    if (!currentBatchId) return;
   417→    const res = await api.scheduleDetail(currentBatchId, scheduleId);
   418→    const item = res.data as {
   419→      topic?: string;
   420→      name?: string;
   421→      attachmentIds?: Array<{ baseUrl?: string; key?: string }>;
   422→      homeworkIds?: Array<{ topic?: string; attachmentIds?: Array<{ baseUrl?: string; key?: string }> }>;
   423→      exerciseIds?: Array<{ title?: string; attachmentIds?: Array<{ baseUrl?: string; key?: string }> }>;
   424→    } | undefined;
   425→    const rows: AttachmentRow[] = [];
   426→    if (item) {
   427→      (item.attachmentIds || []).forEach((att) => rows.push({ title: item.topic || item.name || 'Document', url: imageUrl(att) }));
   428→      (item.homeworkIds || []).forEach((hw) => (hw.attachmentIds || []).forEach((att) => rows.push({ title: hw.topic || 'Class Note', url: imageUrl(att) })));
   429→      (item.exerciseIds || []).forEach((ex) => (ex.attachmentIds || []).forEach((att) => rows.push({ title: ex.title || 'DPP', url: imageUrl(att) })));
   430→    }
   431→    const deduped = rows.filter((r, i, arr) => arr.findIndex((x) => x.url === r.url) === i);
   432→    if (deduped.length === 0) return;
   433→    if (deduped.length === 1) {
   434→      window.open(deduped[0].url, '_blank');
   435→      addXP(5, 'Viewed Document');
   436→    } else {
   437→      setNoteRows(deduped);
   438→      setNoteDppRows([]);
   439→      setNotesOpen(true);
   440→    }
   441→  };
   442→
   443→  const openDppModal = (meta: DppMeta) => {
   444→    setDppMeta(meta);
   445→    setDppOpen(true);
   446→  };
   447→
   448→  const launchDpp = (mode: 'BATCH_QUIZ' | 'DPP_QUIZ_MODE') => {
   449→    if (!dppMeta || !currentBatchId) return;

   451→    const url = `/quiz?testId=${dppMeta.testId}&batchId=${currentBatchId}&mode=${mode}&scheduleId=${dppMeta.scheduleId}&subjectId=${subjectId}&type=${dppMeta.actionType}`;
   452→    addXP(10, 'Attempted Test/DPP');
   453→    setDppOpen(false);
   454→    setTestFrame({ open: true, url, title: 'DPP Attempt' });
   455→  };
   456→
   457→  const handleUnenroll = (batchId: string) => {
   458→    if (!window.confirm('Are you sure you want to unenroll from this batch?')) return;
   459→    unenroll(batchId);
   460→    setBatchInfoOpen(false);
   461→    if (currentBatchId === batchId) {
   462→      const remaining = enrolled.filter((id) => id !== batchId);
   463→      if (remaining.length > 0) setCurrentBatchId(remaining[0]);
   464→      else {
   465→        setCurrentBatchId(null);
   466→        nav('home-view');
   467→      }
   468→    }
   469→  };

   474→  const renderLectureCard = (node: ContentNodePublic) => {
   475→    const d = node.data;
   476→    const thumb = d.videoDetails?.image || DEFAULT_THUMB;
   477→    const title = d.topic || d.name || 'Lecture';
   478→    const date = d.date ? new Date(d.date).toLocaleDateString('en-GB', { day: 'numeric', month: 'short', year: 'numeric' }) : '';
   479→    const vId = d.videoDetails?._id || d._id;
   480→    const vUrl = d.videoDetails?.videoUrl || '';
   481→    const isWatched = watched.includes(vId);
   482→    const bName = currentBatch?.name || 'PW Batch';
   483→    const playerUrl = `/player?videoId=${vId}&schId=${node._id}&batchId=${currentBatchId || ''}&subjectId=${currentSubject?.id || ''}&title=${encodeURIComponent(title)}&batchName=${encodeURIComponent(bName)}&vUrl=${encodeURIComponent(vUrl)}`;
   484→    const hasAttachments = (d.homeworkIds && d.homeworkIds.length > 0) || (d.exerciseIds && d.exerciseIds.length > 0);
   485→
   486→    return (
   487→      <div className="content-card" key={node._id} style={{ flexDirection: 'column', alignItems: 'stretch' }}>
   488→        <div className="d-flex gap-3 mb-3">
   489→          <img src={thumb} className="lecture-thumb" alt={title} />
   490→          <div className="flex-grow-1">
   491→            <span style={{ background: '#eff6ff', color: '#2563eb', fontSize: '0.7rem', fontWeight: 800, padding: '2px 6px', borderRadius: 4, display: 'inline-block', marginBottom: 5 }}>
   492→              {node.type === 'DPP_VIDEOS' ? 'DPP VIDEO' : 'VIDEO'}
   493→            </span>
   494→            <div style={{ fontWeight: 700, fontSize: '1rem', color: 'var(--sb-text-dark)' }}>{title}</div>
   495→            <div style={{ color: 'var(--sb-text-muted)', fontSize: '0.8rem', marginTop: 5 }}>
   496→              <i className="fas fa-clock"></i> {date} &bull; {d.videoDetails?.duration || '1h'}
   497→            </div>
   498→          </div>
   499→        </div>
   500→        <div className="d-flex gap-2">
   501→          <button
   502→            className={`btn-check-done ${isWatched ? 'active' : ''}`}
   503→            title="Mark as Watched"
   504→            onClick={() => {
   505→              const added = toggleWatched(vId);
   506→              if (added) addXP(10, 'Watched Video');
   507→            }}
   508→          >
