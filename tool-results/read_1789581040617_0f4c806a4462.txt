   101→   101→function eventStatus(item: LxScheduleItem, startIso: string, endIso: string): string {
   102→   102→  const tag = String(item.tag || "").toLowerCase();
   103→   103→  if (tag === "live" || tag === "live now") return "LIVE";
   104→   104→  if (tag === "upcoming") return "UPCOMING";
   105→   105→  if (tag === "ended" || tag === "cancelled") return tag === "cancelled" ? "CANCELLED" : "ENDED";
   106→   106→  const now = Date.now();
   107→   107→  const s = new Date(startIso || "").getTime();
   108→   108→  const e = new Date(endIso || "").getTime();
   109→   109→  if (Number.isFinite(s) && Number.isFinite(e)) {
   110→   110→    if (now >= s && now <= e) return "LIVE";
   111→   111→    if (now < s) return "UPCOMING";
   112→   112→  }
   113→   113→  return "ENDED";
   114→   114→}

   144→   144→function contentNodesFromItems(
   145→   145→  batchId: string,
   146→   146→  subjectId: string,
   147→   147→  items: LxScheduleItem[],
   148→   148→  kind: "LECTURE" | "DPP_VIDEOS" | "NOTES",
   149→   149→): Array<{ _id: string; type: string; data: Record<string, unknown> }> {
   150→   150→  const nodes: Array<{ _id: string; type: string; data: Record<string, unknown> }> = [];
   151→   151→  for (const item of items) {
   152→   152→    const pdfId = item._id || "";
   153→   153→
   154→   154→    if (kind === "LECTURE" || kind === "DPP_VIDEOS") {
   155→   155→      const vd = item.videoDetails || {};
   156→   156→      nodes.push({
   157→   157→        _id: item._id || "",
   158→   158→        type: kind,
   159→   159→        data: {
   160→   160→          _id: item._id || "",
   161→   161→          topic: item.topic || vd.name || item.name || "Lecture",
   162→   162→          date: item.date || item.startTime || "",
   163→   163→          urlType: item.urlType || "",
   164→   164→          url: item.url || "",
   165→   165→          videoDetails: {
   166→   166→            _id: item._id || "",
   167→   167→            image: vd.image || item.previewImageUrl || "",
   168→   168→            videoUrl: "",
   169→   169→            duration: vd.duration || "",
   170→   170→          },
   171→   171→          homeworkIds: asArray(item.homeworkIds).flatMap((hw) => {
   172→   172→            const h = hw as { _id?: string; topic?: string; attachmentIds?: unknown[] };
   173→   173→            return asArray(h.attachmentIds).map((att) => {
   174→   174→              const a = att as { _id?: string };
   175→   175→              return {
   176→   176→                _id: h._id || "",
   177→   177→                topic: h.topic || "Class Note",
   178→   178→                attachmentIds: [pdfAttachmentRef(batchId, subjectId, pdfId, a._id || "")],
   179→   179→              };
   180→   180→            });
   181→   181→          }),
   182→   182→          exerciseIds: [],
   183→   183→        },
   184→   184→      });
   185→   185→      continue;
   186→   186→    }
   187→   187→
   188→   188→    // NOTES / DppNotes → PDF cards
   189→   189→    const atts: Array<Record<string, unknown>> = [];
   190→   190→    let title = item.topic || item.name || "";
   191→   191→    const homeworkSources = [
   192→   192→      ...asArray(item.homeworkIds),
   193→   193→      ...asArray(item.dpp?.homeworkIds),
   194→   194→    ];
   195→   195→    for (const hw of homeworkSources) {
   196→   196→      const h = hw as { _id?: string; topic?: string; attachmentIds?: unknown[] };
   197→   197→      if (!title && h.topic) title = h.topic;
   198→   198→      for (const att of asArray(h.attachmentIds)) {
   199→   199→        const a = att as { _id?: string; key?: string; baseUrl?: string; name?: string };
   200→   200→        if (!a?._id) continue;
   201→   201→        if (!title) title = a.name || "";
   202→   202→        // some attachments already carry a real static.pw.live key
   203→   203→        const direct = a.baseUrl && a.key ? `${a.baseUrl}${a.key}` : "";
   204→   204→        const ref = direct
   205→   205→          ? { baseUrl: direct, key: "", _id: a._id }
   206→   206→          : pdfAttachmentRef(batchId, subjectId, pdfId, a._id);
   207→   207→        atts.push({
   208→   208→          _id: a._id,
   209→   209→          topic: h.topic || a.name || item.topic || item.name || "Study Material",
   210→   210→          attachmentIds: [ref],
   211→   211→        });
   212→   212→      }
   213→   213→    }
   214→   214→    if (atts.length === 0) continue;
   215→   215→    nodes.push({
   216→   216→      _id: item._id || "",
   217→   217→      type: "NOTES",
   218→   218→      data: {
   219→   219→        _id: item._id || "",
   220→   220→        topic: title || "Study Material",
   221→   221→        date: item.date || item.startTime || "",
   222→   222→        homeworkIds: atts,
   223→   223→      },
   224→   224→    });
   225→   225→  }
   226→   226→  return nodes;
   227→   227→}

   309→   309→// ---------------------------------------------------------------------------
   310→   310→// Lecture stream signing chain (server-side only).
   311→   311→//
   312→   312→// 1. signedUrl from the upstream video-url API (per-video folder policy)
   313→   313→// 2. LECTURE_SIGN_URL env template (owner's signing service, e.g.
   314→   314→//    https://video-detail.studyparcham.in/?batch_id={batch_id}&video_id={video_id}&subject_id={subject_id})
   315→   315→// 3. non-expired fallback constants registered in pw-cdn
   316→   316→// ---------------------------------------------------------------------------
   317→   317→
   318→   318→async function envSignerSignedQuery(
   319→   319→  batchId: string,
   320→   320→  videoId: string,
   321→   321→  subjectId: string,
   322→   322→): Promise<string> {
   323→   323→  const tpl = process.env.LECTURE_SIGN_URL || "";
   324→   324→  if (!tpl) return "";
   325→   325→  const url = tpl
   326→   326→    .replace(/\{batch_id\}/g, encodeURIComponent(batchId))
   327→   327→    .replace(/\{video_id\}/g, encodeURIComponent(videoId))
   328→   328→    .replace(/\{subject_id\}/g, encodeURIComponent(subjectId));
   329→   329→  try {
   330→   330→    const controller = new AbortController();
   331→   331→    const timer = setTimeout(() => controller.abort(), 12000);
   332→   332→    const res = await fetch(url, {
   333→   333→      cache: "no-store",
   334→   334→      signal: controller.signal,
   335→   335→      headers: { "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) Chrome/126.0" },
   336→   336→    });
   337→   337→    clearTimeout(timer);
   338→   338→    if (!res.ok) return "";
   339→   339→    const body = (await res.json()) as {
   340→   340→      url?: string;
   341→   341→      signedUrl?: string;
   342→   342→      data?: { url?: string; signedUrl?: string };
   343→   343→    };
   344→   344→    const signed = body.signedUrl || body.data?.signedUrl || "";
   345→   345→    if (signed) return signed.startsWith("?") ? signed : `?${signed}`;
   346→   346→    // a signed absolute URL works too — peel off its query
   347→   347→    const abs = body.url || body.data?.url || "";
   348→   348→    if (abs.includes("Signature=")) {
   349→   349→      const q = abs.slice(abs.indexOf("?"));
   350→   350→      return q;
   351→   351→    }
   352→   352→    return "";
   353→   353→  } catch {
   354→   354→    return "";
   355→   355→  }
   356→   356→}

   358→   358→/** Resolve the folder-wide signed query for a video through the whole chain. */
   359→   359→async function resolveSignedQuery(
   360→   360→  folder: string,
   361→   361→  opts: { batchId: string; videoId: string; subjectId: string; signedUrl?: string },
   362→   362→): Promise<string> {
   363→   363→  if (opts.signedUrl) {
   364→   364→    registerSignedFolder(folder, opts.signedUrl);
   365→   365→    return opts.signedUrl.startsWith("?") ? opts.signedUrl : `?${opts.signedUrl}`;
   366→   366→  }
   367→   367→  const envSig = await envSignerSignedQuery(opts.batchId, opts.videoId, opts.subjectId);
   368→   368→  if (envSig) {
   369→   369→    registerSignedFolder(folder, envSig);
   370→   370→    return envSig;
   371→   371→  }
   372→   372→  const fb = getFallbackSignedQuery(folder);
   373→   373→  if (fb) {
   374→   374→    registerSignedFolder(folder, fb);
   375→   375→    return fb;
   376→   376→  }
   377→   377→  const cached = getSignedQuery(folder);
   378→   378→  return cached || "";
   379→   379→}

   381→   381→/**
   382→   382→ * Same-origin relay URL for a resolved CloudFront asset — keeps the REAL
   383→   383→ * manifest path from the upstream url (e.g. hls/720/main.m3u8, master.mpd)
   384→   384→ * instead of guessing, and carries the signature statelessly in the path.
   385→   385→ */
   386→   386→async function relayUrlFor(
   387→   387→  rawUrl: string,
   388→   388→  opts: { batchId: string; videoId: string; subjectId: string; signedUrl?: string },
   389→   389→): Promise<{ url: string; signedQuery: string } | null> {
   390→   390→  const host = hostFromUrl(rawUrl);
   391→   391→  const folder = folderFromUrl(rawUrl);
   392→   392→  const sub = subPathFromUrl(rawUrl);
   393→   393→  if (!folder || !sub) return null;
   394→   394→  const signedQuery = await resolveSignedQuery(folder, { ...opts, signedUrl: opts.signedUrl || "" });
   395→   395→  if (host) registerFolderHost(folder, host);
   396→   396→  if (!signedQuery) return null;
   397→   397→  const sig = sigPathSegment(signedQuery);
   398→   398→  const hostSeg = host && host !== "d1d34p8vz63oiq.cloudfront.net" ? `/h/${host}` : "";
   399→   399→  return { url: `/api/pw${hostSeg}/${folder}${sig ? `/s/${sig}` : ""}/${sub}`, signedQuery };
   400→   400→}

   402→   402→/** Resolve a video → same-origin playable payload (dash/hls/youtube/mp4). */
   403→   403→async function resolveVideoPayload(
   404→   404→  childId: string,
   405→   405→  batchId: string,
   406→   406→  subjectId: string,
   407→   407→  vUrlParam = "",
   408→   408→  realVideoId = "",
   409→   409→) {
   410→   410→  let vid = await lxVideoUrl(batchId, subjectId, childId).catch(() => null);
   411→   411→  if (!vid && realVideoId && realVideoId !== childId) {
   412→   412→    vid = await lxVideoUrl(batchId, subjectId, realVideoId).catch(() => null);
   413→   413→  }
   414→   414→  const sigOpts = { batchId, videoId: realVideoId || childId, subjectId };
   415→   415→  const rawUrl = vid?.url?.trim() || vUrlParam.trim();
   416→   416→  if (rawUrl) {
   417→   417→    const url = rawUrl;
   418→   418→    if (/\.mpd(\?|$)/i.test(url)) {
   419→   419→      const relay = await relayUrlFor(url, { ...sigOpts, signedUrl: vid?.signedUrl || "" });
   420→   420→      if (!relay) return null;
   421→   421→      let clearKeys: Record<string, string> | null = null;
   422→   422→      try {
   423→   423→        // Probe the manifest through the same upstream lane chain the relay
   424→   424→        // uses (delta → pearl → direct) to extract the ClearKey KID.
   425→   425→        const u = new URL(url);
   426→   426→        const mpdRes = await fetchSignedAsset(
   427→   427→          u.host,
   428→   428→          u.pathname.replace(/^\/+/, ""),
   429→   429→          relay.signedQuery,
   430→   430→          { timeoutMs: 30000 },
   431→   431→        );
   432→   432→        if (mpdRes.ok) {
   433→   433→          const text = await mpdRes.text();
   434→   434→          const kidMatch = text.match(/default_KID="([0-9a-fA-F-]+)"/);
   435→   435→          if (kidMatch) {
   436→   436→            clearKeys = await lxOtp(kidMatch[1]);
   437→   437→          }
   438→   438→        }
   439→   439→      } catch {
   440→   440→        /* keys unavailable — player will surface a decrypt error */
   441→   441→      }
   442→   442→      return {
   443→   443→        kind: "dash",
   444→   444→        url: relay.url,
   445→   445→        signedUrl: "",
   446→   446→        clearKeys,
   447→   447→        poster: "",
   448→   448→        drmDetails: null,
   449→   449→      };
   450→   450→    }
   451→   451→    if (/\.m3u8(\?|$)/i.test(url)) {
   452→   452→      const relay = await relayUrlFor(url, { ...sigOpts, signedUrl: vid?.signedUrl || "" });
   453→   453→      if (!relay) return null;
   454→   454→      return {
   455→   455→        kind: "hls",
   456→   456→        url: relay.url,
   457→   457→        signedUrl: "",
   458→   458→        clearKeys: null,
   459→   459→        poster: "",
   460→   460→        drmDetails: null,
   461→   461→      };
   462→   462→    }
   463→   463→    if (/youtube\.com|youtu\.be/i.test(url)) {
   464→   464→      return { kind: "youtube", url, signedUrl: "", clearKeys: null, poster: "", drmDetails: null };
   465→   465→    }
   466→   466→    // plain file (mp4 etc.) inside the signed folder
   467→   467→    const relay = await relayUrlFor(url, { ...sigOpts, signedUrl: vid?.signedUrl || "" });
   468→   468→    if (!relay) return null;
   469→   469→    return {
   470→   470→      kind: "mp4",
   471→   471→      url: relay.url,
   472→   472→      signedUrl: "",
   473→   473→      clearKeys: null,
   474→   474→      poster: "",
   475→   475→      drmDetails: null,
   476→   476→    };
   477→   477→  }
   478→   478→
   479→   479→  // No signed stream — maybe a youtube lecture.
   480→   480→  if (subjectId) {
   481→   481→    const detail = await lxScheduleDetail(batchId, subjectId, childId).catch(() => null);
   482→   482→    const url = (detail as { url?: string } | null)?.url || "";
   483→   483→    if (url && /youtube\.com|youtu\.be/i.test(url)) {
   484→   484→      return { kind: "youtube", url, signedUrl: "", clearKeys: null, poster: "", drmDetails: null };
   485→   485→    }
   486→   486→  }
   487→   487→  return null;
   488→   488→}

   851→   851→/** Flatten test ids from scalars / arrays / objects ({testId}|{_id}|{id}). */
   852→   852→function collectTestIds(source: unknown, out: string[], seen: Set<string>): void {
   853→   853→  if (source === undefined || source === null) return;
   854→   854→  if (Array.isArray(source)) {
   855→   855→    source.forEach((item) => collectTestIds(item, out, seen));
   856→   856→    return;
   857→   857→  }
   858→   858→  if (typeof source === "object") {
   859→   859→    const o = source as Record<string, unknown>;
   860→   860→    collectTestIds(o.testId ?? o._id ?? o.id, out, seen);
   861→   861→    return;
   862→   862→  }
   863→   863→  const s = String(source).trim();
   864→   864→  if (s && !seen.has(s)) {
   865→   865→    seen.add(s);
   866→   866→    out.push(s);
   867→   867→  }
   868→   868→}
   869→   869→
