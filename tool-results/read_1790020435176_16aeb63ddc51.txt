    41→type MemoEntry = { at: number; ttl: number; value: unknown };
    42→const MEMO_MAX = 400;
    43→
    44→function memoGet(map: Map<string, MemoEntry>, key: string): MemoEntry | null {
    45→  const hit = map.get(key);
    46→  if (!hit) return null;
    47→  if (Date.now() - hit.at > hit.ttl) {
    48→    map.delete(key);
    49→    return null;
    50→  }
    51→  return hit;
    52→}
    53→
    54→function memoSet(map: Map<string, MemoEntry>, key: string, value: unknown, ttl: number): void {
    55→  if (map.size >= MEMO_MAX) {
    56→    const oldest = map.keys().next().value;
    57→    if (oldest) map.delete(oldest);
    58→  }
    59→  map.set(key, { at: Date.now(), ttl, value });
    60→}

    62→/** True when the query has no Policy expiry, or the expiry is still ahead. */
    63→function nonExpiredQuery(q: string | null): string | null {
    64→  if (!q) return null;
    65→  const policy = parsePolicy(q.startsWith("?") ? q : `?${q}`);
    66→  if (policy?.exp && Math.floor(Date.now() / 1000) >= policy.exp) return null;
    67→  return q;
    68→}

   137→function eventStatus(item: LxScheduleItem, startIso: string, endIso: string): string {
   138→  const tag = String(item.tag || "").toLowerCase();
   139→  if (tag === "live" || tag === "live now") return "LIVE";
   140→  if (tag === "upcoming") return "UPCOMING";
   141→  if (tag === "ended" || tag === "cancelled") return tag === "cancelled" ? "CANCELLED" : "ENDED";
   142→  const now = Date.now();
   143→  const s = new Date(startIso || "").getTime();
   144→  const e = new Date(endIso || "").getTime();
   145→  if (Number.isFinite(s) && Number.isFinite(e)) {
   146→    if (now >= s && now <= e) return "LIVE";
   147→    if (now < s) return "UPCOMING";
   148→  }
   149→  return "ENDED";
   150→}

   180→function contentNodesFromItems(
   181→  batchId: string,
   182→  subjectId: string,
   183→  items: LxScheduleItem[],
   184→  kind: "LECTURE" | "DPP_VIDEOS" | "NOTES",
   185→): Array<{ _id: string; type: string; data: Record<string, unknown> }> {
   186→  const nodes: Array<{ _id: string; type: string; data: Record<string, unknown> }> = [];
   187→  for (const item of items) {
   188→    const pdfId = item._id || "";
   189→
   190→    if (kind === "LECTURE" || kind === "DPP_VIDEOS") {
   191→      const vd = item.videoDetails || {};
   192→      nodes.push({
   193→        _id: item._id || "",
   194→        type: kind,
   195→        data: {
   196→          _id: item._id || "",
   197→          topic: item.topic || vd.name || item.name || "Lecture",
   198→          date: item.date || item.startTime || "",
   199→          urlType: item.urlType || "",
   200→          url: item.url || "",
   201→          videoDetails: {
   202→            _id: item._id || "",
   203→            image: vd.image || item.previewImageUrl || "",
   204→            videoUrl: "",
   205→            duration: vd.duration || "",
   206→          },
   207→          homeworkIds: asArray(item.homeworkIds).flatMap((hw) => {
   208→            const h = hw as { _id?: string; topic?: string; attachmentIds?: unknown[] };
   209→            return asArray(h.attachmentIds).map((att) => {
   210→              const a = att as { _id?: string };
   211→              return {
   212→                _id: h._id || "",
   213→                topic: h.topic || "Class Note",
   214→                attachmentIds: [pdfAttachmentRef(batchId, subjectId, pdfId, a._id || "")],
   215→              };
   216→            });
   217→          }),
   218→          exerciseIds: [],
   219→        },
   220→      });
   221→      continue;
   222→    }
   223→
   224→    // NOTES / DppNotes → PDF cards
   225→    const atts: Array<Record<string, unknown>> = [];
   226→    let title = item.topic || item.name || "";
   227→    const homeworkSources = [
   228→      ...asArray(item.homeworkIds),
   229→      ...asArray(item.dpp?.homeworkIds),
   230→    ];
   231→    for (const hw of homeworkSources) {
   232→      const h = hw as { _id?: string; topic?: string; attachmentIds?: unknown[] };
   233→      if (!title && h.topic) title = h.topic;
   234→      for (const att of asArray(h.attachmentIds)) {
   235→        const a = att as { _id?: string; key?: string; baseUrl?: string; name?: string };
   236→        if (!a?._id) continue;
   237→        if (!title) title = a.name || "";
   238→        // some attachments already carry a real static.pw.live key
   239→        const direct = a.baseUrl && a.key ? `${a.baseUrl}${a.key}` : "";
   240→        const ref = direct
   241→          ? { baseUrl: direct, key: "", _id: a._id }
   242→          : pdfAttachmentRef(batchId, subjectId, pdfId, a._id);
   243→        atts.push({
   244→          _id: a._id,
   245→          topic: h.topic || a.name || item.topic || item.name || "Study Material",
   246→          attachmentIds: [ref],
   247→        });
   248→      }
   249→    }
   250→    if (atts.length === 0) continue;
   251→    nodes.push({
   252→      _id: item._id || "",
   253→      type: "NOTES",
   254→      data: {
   255→        _id: item._id || "",
   256→        topic: title || "Study Material",
   257→        date: item.date || item.startTime || "",
   258→        homeworkIds: atts,
   259→      },
   260→    });
   261→  }
   262→  return nodes;
   263→}

   293→/**
   294→ * Direct (unsigned) CloudFront urls per schedule item from the owner's
   295→ * V2 listing API — the same source the production site merges into its
   296→ * content lists. Best-effort: failure simply leaves the nodes untouched.
   297→ * Memoized 5 min (60s when empty) so chapter revisits skip the roundtrip.
   298→ */
   299→const v2Memo = new Map<string, MemoEntry>();
   300→
   301→async function v2DirectUrls(
   302→  batchId: string,
   303→  subjectId: string,
   304→  tagId: string,
   305→): Promise<Map<string, { url: string; videoId: string }>> {
   306→  const memoKey = `${batchId}|${subjectId}|${tagId}`;
   307→  const hit = memoGet(v2Memo, memoKey);
   308→  if (hit) return hit.value as Map<string, { url: string; videoId: string }>;
   309→  const out = await fetchV2DirectUrls(batchId, subjectId, tagId);
   310→  memoSet(v2Memo, memoKey, out, out.size > 0 ? 5 * 60_000 : 60_000);
   311→  return out;
   312→}

   314→async function fetchV2DirectUrls(
   315→  batchId: string,
   316→  subjectId: string,
   317→  tagId: string,
   318→): Promise<Map<string, { url: string; videoId: string }>> {
   319→  const out = new Map<string, { url: string; videoId: string }>();
   320→  try {
   321→    const controller = new AbortController();
   322→    const timer = setTimeout(() => controller.abort(), 9000);
   323→    const res = await fetch("https://pw.studyparcham.in/api/data", {
   324→      method: "POST",
   325→      headers: { "Content-Type": "application/json" },
   326→      cache: "no-store",
   327→      signal: controller.signal,
   328→      body: JSON.stringify({
   329→        action: "pw_v2_list",
   330→        params: { batchId, subjectId, tagId, contentType: "videos", page: 1 },
   331→        method: "GET",
   332→        payload: null,
   333→      }),
   334→    });
   335→    clearTimeout(timer);
   336→    if (!res.ok) return out;
   337→    const body = (JSON.parse(await res.text()) as {
   338→      data?: Array<{
   339→        _id?: string;
   340→        url?: string;
   341→        videoDetails?: { videoUrl?: string; _id?: string };
   342→      }>;
   343→    });
   344→    for (const it of body.data || []) {
   345→      const u = it.url || it.videoDetails?.videoUrl || "";
   346→      if (it._id && u) out.set(it._id, { url: u, videoId: it.videoDetails?._id || "" });
   347→    }
   348→  } catch {
   349→    /* best-effort merge */
   350→  }
   351→  return out;
   352→}

   361→// ---------------------------------------------------------------------------
   362→// Lecture stream signing chain (server-side only).
   363→//
   364→// 1. signedUrl from the upstream video-url API (per-video folder policy)
   365→// 2. LECTURE_SIGN_URL env template (owner's signing service, e.g.
   366→//    https://video-detail.studyparcham.in/?batch_id={batch_id}&video_id={video_id}&subject_id={subject_id})
   367→// 3. non-expired fallback constants registered in pw-cdn
   368→// ---------------------------------------------------------------------------
   369→
   370→async function envSignerSignedQuery(
   371→  batchId: string,
   372→  videoId: string,
   373→  subjectId: string,
   374→): Promise<string> {
   375→  const tpl = process.env.LECTURE_SIGN_URL || "";
   376→  if (!tpl) return "";
   377→  const url = tpl
   378→    .replace(/\{batch_id\}/g, encodeURIComponent(batchId))
   379→    .replace(/\{video_id\}/g, encodeURIComponent(videoId))
   380→    .replace(/\{subject_id\}/g, encodeURIComponent(subjectId));
   381→  try {
   382→    const controller = new AbortController();
   383→    const timer = setTimeout(() => controller.abort(), 8000);
   384→    const res = await fetch(url, {
   385→      cache: "no-store",
   386→      signal: controller.signal,
   387→      headers: { "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) Chrome/126.0" },
   388→    });
   389→    clearTimeout(timer);
   390→    if (!res.ok) return "";
   391→    const body = (await res.json()) as {
   392→      url?: string;
   393→      signedUrl?: string;
   394→      data?: { url?: string; signedUrl?: string };
   395→    };
   396→    const signed = body.signedUrl || body.data?.signedUrl || "";
   397→    if (signed) return signed.startsWith("?") ? signed : `?${signed}`;
   398→    // a signed absolute URL works too — peel off its query
   399→    const abs = body.url || body.data?.url || "";
   400→    if (abs.includes("Signature=")) {
   401→      const q = abs.slice(abs.indexOf("?"));
   402→      return q;
   403→    }
   404→    return "";
   405→  } catch {
   406→    return "";
   407→  }
   408→}

   414→  folder: string,

   426→/** Resolve the folder-wide signed query for a video through the whole chain. */
   427→async function resolveSignedQuery(
   428→  folder: string,
   429→  opts: { batchId: string; videoId: string; subjectId: string; signedUrl?: string },
   430→): Promise<string> {
   431→  if (opts.signedUrl) {
   432→    registerSignedFolder(folder, opts.signedUrl);
   433→    return opts.signedUrl.startsWith("?") ? opts.signedUrl : `?${opts.signedUrl}`;
   434→  }
   435→  // 1) Non-expired LOCAL signatures first — zero network, saves the env
   436→  //    signer roundtrip (1-12s) on every cold load of a known folder.
   437→  const local =
   438→    nonExpiredQuery(getFallbackSignedQuery(folder)) ||
   439→    nonExpiredQuery(lookupStoredSignature(folder)) ||
   440→    nonExpiredQuery(getSignedQuery(folder));
   441→  if (local) {
   442→    registerSignedFolder(folder, local);
   443→    return local;
   444→  }
   445→  // 2) Env signing service (memoized per folder, 8s cap).
   446→  const envSig = await envSignerSignedQueryCached(folder, opts.batchId, opts.videoId, opts.subjectId);
   447→  if (envSig) {
   448→    registerSignedFolder(folder, envSig);
   449→    return envSig;
   450→  }
   451→  // Nothing usable — let the caller decide (upstream signedUrl retry, error).
   452→  return "";
   453→}

   455→/**
   456→ * Same-origin relay URL for a resolved CloudFront asset — keeps the REAL
   457→ * manifest path from the upstream url (e.g. hls/720/main.m3u8, master.mpd)
   458→ * instead of guessing, and carries the signature statelessly in the path.
   459→ */
   460→async function relayUrlFor(
   461→  rawUrl: string,
   462→  opts: { batchId: string; videoId: string; subjectId: string; signedUrl?: string },
   463→): Promise<{ url: string; signedQuery: string } | null> {
   464→  const host = hostFromUrl(rawUrl);
   465→  const folder = folderFromUrl(rawUrl);
   466→  const sub = subPathFromUrl(rawUrl);
   467→  if (!folder || !sub) return null;
   468→  const signedQuery = await resolveSignedQuery(folder, { ...opts, signedUrl: opts.signedUrl || "" });
   469→  if (host) registerFolderHost(folder, host);
   470→  if (!signedQuery) return null;
   471→  const sig = sigPathSegment(signedQuery);
   472→  const hostSeg = host && host !== "d1d34p8vz63oiq.cloudfront.net" ? `/h/${host}` : "";
   473→  return { url: `/api/pw${hostSeg}/${folder}${sig ? `/s/${sig}` : ""}/${sub}`, signedQuery };
   474→}

   476→/**
   477→ * ClearKey cache — the KID and keys are stable per video folder, but probing
   478→ * the manifest through the lane chain + otp can cost up to ~30s. Memoize per
   479→ * folder (60 min for keys, 60s negative) with single-flight.
   480→ */
   481→const folderKeysMemo = new Map<string, MemoEntry>();
   482→const folderKeysInflight = new Map<string, Promise<Record<string, string> | null>>();
   483→
   484→async function clearKeysForFolder(
   485→  folder: string,
   486→  host: string,
   487→  manifestPath: string,
   488→  signedQuery: string,
   489→): Promise<Record<string, string> | null> {
   490→  if (!folder) return null;
   491→  const hit = memoGet(folderKeysMemo, `ck:${folder}`);
   492→  if (hit) return hit.value as Record<string, string> | null;
   493→  const running = folderKeysInflight.get(folder);
   494→  if (running) return running;
   495→  const p = (async () => {
   496→    try {
   497→      // Probe the manifest through the same upstream lane chain the relay
   498→      // uses (delta → pearl → direct) to extract the ClearKey KID. 8s per
   499→      // lane keeps a dead-lane worst case bounded.
   500→      const mpdRes = await fetchSignedAsset(host, manifestPath, signedQuery, { timeoutMs: 8000 });
   501→      if (!mpdRes.ok) return null;
   502→      const text = await mpdRes.text();
   503→      const kidMatch = text.match(/default_KID="([0-9a-fA-F-]+)"/);
   504→      if (!kidMatch) return null;
   505→      const keys = await lxOtp(kidMatch[1]);
   506→      if (keys && Object.keys(keys).length > 0) {
   507→        memoSet(folderKeysMemo, `ck:${folder}`, keys, 60 * 60_000);
   508→        return keys;
   509→      }
   510→      // keys unavailable right now — brief negative cache so a retry soon
   511→      // after isn't blocked, but the next cold load re-probes.
   512→      memoSet(folderKeysMemo, `ck:${folder}`, null, 60_000);
   513→      return null;
   514→    } catch {
   515→      return null;
   516→    } finally {
   517→      folderKeysInflight.delete(folder);
   518→    }
   519→  })();
   520→  folderKeysInflight.set(folder, p);
   521→  return p;
   522→}

   533→/** Build a same-origin playable payload for a resolved manifest/file url. */
   534→async function payloadFromRawUrl(
   535→  rawUrl: string,
   536→  upstreamSignedUrl: string,
   537→  sigOpts: { batchId: string; videoId: string; subjectId: string },
   538→): Promise<VideoPayload | null> {
   539→  const url = rawUrl.trim();
   540→  if (!url) return null;
   541→  if (/\.mpd(\?|$)/i.test(url)) {
   542→    const relay = await relayUrlFor(url, { ...sigOpts, signedUrl: upstreamSignedUrl });
   543→    if (!relay) return null;
   544→    let clearKeys: Record<string, string> | null = null;
   545→    try {
   546→      const u = new URL(url);
   547→      const folder = folderFromUrl(url);
   548→      clearKeys = await clearKeysForFolder(folder, u.host, u.pathname.replace(/^\/+/, ""), relay.signedQuery);
   549→    } catch {
   550→      /* keys unavailable — player will surface a decrypt error */
   551→    }
   552→    return { kind: "dash", url: relay.url, signedUrl: "", clearKeys, poster: "", drmDetails: null };
   553→  }
   554→  if (/\.m3u8(\?|$)/i.test(url)) {
   555→    // Root-level LIVE manifest (PW live channel: https://<dist>/index.m3u8, no
   556→    // UUID folder) → the /api/live relay, which signs it from the persistent
   557→    // store's "live:<host>" entry (owner-pasted signed live URL).
   558→    try {
   559→      const u = new URL(url);
   560→      const segs = u.pathname.split("/").filter(Boolean);
   561→      if (segs.length === 1 && /\.cloudfront\.net$/i.test(u.host)) {
   562→        return {
   563→          kind: "hls",
   564→          url: `/api/live/h/${u.host}/${segs[0]}`,
   565→          signedUrl: "",
   566→          clearKeys: null,
   567→          poster: "",
   568→          drmDetails: null,
   569→        };
   570→      }
   571→    } catch {
   572→      /* fall through to the folder-based relay */
   573→    }
   574→    const relay = await relayUrlFor(url, { ...sigOpts, signedUrl: upstreamSignedUrl });
   575→    if (!relay) return null;
   576→    return { kind: "hls", url: relay.url, signedUrl: "", clearKeys: null, poster: "", drmDetails: null };
   577→  }
   578→  if (/youtube\.com|youtu\.be/i.test(url)) {
   579→    return { kind: "youtube", url, signedUrl: "", clearKeys: null, poster: "", drmDetails: null };
   580→  }
   581→  // plain file (mp4 etc.) inside the signed folder
   582→  const relay = await relayUrlFor(url, { ...sigOpts, signedUrl: upstreamSignedUrl });
   583→  if (!relay) return null;
   584→  return { kind: "mp4", url: relay.url, signedUrl: "", clearKeys: null, poster: "", drmDetails: null };
   585→}

   587→/** Resolve a video → same-origin playable payload (dash/hls/youtube/mp4). */
   588→async function resolveVideoPayload(
   589→  childId: string,
   590→  batchId: string,
   591→  subjectId: string,
   592→  vUrlParam = "",
   593→  realVideoId = "",
   594→) {
   595→  const sigOpts = { batchId, videoId: realVideoId || childId, subjectId };
   596→  const direct = vUrlParam.trim();
   597→
   598→  // FAST PATH: the lecture list already carries the CloudFront manifest url
   599→  // (v2 merge). Skip the slow upstream video-url roundtrip entirely and sign
   600→  // from local/env sources; only when no signature is available do we fall
   601→  // through to the full chain below.
   602→  if (direct && /\.cloudfront\.net/i.test(direct)) {
   603→    const fast = await payloadFromRawUrl(direct, "", sigOpts);
   604→    if (fast) return fast;
   605→  }
   606→
   607→  let vid = await lxVideoUrl(batchId, subjectId, childId).catch(() => null);
   608→  if (!vid && realVideoId && realVideoId !== childId) {
   609→    vid = await lxVideoUrl(batchId, subjectId, realVideoId).catch(() => null);
   610→  }
   611→  const rawUrl = vid?.url?.trim() || direct;
   612→  if (rawUrl) {
   613→    const payload = await payloadFromRawUrl(rawUrl, vid?.signedUrl || "", sigOpts);
   614→    if (payload) return payload;
   615→  }
   616→
   617→  // No video-doc stream — the SCHEDULE itself may carry the manifest. Live
   618→  // classes (urlType awsVideo) keep an UNSIGNED CloudFront HLS url here and
   619→  // have no video doc at all; past live classes keep their replay url. The
   620→  // schedule url is the only remaining source — use it for every kind
   621→  // (youtube / m3u8 / mpd / mp4), not just youtube.
   622→  if (subjectId) {
   623→    const detail = await lxScheduleDetail(batchId, subjectId, childId).catch(() => null);
   624→    const schedUrl = (detail as { url?: string } | null)?.url || "";
   625→    if (schedUrl && /^https?:\/\//i.test(schedUrl.trim())) {
   626→      const payload = await payloadFromRawUrl(schedUrl.trim(), "", sigOpts);
   627→      if (payload) return payload;
   628→      // youtube schedules stay a direct passthrough even if nothing resolved
   629→      if (/youtube\.com|youtu\.be/i.test(schedUrl)) {
   630→        return { kind: "youtube", url: schedUrl.trim(), signedUrl: "", clearKeys: null, poster: "", drmDetails: null };
   631→      }
   632→    }
   633→  }
   634→  return null;
   635→}

   637→/**
   638→ * Memoized entry point for the parcham_vid action.
   639→ *  - full payload .............. 10 min
   640→ *  - dash without clearKeys ..... 45s (probe failed — retry soon)
   641→ *  - unavailable (null) ......... 30s negative cache (no upstream hammering)
   642→ * Concurrent identical requests share one in-flight resolution.
   643→ */
   644→const videoMemo = new Map<string, MemoEntry>();
   645→const videoInflight = new Map<string, Promise<VideoPayload | null>>();
   646→
   647→async function resolveVideoPayloadCached(
   648→  childId: string,
   649→  batchId: string,
   650→  subjectId: string,
   651→  vUrlParam = "",
   652→  realVideoId = "",
   653→): Promise<VideoPayload | null> {
   654→  const key = `${batchId}|${subjectId}|${childId}|${realVideoId}|${(vUrlParam || "").split("?")[0]}`;
   655→  const hit = memoGet(videoMemo, key);
   656→  if (hit) return hit.value as VideoPayload | null;
   657→  const running = videoInflight.get(key);
   658→  if (running) return running;
   659→  const p = (async () => {
   660→    try {
   661→      const value = await resolveVideoPayload(childId, batchId, subjectId, vUrlParam, realVideoId);
   662→      const hasKeys = value?.kind === "dash" ? !!value.clearKeys : true;
   663→      const ttl = !value ? 30_000 : hasKeys ? 10 * 60_000 : 45_000;
   664→      memoSet(videoMemo, key, value, ttl);
   665→      return value;
   666→    } finally {
   667→      videoInflight.delete(key);
   668→    }
   669→  })();
   670→  videoInflight.set(key, p);
   671→  return p;
   672→}

  1044→/** Flatten test ids from scalars / arrays / objects ({testId}|{_id}|{id}). */
  1045→function collectTestIds(source: unknown, out: string[], seen: Set<string>): void {
  1046→  if (source === undefined || source === null) return;
  1047→  if (Array.isArray(source)) {
  1048→    source.forEach((item) => collectTestIds(item, out, seen));
  1049→    return;
