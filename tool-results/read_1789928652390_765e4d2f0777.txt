    41→function formatTime(t: number): string {
    42→  if (!isFinite(t) || t < 0) return '00:00';
    43→  const s = Math.floor(t % 60);
    44→  const m = Math.floor((t / 60) % 60);
    45→  const h = Math.floor(t / 3600);
    46→  const mm = String(m).padStart(2, '0');
    47→  const ss = String(s).padStart(2, '0');
    48→  return h > 0 ? `${h}:${mm}:${ss}` : `${mm}:${ss}`;
    49→}

    51→function timeAgo(iso: string): string {
    52→  const then = new Date(iso).getTime();
    53→  if (!isFinite(then)) return '';
    54→  const secs = Math.max(0, Math.floor((Date.now() - then) / 1000));
    55→  if (secs < 60) return 'just now';
    56→  const mins = Math.floor(secs / 60);
    57→  if (mins < 60) return `${mins}m ago`;
    58→  const hrs = Math.floor(mins / 60);
    59→  if (hrs < 24) return `${hrs}h ago`;
    60→  const days = Math.floor(hrs / 24);
    61→  if (days < 30) return `${days}d ago`;
    62→  return new Date(then).toLocaleDateString();
    63→}
