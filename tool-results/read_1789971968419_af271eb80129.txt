     1→'use client';
     2→
     3→// ============================================================================
     4→// Study Bison — main SPA (rebranded from the studybee pro clone)
     5→// Views: All Courses (home) / Study Dashboard / All Classes / Chapters /
     6→// Content / All Tests. Every content action is backed by the REAL PW data
     7→// relayed from the LearnxPW upstream (batches, subjects, chapters, lectures,
     8→// notes & DPP PDFs, tests, live schedule).
     9→// ============================================================================
    10→
    11→import { useCallback, useEffect, useRef, useState } from 'react';
    12→import {
    13→  api,
    14→  fetchCatalog,
    15→  imageUrl,
    16→  BatchCatalogItem,
    17→  BatchDetailPublic,
    18→  BatchSubjectPublic,
    19→  ChapterItemPublic,
    20→  ContentNodePublic,
    21→  ScheduleEventPublic,
    22→  TestItemPublic,
    23→} from '@/lib/api-client';
    24→import { addXP, useEnrolledBatches, useXP, useWatchedVideos } from '@/lib/local-state';
    25→import {
    26→  SbDownload,
    27→  listDownloads,
    28→  subscribeDownloads,
    29→  deleteDownload,
    30→  pauseDownload,
    31→  startDownload,
    32→  estimateOfflineStorage,
    33→  formatBytes,
    34→} from '@/lib/offline';
    35→
    36→export type ViewId = 'home-view' | 'study-view' | 'all-classes-view' | 'chapters-view' | 'content-view' | 'all-tests-view' | 'downloads-view';
    37→
    38→interface SubjectRef { id: string; internalId: string; name: string; masterId: string; batchSubjectId: string }
    39→interface TopicRef { id: string; name: string }
    40→interface AttachmentRow { title: string; url: string }
    41→interface DppMeta {
    42→  testId: string;
    43→  scheduleId: string;
    44→  subjectId: string;
    45→  name: string;
    46→}
    47→
    48→const DEFAULT_THUMB = 'https://store.pw.live/footer-logo.svg';
    49→
    50→export default function StudyBisonApp() {
    51→  // ---------------- global nav ----------------
    52→  const [view, setView] = useState<ViewId>('home-view');
    53→  const [sidebarOpen, setSidebarOpen] = useState(false);
    54→
    55→  // ---------------- home / catalog ----------------
    56→  const [batches, setBatches] = useState<BatchCatalogItem[]>([]);
    57→  const [page, setPage] = useState(1);
    58→  const [hasMore, setHasMore] = useState(false);
    59→  const [searchTerm, setSearchTerm] = useState('');
    60→  const [searching, setSearching] = useState(false);
    61→  const [catalogError, setCatalogError] = useState('');
    62→  const searchTimeout = useRef<ReturnType<typeof setTimeout> | null>(null);
    63→  const debounceTerm = useRef('');
    64→
    65→  // ---------------- enrollment ----------------
    66→  const { enrolled, enroll, unenroll } = useEnrolledBatches();
    67→  const { watched, toggleWatched } = useWatchedVideos();
    68→  const xp = useXP();
    69→  const batchDetailCache = useRef<Map<string, BatchDetailPublic>>(new Map());
    70→
    71→  // ---------------- study ----------------
    72→  const [currentBatchId, setCurrentBatchId] = useState<string | null>(null);
    73→  const [enrolledDetails, setEnrolledDetails] = useState<BatchDetailPublic[]>([]);
    74→  const [events, setEvents] = useState<ScheduleEventPublic[]>([]);
    75→  const [eventsLoading, setEventsLoading] = useState(false);
    76→
    77→  // ---------------- classes / chapters / content ----------------
    78→  const [subjects, setSubjects] = useState<BatchSubjectPublic[]>([]);
    79→  const [subjectsLoading, setSubjectsLoading] = useState(false);
    80→  const [currentSubject, setCurrentSubject] = useState<SubjectRef | null>(null);
    81→  const [chaptersTab, setChaptersTab] = useState<'UNITS' | 'STUDY_MATERIAL'>('UNITS');
    82→  const [chapters, setChapters] = useState<ChapterItemPublic[]>([]);
    83→  const [chaptersLoading, setChaptersLoading] = useState(false);
    84→  const [currentTopic, setCurrentTopic] = useState<TopicRef | null>(null);
    85→  const [contentTab, setContentTab] = useState('ALL');
    86→  const [contentNodes, setContentNodes] = useState<ContentNodePublic[]>([]);
    87→  const [dppItems, setDppItems] = useState<TestItemPublic[]>([]);
    88→  const [contentLoading, setContentLoading] = useState(false);
    89→
    90→  // ---------------- tests ----------------
    91→  const [tests, setTests] = useState<TestItemPublic[]>([]);
    92→  const [testsLoading, setTestsLoading] = useState(false);
    93→
    94→  // ---------------- PWA offline downloads ----------------
    95→  const [downloads, setDownloads] = useState<SbDownload[]>([]);
    96→  const [storageEst, setStorageEst] = useState<{ usage: number; quota: number } | null>(null);
    97→
    98→  // ---------------- resources tab ----------------
    99→  const [resourcesTab, setResourcesTab] = useState<'subjects' | 'resources'>('subjects');
   100→  const [resourceTopics, setResourceTopics] = useState<ChapterItemPublic[]>([]);
   101→  const [resourcesLoading, setResourcesLoading] = useState(false);
   102→
   103→  // ---------------- modals ----------------
   104→  const [batchInfoOpen, setBatchInfoOpen] = useState(false);
   105→  const [batchInfo, setBatchInfo] = useState<BatchDetailPublic | null>(null);
   106→  const [batchInfoLoading, setBatchInfoLoading] = useState(false);
   107→
   108→  const [notesOpen, setNotesOpen] = useState(false);
   109→  const [notesLoading, setNotesLoading] = useState(false);
   110→  const [noteRows, setNoteRows] = useState<AttachmentRow[]>([]);
   111→
   112→  const [dppOpen, setDppOpen] = useState(false);
   113→  const [dppMeta, setDppMeta] = useState<DppMeta | null>(null);
   114→
   115→  const [testFrame, setTestFrame] = useState<{ open: boolean; url: string; title: string }>({ open: false, url: '', title: 'Test Attempt' });
   116→
   117→  const [instructions, setInstructions] = useState<{ open: boolean; loading: boolean; title: string; html: string }>({ open: false, loading: false, title: '', html: '' });
   118→
   119→  // ==================================================================
   120→  // NAV
   121→  // ==================================================================
   122→  const nav = useCallback((v: ViewId) => {
   123→    setView(v);
   124→    setSidebarOpen(false);
   125→    window.scrollTo(0, 0);
   126→  }, []);
   127→
   128→  // ==================================================================
   129→  // OFFLINE LECTURE LIBRARY (PWA downloads)
   130→  // ==================================================================
   131→  const refreshDownloads = useCallback(() => {
   132→    void listDownloads().then((list) => {
   133→      setDownloads(list);
   134→      void estimateOfflineStorage().then(setStorageEst);
   135→    });
   136→  }, []);
   137→
   138→  useEffect(() => {
   139→    refreshDownloads();
   140→    const unsub = subscribeDownloads(refreshDownloads);
   141→    return () => unsub();
   142→  }, [refreshDownloads]);
   143→
   144→  // manifest shortcut / deep link → open the Downloads view directly
   145→  useEffect(() => {
   146→    try {
   147→      const v = new URLSearchParams(window.location.search).get('view');
   148→      if (v === 'downloads') {
   149→        setView('downloads-view');
   150→        window.history.replaceState({}, '', '/');
   151→      }
   152→    } catch {
   153→      /* ignore */
   154→    }
   155→  }, []);
   156→
   157→  const playDownload = useCallback((rec: SbDownload) => {
   158→    const q = new URLSearchParams({
   159→      videoId: rec.id,
   160→      batchId: rec.batchId || '',
   161→      subjectId: rec.subjectId || '',
   162→      schId: rec.schId || rec.id,
   163→      title: rec.title || 'PW Lecture',
   164→      batchName: rec.batchName || '',
   165→      vUrl: rec.vUrl || '',
   166→      ...(rec.realVideoId ? { lectureVideoId: rec.realVideoId } : {}),
   167→    });
   168→    window.open(`/player?${q.toString()}`, '_blank');
   169→  }, []);
   170→
   171→  // ==================================================================
   172→  // CATALOG (home view)
   173→  // ==================================================================
   174→  const loadCatalog = useCallback(async (q: string, p: number, isNew: boolean) => {
   175→    try {
   176→      if (isNew) {
   177→        setSearching(true);
   178→        setBatches([]);
   179→      }
   180→      setCatalogError('');
   181→      const res = await fetchCatalog(q, p, 20);
   182→      setBatches((prev) => (isNew ? res.data : [...prev, ...res.data]));
   183→      setHasMore(res.hasMore);
   184→      setPage(p);
   185→    } catch {
   186→      setCatalogError('Failed to load batches.');
   187→    } finally {
   188→      setSearching(false);
   189→    }
   190→  }, []);
   191→
   192→  useEffect(() => {
   193→    loadCatalog('', 1, true);
   194→  }, [loadCatalog]);
   195→
   196→  const handleSearchInput = (value: string) => {
   197→    setSearchTerm(value);
   198→    debounceTerm.current = value;
   199→    if (searchTimeout.current) clearTimeout(searchTimeout.current);
   200→    searchTimeout.current = setTimeout(() => {
   201→      loadCatalog(debounceTerm.current, 1, true);
   202→    }, 500);
   203→  };
   204→
   205→  const loadNextPage = () => {
   206→    loadCatalog(searchTerm, page + 1, false);
   207→  };
   208→
   209→  /** Catalog card click — open the real-data batch detail sheet. */
   210→  const openCatalogInfo = async (b: BatchCatalogItem) => {
   211→    setBatchInfoOpen(true);
   212→    setBatchInfoLoading(true);
   213→    setBatchInfo(b);
   214→    const res = await api.batchDetail(b._id);
   215→    if (res.success && res.data) setBatchInfo(res.data);
   216→    setBatchInfoLoading(false);
   217→  };
   218→
   219→  const enrollInBatch = (batchId: string) => {
   220→    enroll(batchId);
   221→    setCurrentBatchId(batchId);
   222→    setBatchInfoOpen(false);
   223→    nav('study-view');
   224→  };
   225→
   226→  const studyBatch = (batchId: string) => {
   227→    setBatchInfoOpen(false);
   228→    setCurrentBatchId(batchId);
   229→    nav('study-view');
   230→  };
   231→
   232→  // ==================================================================
   233→  // STUDY DASHBOARD
   234→  // ==================================================================
   235→  const refreshEnrolledDetails = useCallback(async () => {
   236→    const results: BatchDetailPublic[] = [];
   237→    for (const id of enrolled) {
   238→      const cached = batchDetailCache.current.get(id);
   239→      if (cached) {
   240→        results.push(cached);
   241→        continue;
   242→      }
   243→      const res = await api.batchDetail(id);
   244→      if (res.success && res.data) {
   245→        batchDetailCache.current.set(id, res.data);
   246→        results.push(res.data);
   247→      }
   248→    }
   249→    setEnrolledDetails(results);
   250→    setCurrentBatchId((cur) => {
   251→      if (cur && results.some((r) => r._id === cur)) return cur;
   252→      return results.length > 0 ? results[0]._id : null;
   253→    });
   254→  }, [enrolled]);
   255→
   256→  useEffect(() => {
   257→    refreshEnrolledDetails();
   258→  }, [refreshEnrolledDetails]);
   259→
   260→  const loadEvents = useCallback(async (batchId: string) => {
   261→    setEventsLoading(true);
   262→    const res = await api.todaySchedule(batchId);
   263→    setEvents(res.success && res.data ? res.data : []);
   264→    setEventsLoading(false);
   265→  }, []);
   266→
   267→  useEffect(() => {
   268→    if (currentBatchId) loadEvents(currentBatchId);
   269→    else setEvents([]);
   270→  }, [currentBatchId, loadEvents]);
   271→
   272→  const currentBatch = enrolledDetails.find((b) => b._id === currentBatchId) || null;
   273→
   274→  // ==================================================================
   275→  // ALL CLASSES — SUBJECTS & RESOURCES
   276→  // ==================================================================
   277→  const loadSubjects = useCallback(async (batchId: string) => {
   278→    setSubjectsLoading(true);
   279→    // Batch detail carries the real subject mapping (slug-addressed).
   280→    const res = await api.batchDetail(batchId);
   281→    if (res.success && res.data) {
   282→      batchDetailCache.current.set(batchId, res.data);
   283→      setSubjects(res.data.subjects || []);
   284→    } else {
   285→      setSubjects([]);
   286→    }
   287→    setSubjectsLoading(false);
   288→  }, []);
   289→
   290→  const loadResources = useCallback(async (batchId: string) => {
   291→    setResourcesLoading(true);
   292→    const res = await api.resourceTopics(batchId, '');
   293→    const topics = res.success && res.data ? (res.data as unknown as { data?: ChapterItemPublic[] }).data || [] : [];
   294→    setResourceTopics(topics);
   295→    setResourcesLoading(false);
   296→  }, []);
   297→
   298→  const openAllClasses = () => {
   299→    setResourcesTab('subjects');
   300→    nav('all-classes-view');
   301→    if (currentBatchId) loadSubjects(currentBatchId);
   302→  };
   303→
   304→  const switchClassesTab = (tab: 'subjects' | 'resources') => {
   305→    setResourcesTab(tab);
   306→    if (tab === 'subjects' && currentBatchId) loadSubjects(currentBatchId);
   307→    if (tab === 'resources' && currentBatchId) loadResources(currentBatchId);
   308→  };
   309→
   310→  // ==================================================================
   311→  // CHAPTERS
   312→  // ==================================================================
   313→  const loadChaptersList = useCallback(async (subjectId: string, tagType: 'UNITS' | 'STUDY_MATERIAL') => {
   314→    if (!currentBatchId) return;
   315→    setChaptersLoading(true);
   316→    const res = await api.subjectTopics(currentBatchId, subjectId, tagType);
   317→    const list = res.success && res.data ? (res.data as unknown as { data?: ChapterItemPublic[] }).data || [] : [];
   318→    setChapters(list);
   319→    setChaptersLoading(false);
   320→  }, [currentBatchId]);
   321→
   322→  const openSubject = (sub: BatchSubjectPublic) => {
   323→    setCurrentSubject({
   324→      id: sub.subjectId || sub._id,
   325→      internalId: sub.subjectId || sub._id,
   326→      name: sub.subject,
   327→      masterId: sub.masterId || '',
   328→      batchSubjectId: sub.batchSubjectId || '',
   329→    });
   330→    setChaptersTab('UNITS');
   331→    nav('chapters-view');
   332→    loadChaptersList(sub.subjectId || sub._id, 'UNITS');
   333→  };
   334→
   335→  const switchChaptersTab = (tagType: 'UNITS' | 'STUDY_MATERIAL') => {
   336→    setChaptersTab(tagType);
   337→    if (currentSubject) loadChaptersList(currentSubject.id, tagType);
   338→  };
   339→
   340→  // ==================================================================
   341→  // CONTENT
   342→  // ==================================================================
   343→  const loadContent = useCallback(async (subjectId: string, tagId: string, filter: string) => {
   344→    if (!currentBatchId) return;
   345→    setContentLoading(true);
   346→    if (filter === 'DPP_TEST') {
   347→      const subject = currentSubject;
   348→      const res = await api.dppList(currentBatchId, subject?.batchSubjectId || '', tagId);
   349→      const list = res.success && res.data ? (res.data as unknown as { data?: TestItemPublic[] }).data || [] : [];
   350→      setDppItems(list);
   351→      setContentNodes([]);
   352→    } else {
   353→      const res = await api.scheduleContent(currentBatchId, subjectId, tagId, filter);
   354→      const nodes = res.success && res.data ? (res.data as ContentNodePublic[]) : [];
   355→      setContentNodes(nodes);
   356→      setDppItems([]);
   357→    }
   358→    setContentLoading(false);
   359→  }, [currentBatchId, currentSubject]);
   360→
   361→  const openTopic = (t: ChapterItemPublic) => {
   362→    setCurrentTopic({ id: t._id, name: t.name });
   363→    setContentTab('ALL');
   364→    nav('content-view');
   365→    if (currentSubject) loadContent(currentSubject.id, t._id, 'ALL');
   366→  };
   367→
   368→  const openResourceTopic = (t: ChapterItemPublic) => {
   369→    if (!currentSubject) return;
   370→    setCurrentTopic({ id: t._id, name: t.name });
   371→    setContentTab('ALL');
   372→    nav('content-view');
   373→    loadContent(currentSubject.id, t._id, 'ALL');
   374→  };
   375→
   376→  const filterContentTab = (filter: string) => {
   377→    setContentTab(filter);
   378→    if (currentSubject && currentTopic) loadContent(currentSubject.id, currentTopic.id, filter);
   379→  };
   380→
   381→  // ==================================================================
   382→  // TESTS
   383→  // ==================================================================
   384→  const openAllTests = () => {
   385→    nav('all-tests-view');
   386→    if (!currentBatchId) return;
   387→    setTestsLoading(true);
   388→    api.testList(currentBatchId).then((res) => {
   389→      setTests(res.success && res.data ? res.data : []);
   390→      setTestsLoading(false);
   391→    }).catch(() => {
   392→      setTests([]);
   393→      setTestsLoading(false);
   394→    });
   395→  };
   396→
   397→  const openInstructions = async (t: TestItemPublic) => {
   398→    if (!currentBatchId) return;
   399→    setInstructions({ open: true, loading: true, title: t.name || 'Test', html: '' });
   400→    const res = await api.testDetail(t._id, currentBatchId);
   401→    const d = (res.data || {}) as Record<string, unknown>;
   402→    // upstream instructions arrive as {en: "<html>"} or a plain html string
   403→    const pick = (v: unknown): string => {
   404→      if (typeof v === 'string') return v;
   405→      if (v && typeof v === 'object') return String((v as { en?: string }).en || '');
   406→      return '';
   407→    };
   408→    const raw = pick(d.multiGeneralInstructions) || pick(d.generalInstructions);
   409→    const html = raw.replace(/<script[\s\S]*?<\/script>/gi, '');
   410→    setInstructions({ open: true, loading: false, title: t.name || 'Test', html: html || '<p>No instructions available.</p>' });
   411→  };
   412→
   413→  const launchQuiz = (p: { testId: string; scheduleId?: string; subjectId?: string; title: string }) => {
   414→    if (!currentBatchId) return;
   415→    const q = new URLSearchParams({
   416→      testId: p.testId,
   417→      batchId: currentBatchId,
   418→      scheduleId: p.scheduleId || '',
   419→      subjectId: p.subjectId || currentSubject?.masterId || '',
   420→      type: 'Start',
   421→    });
   422→    addXP(10, 'Attempted Test/DPP');
   423→    setDppOpen(false);
   424→    setTestFrame({ open: true, url: `/quiz?${q.toString()}`, title: p.title || 'Test Attempt' });
   425→  };
   426→
   427→  // ==================================================================
   428→  // MODALS
   429→  // ==================================================================
   430→  const openBatchInfo = async () => {
   431→    if (!currentBatchId) return;
   432→    setBatchInfoOpen(true);
   433→    setBatchInfoLoading(true);
   434→    const res = await api.batchDetail(currentBatchId);
   435→    setBatchInfo(res.success && res.data ? res.data : null);
   436→    setBatchInfoLoading(false);
   437→  };
   438→
   439→  const openNotesModal = async (scheduleId: string, subjectMasterId: string) => {
   440→    if (!currentBatchId) return;
   441→    setNotesOpen(true);
   442→    setNotesLoading(true);
   443→    setNoteRows([]);
   444→    const res = await api.scheduleDetail(currentBatchId, scheduleId, subjectMasterId);
   445→    const item = res.data as {
   446→      topic?: string;
   447→      name?: string;
   448→      homeworkIds?: Array<{ topic?: string; attachmentIds?: Array<{ baseUrl?: string; key?: string }> }>;
   449→    } | undefined;
   450→
   451→    const rows: AttachmentRow[] = [];
   452→    if (item) {
   453→      (item.homeworkIds || []).forEach((hw) => {
   454→        (hw.attachmentIds || []).forEach((att) => {
   455→          rows.push({ title: hw.topic || 'Class Note', url: imageUrl(att) });
   456→        });
   457→      });
   458→    }
   459→    setNoteRows(rows);
   460→    setNotesLoading(false);
   461→  };
   462→
   463→  const handleUnenroll = (batchId: string) => {
   464→    if (!window.confirm('Are you sure you want to unenroll from this batch?')) return;
   465→    unenroll(batchId);
   466→    setBatchInfoOpen(false);
   467→    if (currentBatchId === batchId) {
   468→      const remaining = enrolled.filter((id) => id !== batchId);
   469→      if (remaining.length > 0) setCurrentBatchId(remaining[0]);
   470→      else {
   471→        setCurrentBatchId(null);
   472→        nav('home-view');
   473→      }
   474→    }
   475→  };
   476→
   477→  // ==================================================================
   478→  // RENDER HELPERS
   479→  // ==================================================================
   480→  const renderLectureCard = (node: ContentNodePublic) => {
   481→    const d = node.data;
   482→    const thumb = d.videoDetails?.image || DEFAULT_THUMB;
   483→    const title = d.topic || d.name || 'Lecture';
   484→    const date = d.date ? new Date(d.date).toLocaleDateString('en-GB', { day: 'numeric', month: 'short', year: 'numeric' }) : '';
   485→    const vId = d._id || d.videoDetails?._id || node._id;
   486→    const master = currentSubject?.masterId || currentSubject?.id || '';
   487→    const rawVUrl = d.url || d.videoDetails?.videoUrl || '';
   488→    const realVideoId = d.videoDetails?._id && d.videoDetails._id !== vId ? d.videoDetails._id : '';
   489→    const isWatched = watched.includes(vId);
   490→    const bName = currentBatch?.name || 'PW Batch';
   491→    const base = node.type === 'DPP_VIDEOS' ? '/player' : '/player';
   492→    const playerUrl = `${base}?videoId=${encodeURIComponent(vId)}&batchId=${encodeURIComponent(currentBatchId || '')}&subjectId=${encodeURIComponent(master)}&schId=${encodeURIComponent(node._id)}&title=${encodeURIComponent(title)}&batchName=${encodeURIComponent(bName)}&vUrl=${encodeURIComponent(rawVUrl)}${realVideoId ? `&lectureVideoId=${encodeURIComponent(realVideoId)}` : ''}`;
   493→    const hasAttachments = d.homeworkIds && d.homeworkIds.length > 0;
   494→
   495→    return (
   496→      <div className="content-card" key={node._id} style={{ flexDirection: 'column', alignItems: 'stretch' }}>
   497→        <div className="d-flex gap-3 mb-3">
   498→          <img src={thumb} className="lecture-thumb" alt={title} />
   499→          <div className="flex-grow-1">
   500→            <span style={{ background: '#eff6ff', color: '#2563eb', fontSize: '0.7rem', fontWeight: 800, padding: '2px 6px', borderRadius: 4, display: 'inline-block', marginBottom: 5 }}>
   501→              {node.type === 'DPP_VIDEOS' ? 'DPP VIDEO' : 'VIDEO'}
   502→            </span>
   503→            <div style={{ fontWeight: 700, fontSize: '1rem', color: 'var(--sb-text-dark)' }}>{title}</div>
   504→            <div style={{ color: 'var(--sb-text-muted)', fontSize: '0.8rem', marginTop: 5 }}>
   505→              <i className="fas fa-clock"></i> {date}
   506→            </div>
   507→          </div>
   508→        </div>
   509→        <div className="d-flex gap-2">
   510→          <button
   511→            className={`btn-check-done ${isWatched ? 'active' : ''}`}
   512→            title="Mark as Watched"
   513→            onClick={() => {
   514→              const added = toggleWatched(vId);
   515→              if (added) addXP(10, 'Watched Video');
   516→            }}
   517→          >
   518→            <i className="fas fa-check"></i>
   519→          </button>
   520→          <button className="btn-play" style={{ flexGrow: 1 }} onClick={() => window.open(playerUrl, '_blank')}>
   521→            <i className="fas fa-play"></i> Watch
   522→          </button>
   523→          {hasAttachments ? (
   524→            <button className="btn-notes" onClick={() => openNotesModal(node._id, master)}>
   525→              <i className="fas fa-file-alt"></i>
   526→            </button>
   527→          ) : null}
   528→        </div>
   529→      </div>
   530→    );
