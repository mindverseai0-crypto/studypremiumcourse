    46→const asParams = (v: unknown): Params =>
    47→  v && typeof v === "object" && !Array.isArray(v) ? (v as Params) : {};

    49→const str = (v: unknown, fallback = ""): string =>
    50→  v === undefined || v === null ? fallback : String(v);

    57→/** Payload may arrive as object or JSON string. */
    58→function parsePayload(payload: unknown): unknown {
    59→  if (typeof payload === "string") {
    60→    try {
    61→      return JSON.parse(payload);
    62→    } catch {
    63→      return payload;
    64→    }
    65→  }
    66→  return payload;
    67→}

    77→/** PDF attachments are baked as /api/lxpdf links that resolve the real file. */
    78→function pdfAttachmentRef(batchId: string, subjectId: string, pdfId: string, attachmentId: string) {
    79→  const q = `batchId=${encodeURIComponent(batchId)}&subjectId=${encodeURIComponent(
    80→    subjectId,
    81→  )}&pdfId=${encodeURIComponent(pdfId)}&attachmentId=${encodeURIComponent(attachmentId)}`;
    82→  return { baseUrl: `/api/lxpdf?${q}`, key: "", _id: attachmentId };
    83→}

    85→function subjectRef(s: unknown): { _id: string; slug: string; name: string } {
    86→  if (s && typeof s === "object") {
    87→    const o = s as { _id?: string; slug?: string; name?: string };
    88→    return { _id: o._id || "", slug: o.slug || o._id || "", name: o.name || "" };
    89→  }
    90→  return { _id: String(s || ""), slug: String(s || ""), name: "" };
    91→}

    93→function eventStatus(item: LxScheduleItem, startIso: string, endIso: string): string {
    94→  const tag = String(item.tag || "").toLowerCase();
    95→  if (tag === "live" || tag === "live now") return "LIVE";
    96→  if (tag === "upcoming") return "UPCOMING";
    97→  if (tag === "ended" || tag === "cancelled") return tag === "cancelled" ? "CANCELLED" : "ENDED";
    98→  const now = Date.now();
    99→  const s = new Date(startIso || "").getTime();
   100→  const e = new Date(endIso || "").getTime();
   101→  if (Number.isFinite(s) && Number.isFinite(e)) {
   102→    if (now >= s && now <= e) return "LIVE";
   103→    if (now < s) return "UPCOMING";
   104→  }
   105→  return "ENDED";
   106→}

   136→function contentNodesFromItems(
   137→  batchId: string,
   138→  subjectId: string,
   139→  items: LxScheduleItem[],
   140→  kind: "LECTURE" | "DPP_VIDEOS" | "NOTES",
   141→): Array<{ _id: string; type: string; data: Record<string, unknown> }> {
   142→  const nodes: Array<{ _id: string; type: string; data: Record<string, unknown> }> = [];
   143→  for (const item of items) {
   144→    const pdfId = item._id || "";
   145→
   146→    if (kind === "LECTURE" || kind === "DPP_VIDEOS") {
   147→      const vd = item.videoDetails || {};
   148→      nodes.push({
   149→        _id: item._id || "",
   150→        type: kind,
   151→        data: {
   152→          _id: item._id || "",
   153→          topic: item.topic || vd.name || item.name || "Lecture",
   154→          date: item.date || item.startTime || "",
   155→          urlType: item.urlType || "",
   156→          url: item.url || "",
   157→          videoDetails: {
   158→            _id: item._id || "",
   159→            image: vd.image || item.previewImageUrl || "",
   160→            videoUrl: "",
   161→            duration: vd.duration || "",
   162→          },
   163→          homeworkIds: asArray(item.homeworkIds).flatMap((hw) => {
   164→            const h = hw as { _id?: string; topic?: string; attachmentIds?: unknown[] };
   165→            return asArray(h.attachmentIds).map((att) => {
   166→              const a = att as { _id?: string };
   167→              return {
   168→                _id: h._id || "",
   169→                topic: h.topic || "Class Note",
   170→                attachmentIds: [pdfAttachmentRef(batchId, subjectId, pdfId, a._id || "")],
   171→              };
   172→            });
   173→          }),
   174→          exerciseIds: [],
   175→        },
   176→      });
   177→      continue;
   178→    }
   179→
   180→    // NOTES / DppNotes → PDF cards
   181→    const atts: Array<Record<string, unknown>> = [];
   182→    let title = item.topic || item.name || "";
   183→    const homeworkSources = [
   184→      ...asArray(item.homeworkIds),
   185→      ...asArray(item.dpp?.homeworkIds),
   186→    ];
   187→    for (const hw of homeworkSources) {
   188→      const h = hw as { _id?: string; topic?: string; attachmentIds?: unknown[] };
   189→      if (!title && h.topic) title = h.topic;
   190→      for (const att of asArray(h.attachmentIds)) {
   191→        const a = att as { _id?: string; key?: string; baseUrl?: string; name?: string };
   192→        if (!a?._id) continue;
   193→        if (!title) title = a.name || "";
   194→        // some attachments already carry a real static.pw.live key
   195→        const direct = a.baseUrl && a.key ? `${a.baseUrl}${a.key}` : "";
   196→        const ref = direct
   197→          ? { baseUrl: direct, key: "", _id: a._id }
   198→          : pdfAttachmentRef(batchId, subjectId, pdfId, a._id);
   199→        atts.push({
   200→          _id: a._id,
   201→          topic: h.topic || a.name || item.topic || item.name || "Study Material",
   202→          attachmentIds: [ref],
   203→        });
   204→      }
   205→    }
   206→    if (atts.length === 0) continue;
   207→    nodes.push({
   208→      _id: item._id || "",
   209→      type: "NOTES",
   210→      data: {
   211→        _id: item._id || "",
   212→        topic: title || "Study Material",
   213→        date: item.date || item.startTime || "",
   214→        homeworkIds: atts,
   215→      },
   216→    });
   217→  }
   218→  return nodes;
   219→}

   221→function resolveContentKind(contentType: string): "LECTURE" | "DPP_VIDEOS" | "NOTES" | null {
   222→  switch ((contentType || "ALL").toUpperCase()) {
   223→    case "ALL":
   224→    case "LECTURE":
   225→      return "LECTURE";
   226→    case "DPP_VIDEOS":
   227→      return "DPP_VIDEOS";
   228→    case "NOTES":
   229→    case "DPP_PDF":
   230→      return "NOTES";
   231→    default:
   232→      return null;
   233→  }

   249→/** base64url-encode a signed query for the stateless /s/<sig>/ relay path. */
   250→function sigPathSegment(signedQuery: string): string {
   251→  const q = signedQuery.startsWith("?") ? signedQuery.slice(1) : signedQuery;
   252→  if (!q) return "";
   253→  return Buffer.from(q, "utf8").toString("base64url");
   254→}

   256→/** Resolve a video → same-origin playable payload (dash/hls/youtube/mp4). */
   257→async function resolveVideoPayload(childId: string, batchId: string, subjectId: string) {
   258→  const vid = await lxVideoUrl(batchId, subjectId, childId);
   259→  if (vid?.url) {
   260→    const url = vid.url.trim();
   261→    if (/\.mpd(\?|$)/i.test(url)) {
   262→      const folder = folderFromUrl(url);
   263→      const sig = sigPathSegment(vid.signedUrl || "");
   264→      if (folder && sig) registerSignedFolder(folder, vid.signedUrl || "");
   265→      let clearKeys: Record<string, string> | null = null;
   266→      try {
   267→        const mpdUrl = `${url}${vid.signedUrl || ""}`;
   268→        const controller = new AbortController();
   269→        const timer = setTimeout(() => controller.abort(), 30000);
   270→        const mpdRes = await fetch(mpdUrl, {
   271→          cache: "no-store",
   272→          signal: controller.signal,
   273→          headers: { "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64)" },
   274→        });
   275→        clearTimeout(timer);
   276→        if (mpdRes.ok) {
   277→          const text = await mpdRes.text();
   278→          const kidMatch = text.match(/default_KID="([0-9a-fA-F-]+)"/);
   279→          if (kidMatch) {
   280→            clearKeys = await lxOtp(kidMatch[1]);
   281→          }
   282→        }
   283→      } catch {
   284→        /* keys unavailable — player will surface a decrypt error */
   285→      }
   286→      // Stateless relay URL: relative DASH segments inherit the signature.
   287→      const manifestUrl = folder
   288→        ? `/api/pw/${folder}${sig ? `/s/${sig}` : ""}/master.mpd`
   289→        : "";
   290→      if (!manifestUrl) return null;
   291→      return {
   292→        kind: "dash",
   293→        url: manifestUrl,
   294→        signedUrl: "",
   295→        clearKeys,
   296→        poster: "",
   297→        drmDetails: null,
   298→      };
   299→    }
   300→    if (/\.m3u8(\?|$)/i.test(url)) {
   301→      const folder = folderFromUrl(url);
   302→      const sig = sigPathSegment(vid.signedUrl || "");
   303→      if (!folder) return null;
   304→      if (sig) registerSignedFolder(folder, vid.signedUrl || "");
   305→      // Root-level LIVE manifest (…/index.m3u8?Signature=… — no uuid folder):
   306→      // each live class gets its own channel like this, on its own live host.
   307→      // The host is embedded in the relay path (validated against an
   308→      // allowlist in pw-cdn); the manifest itself is the relayed asset and
   309→      // segments resolve as its siblings through the same /s/<sig>/ path.
   310→      const isUuidFolder = /^[a-f0-9-]{16,}$/i.test(folder);
   311→      if (!isUuidFolder && /^[A-Za-z0-9._-]+\.m3u8$/i.test(folder)) {
   312→        const liveHost = hostFromUrl(url);
   313→        return {
   314→          kind: "hls",
   315→          url: `/api/pw/h/${liveHost}/${folder}${sig ? `/s/${sig}` : ""}`,
   316→          signedUrl: "",
   317→          clearKeys: null,
   318→          poster: "",
   319→          drmDetails: null,
   320→        };
   321→      }
   322→      return {
   323→        kind: "hls",
   324→        url: `/api/pw/${folder}${sig ? `/s/${sig}` : ""}/master.m3u8`,
   325→        signedUrl: "",
   326→        clearKeys: null,
   327→        poster: "",
   328→        drmDetails: null,
   329→      };
   330→    }
   331→    if (/youtube\.com|youtu\.be/i.test(url)) {
   332→      return { kind: "youtube", url, signedUrl: "", clearKeys: null, poster: "", drmDetails: null };
   333→    }
   334→    // plain file (mp4 etc.) inside the signed folder
   335→    const folder = folderFromUrl(url);
   336→    const sig = sigPathSegment(vid.signedUrl || "");
   337→    if (folder) {
   338→      if (sig) registerSignedFolder(folder, vid.signedUrl || "");
   339→      const fileName = url.split("/").pop() || "video.mp4";
   340→      return {
   341→        kind: "mp4",
   342→        url: `/api/pw/${folder}${sig ? `/s/${sig}` : ""}/${fileName}`,
   343→        signedUrl: "",
   344→        clearKeys: null,
   345→        poster: "",
   346→        drmDetails: null,
   347→      };
   348→    }
   349→    return { kind: "mp4", url, signedUrl: "", clearKeys: null, poster: "", drmDetails: null };
   350→  }
   351→
   352→  // No signed stream — maybe a youtube lecture.
   353→  if (subjectId) {
   354→    const detail = await lxScheduleDetail(batchId, subjectId, childId);
   355→    const url = detail?.url || "";
   356→    if (url && /youtube\.com|youtu\.be/i.test(url)) {
   357→      return { kind: "youtube", url, signedUrl: "", clearKeys: null, poster: "", drmDetails: null };
   358→    }
   359→  }
   360→  return null;
   361→}

   366→export async function POST(request: NextRequest) {
   367→  try {
   368→    const body = (await request.json().catch(() => ({}))) as ActionBody;
   369→    const action = typeof body.action === "string" ? body.action : "";
   370→    const params = asParams(body.params);
   371→    const payload = parsePayload(body.payload);
   372→
   373→    switch (action) {
   374→      case "pw_btch_dtl": {
   375→        const batchId = str(params.batchId).trim();
   376→        if (!batchId) return NextResponse.json({ success: false }, { headers: CORS_HEADERS });
   377→        const d = await lxBatchDetails(batchId);
   378→        if (!d) return NextResponse.json({ success: false }, { headers: CORS_HEADERS });
   379→
   380→        const exams = Array.isArray(d.exam)
   381→          ? (d.exam as unknown[])
   382→              .map((e) => (typeof e === "string" ? e : (e as { name?: string })?.name || ""))
   383→              .filter(Boolean)
   384→          : [];
   385→        const subjects = Array.isArray(d.subjects) ? d.subjects : [];
   386→        return NextResponse.json(
   387→          {
   388→            success: true,
   389→            data: {
   390→              _id: d._id || batchId,
   391→              id: d.slug || batchId,
   392→              name: d.name || "Batch",
   393→              byName: d.byName || "",
   394→              description: typeof d.description === "string" ? d.description : "",
   395→              imageUrl: d.previewImage || d.previewImageUrl || "",
   396→              exams,
   397→              language: d.language || "",
   398→              className:
   399→                typeof d.class === "string"
   400→                  ? d.class
   401→                  : Array.isArray(d.class)
   402→                    ? (d.class as unknown[]).map((c) => String(c)).join(", ")
   403→                    : "",
   404→              amount: typeof d.feeTotal === "number" ? d.feeTotal : null,
   405→              startDate: d.startDate || "",
   406→              endDate: d.endDate || "",
   407→              // subject mapping — subjectId = SLUG (SubjectInfo/TopicInfo are
   408→              // slug-addressed), batchSubjectId & masterId for dpp/schedule apis
   409→              subjects: subjects.map((s) => ({
   410→                _id: s.slug || s.subjectId || s._id || "",
   411→                subject: s.subject || "",
   412→                subjectId: s.slug || s.subjectId || "",
   413→                batchSubjectId: s._id || "",
   414→                masterId: s.subjectId || "",
   415→              })),
   416→            },
   417→          },
   418→          { headers: CORS_HEADERS },
   419→        );
   420→      }
   421→
   422→      case "pw_sub_topics": {
   423→        const batchId = str(params.batchId).trim();
   424→        const subjectId = str(params.subjectId).trim();
   425→        const page = num(params.page, 1);
   426→        if (!batchId || !subjectId) {
   427→          return NextResponse.json({ success: true, data: { data: [] } }, { headers: CORS_HEADERS });
   428→        }
   429→        try {
   430→          const chapters = await lxChapters(batchId, subjectId, page);
   431→          const data = chapters.map((c, i) => ({
   432→            _id: c._id || "",
   433→            name: c.name || "Chapter",
   434→            typeId: subjectId,
   435→            notes: Number(c.notes || 0),
   436→            videos: Number(c.videos || c.lectureVideos || 0),
   437→            exercises: Number(c.exercises || 0),
   438→            order: Number.isFinite(c.displayOrder) ? Number(c.displayOrder) : i,
   439→          }));
   440→          return NextResponse.json({ success: true, data: { data } }, { headers: CORS_HEADERS });
   441→        } catch {
   442→          return NextResponse.json({ success: true, data: { data: [] } }, { headers: CORS_HEADERS });
   443→        }
   444→      }
   445→
   446→      case "pw_res_topics":
   447→        // upstream has no separate resource folders
   448→        return NextResponse.json({ success: true, data: { data: [] } }, { headers: CORS_HEADERS });
   449→
   450→      case "pw_sch_cntnt": {
   451→        const batchId = str(params.batchId).trim();
   452→        const subjectId = str(params.subjectId).trim();
   453→        const tagId = str(params.tagId).trim();
   454→        const contentType = str(params.contentType, "ALL");
   455→        const page = num(params.page, 1);
   456→        if (!batchId || !subjectId || !tagId) {
   457→          return NextResponse.json({ success: true, data: [] }, { headers: CORS_HEADERS });
   458→        }
   459→        const kind = resolveContentKind(contentType);
   460→        if (!kind) return NextResponse.json({ success: true, data: [] }, { headers: CORS_HEADERS });
   461→        try {
   462→          const items = await lxTopicContent(batchId, subjectId, tagId, upstreamContentType(contentType), page);
   463→          const nodes = contentNodesFromItems(batchId, subjectId, items, kind);
   464→          return NextResponse.json({ success: true, data: nodes }, { headers: CORS_HEADERS });
   465→        } catch {
   466→          return NextResponse.json({ success: true, data: [] }, { headers: CORS_HEADERS });
   467→        }
   468→      }
   469→
   470→      case "pw_dpp_lst": {
   471→        const batchId = str(params.batchId).trim();
   472→        const batchSubjectId = str(params.batchSubjectId).trim();
   473→        const chapterId = str(params.chapterId).trim();
   474→        const page = num(params.page, 1);
   475→        const limit = num(params.limit, 20);
   476→        if (!batchId || !batchSubjectId || !chapterId) {
   477→          return NextResponse.json({ success: true, data: { data: [] } }, { headers: CORS_HEADERS });
   478→        }
   479→        try {
   480→          const items = await lxDppList(batchId, batchSubjectId, chapterId, page, limit);
   481→          const data = items
   482→            .map((it) => {
   483→              const q = it.dppQuizDetails;
   484→              const t = q?.test;
   485→              if (!t?._id) return null;
   486→              return {
   487→                _id: t._id,
   488→                name: t.name || "DPP",
   489→                totalQuestions: Number(t.totalQuestions || 0),
   490→                totalMarks: Number(t.totalMarks || 0),
   491→                duration: Number(t.maxDuration || 0) * 60 || 1800,
   492→                testActivityStatus: "Not Started",
   493→                testStudentMapping: null,
   494→                dppQuizDetails: {
   495→                  contentId: q?.contentId || t._id,
   496→                  scheduleId: q?.scheduleId || "",
   497→                  test: {
   498→                    _id: t._id,
   499→                    name: t.name || "DPP",
   500→                    totalQuestions: Number(t.totalQuestions || 0),
   501→                    totalMarks: Number(t.totalMarks || 0),
   502→                  },
   503→                  testStudentMapping: { testActivityStatus: q?.tag || "Start" },
   504→                },
   505→              };
   506→            })
   507→            .filter(Boolean);
   508→          return NextResponse.json({ success: true, data: { data } }, { headers: CORS_HEADERS });
   509→        } catch {
   510→          return NextResponse.json({ success: true, data: { data: [] } }, { headers: CORS_HEADERS });
   511→        }
   512→      }
   513→
   514→      case "pw_dpp_start": {
   515→        // Auth-gated upstream: anonymous calls fail → surfaced honestly.
   516→        const testId = str(params.testId).trim();
   517→        const batchId = str(params.batchId).trim();
   518→        if (!testId || !batchId) {
   519→          return NextResponse.json(
   520→            { success: false, error: "testId and batchId are required" },
   521→            { headers: CORS_HEADERS },
   522→          );
   523→        }
   524→        try {
   525→          const data = await lxStartDpp({
   526→            testId,
   527→            batchId,
   528→            type: str(params.type, "Start"),
   529→            scheduleId: str(params.scheduleId),
   530→            batchScheduleId: str(params.batchScheduleId) || str(params.scheduleId),
   531→            subjectId: str(params.subjectId),
   532→          });
   533→          return NextResponse.json({ success: true, data }, { headers: CORS_HEADERS });
   534→        } catch (e) {
   535→          return NextResponse.json(
   536→            { success: false, error: e instanceof Error ? e.message : "Failed to start DPP" },
   537→            { headers: CORS_HEADERS },
   538→          );
   539→        }
   540→      }
   541→
   542→      case "pw_test_lst": {

   721→
   722→      case "pw_test_map": {

   755→
   756→/** Flatten test ids from scalars / arrays / objects ({testId}|{_id}|{id}). */
   757→function collectTestIds(source: unknown, out: string[], seen: Set<string>): void {
   758→  if (source === undefined || source === null) return;
   759→  if (Array.isArray(source)) {
   760→    source.forEach((item) => collectTestIds(item, out, seen));
   761→    return;
   762→  }
   763→  if (typeof source === "object") {
   764→    const o = source as Record<string, unknown>;
   765→    collectTestIds(o.testId ?? o._id ?? o.id, out, seen);
   766→    return;
   767→  }
   768→  const s = String(source).trim();
   769→  if (s && !seen.has(s)) {
   770→    seen.add(s);
   771→    out.push(s);
   772→  }
   773→}
