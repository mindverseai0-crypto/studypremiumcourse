    31→ * same-origin /api/stream proxy) while the signed policy is valid; falls
    32→ * back to the self-hosted clips afterwards.
    33→ */
    34→function lectureVideoUrl(seed: string): string {

    38→  }
    39→  return SAMPLE_VIDEOS[hashSeed(seed) % SAMPLE_VIDEOS.length];
    40→}

    96→function loadCatalog(): CatalogBatch[] {
    97→  if (catalogCache) return catalogCache;
    98→  const file = path.join(process.cwd(), 'src', 'data', 'all_batches.json');
    99→  try {
   100→    const raw = fs.readFileSync(file, 'utf-8');
   101→    const parsed = JSON.parse(raw);
   102→    catalogCache = Array.isArray(parsed) ? (parsed as CatalogBatch[]) : [];
   103→  } catch {
   104→    catalogCache = [];
   105→  }
   106→  return catalogCache;
   107→}

   127→// ---------------------------------------------------------------------------
   128→// Catalog search — 20/page, search by name / id / byName.
   129→// (Catalog rows keep subjects empty: building content bundles per row would
   130→// be wasteful; subjects are attached on pw_btch_dtl / single-batch detail.)
   131→// ---------------------------------------------------------------------------
   132→export function searchCatalog(query: string, page: number, limit: number): CatalogResult {
   133→  const catalog = loadCatalog();
   134→  const q = (query || '').trim().toLowerCase();
   135→
   136→  const filtered = q
   137→    ? catalog.filter(
   138→        (b) =>
   139→          b.name.toLowerCase().includes(q) ||
   140→          b.batch_id.toLowerCase().includes(q) ||
   141→          b.id.toLowerCase().includes(q) ||
   142→          b.byName.toLowerCase().includes(q),
   143→      )
   144→    : catalog;
   145→
   146→  const safeLimit = Math.min(100, Math.max(1, limit));
   147→  const safePage = Math.max(1, page);
   148→  const start = (safePage - 1) * safeLimit;
   149→  const slice = filtered.slice(start, start + safeLimit);
   150→
   151→  return {
   152→    total: filtered.length,
   153→    page: safePage,
   154→    limit: safeLimit,
   155→    hasMore: start + safeLimit < filtered.length,
   156→    data: slice.map(toDetail),
   157→  };
   158→}

   163→export function getBatchDetail(batchId: string): BatchDetail | null {
   164→  const catalog = loadCatalog();
   165→  const id = (batchId || '').trim().toLowerCase();
   166→  if (!id) return null;
   167→  const found = catalog.find((b) => b.batch_id.toLowerCase() === id);
   168→  if (!found) return null;
   169→  const detail = toDetail(found);
   170→  detail.subjects = getSubjects(found.batch_id);
   171→  return detail;
   172→}

   262→function docFileId(kind: 'NOTE' | 'DPP', title: string, subject: string, chapter: string): string {
   263→  const payload = JSON.stringify({ k: kind, t: title, s: subject, c: chapter });
   264→  const b64 = Buffer.from(payload, 'utf-8').toString('base64url');
   265→  return `d_${b64}`;
   266→}

   268→function fmtDuration(rng: ReturnType<typeof makeRng>): string {
   269→  const h = rng.int(0, 2);
   270→  const m = rng.int(8, 55);
   271→  return h > 0 ? `${h}h ${m}m` : `${m}m`;
   272→}

   274→/** Spread n lecture dates evenly over the batch validity window (capped at now + 30d). */
   275→function lectureDates(startDate: string, endDate: string, n: number): string[] {
   276→  const now = Date.now();
   277→  let start = new Date(startDate || '').getTime();
   278→  let end = new Date(endDate || '').getTime();
   279→  if (!Number.isFinite(start)) start = now - 90 * 86400000;
   280→  if (!Number.isFinite(end)) end = start + 120 * 86400000;
   281→  start = Math.max(start, now - 90 * 86400000);
   282→  end = Math.min(end, now + 30 * 86400000);
   283→  if (end <= start) end = start + 86400000;
   284→  const out: string[] = [];
   285→  for (let i = 0; i < n; i++) {
   286→    const frac = (i + 1) / (n + 1);
   287→    out.push(new Date(start + frac * (end - start)).toISOString());
   288→  }
   289→  return out;
   290→}

   292→function pad(n: number): string {
   293→  return n < 10 ? `0${n}` : String(n);

   296→function getBatchContent(batchId: string): BatchContent | null {
   297→  const batch = findBatch(batchId);
   298→  if (!batch) return null;
   299→
   300→  const cached = bundleCache.get(batch.batch_id);
   301→  if (cached) return cached;
   302→
   303→  const short = isShortCourse(batch.name, batch.byName);
   304→  const templates: SubjectTemplate[] = pickSubjects({
   305→    name: batch.name,
   306→    byName: batch.byName,
   307→    exam: Array.isArray(batch.exam) ? batch.exam : [],
   308→    className: batch.class || '',
   309→  });
   310→
   311→  const subjects: SubjectData[] = templates.map((tpl) => {
   312→    const subjectId = stableId('sub', batch.batch_id, tpl.subject);
   313→    const rngChapters = makeRng('chapters', batch.batch_id, tpl.subject);
   314→
   315→    const bank = tpl.chapters;
   316→    const chapterCount = short ? rngChapters.int(5, 8) : Math.min(bank.length, rngChapters.int(8, 14));
   317→    const chapters: ChapterRef[] = [];
   318→    const lectures: LectureRef[] = [];
   319→    const notes: NotesRef[] = [];
   320→    const dpps: DppRef[] = [];
   321→
   322→    const pickedChapters = short ? rngChapters.sample(bank, chapterCount) : bank.slice(0, chapterCount);
   323→
   324→    pickedChapters.forEach((chapterName, ci) => {
   325→      const chapterId = stableId('ch', batch.batch_id, tpl.subject, chapterName);
   326→      const rng = makeRng('chapter', batch.batch_id, chapterName);
   327→
   328→      const lectureCount = short ? rng.int(3, 5) : rng.int(4, 9);
   329→      const notesCount = short ? 1 : rng.int(1, 2);
   330→      const dppCount = short ? rng.int(1, 3) : rng.int(2, 4);
   331→
   332→      chapters.push({
   333→        id: chapterId,
   334→        name: chapterName,
   335→        subjectId,
   336→        subjectName: tpl.subject,
   337→        lectureCount,
   338→        notesCount,
   339→        dppCount,
   340→      });
   341→
   342→      const dates = lectureDates(batch.start_date, batch.end_date, lectureCount);
   343→      const subtopics = rng.shuffle(LECTURE_SUBTOPICS);
   344→
   345→      // DPP refs for this chapter (shared across its lectures).
   346→      const chapterDpps: DppRef[] = [];
   347→      for (let di = 0; di < dppCount; di++) {
   348→        const dppId = stableId('dpp', batch.batch_id, chapterId, di);
   349→        const questions = short ? rng.int(5, 8) : rng.int(8, 15);
   350→        chapterDpps.push({
   351→          testId: dppId,
   352→          name: `DPP ${pad(di + 1)} : ${chapterName}`,
   353→          chapterId,
   354→          chapterName,
   355→          subjectId,
   356→          subjectName: tpl.subject,
   357→          totalQuestions: questions,
   358→          totalMarks: questions * 4,
   359→          docId: docFileId('DPP', `${chapterName} — DPP ${pad(di + 1)}`, tpl.subject, chapterName),
   360→        });
   361→      }
   362→      dpps.push(...chapterDpps);
   363→
   364→      for (let li = 0; li < lectureCount; li++) {
   365→        const nodeId = stableId('lec', batch.batch_id, chapterId, li);
   366→        const videoId = stableId('vid', batch.batch_id, chapterId, li);
   367→        const subtopic = subtopics[li % subtopics.length];
   368→        const linked = chapterDpps.length > 0 ? [chapterDpps[li % chapterDpps.length]] : [];
   369→        lectures.push({
   370→          nodeId,
   371→          videoId,
   372→          topic: `Lecture ${pad(li + 1)} : ${subtopic}`,
   373→          subtopic,
   374→          date: dates[li],
   375→          duration: fmtDuration(rng),
   376→          chapterId,
   377→          chapterName,
   378→          subjectId,
   379→          subjectName: tpl.subject,
   380→          subjectTeacher: '', // filled after teacher selection below
   381→          notesDocId: docFileId('NOTE', `${chapterName} — Class Notes`, tpl.subject, chapterName),
   382→          dppRefs: linked.map((d) => ({ id: d.testId, docId: d.docId, title: d.name })),
   383→        });
   384→      }
   385→
   386→      for (let ni = 0; ni < notesCount; ni++) {
   387→        const title = STUDY_MATERIAL_TITLES[(ci + ni) % STUDY_MATERIAL_TITLES.length];
   388→        notes.push({
   389→          nodeId: stableId('nt', batch.batch_id, chapterId, ni),
   390→          topic: `${chapterName} : ${title}`,
   391→          chapterId,
   392→          chapterName,
   393→          docId: docFileId('NOTE', `${chapterName} — ${title}`, tpl.subject, chapterName),
   394→          date: dates[Math.min(ni, dates.length - 1)] ?? new Date().toISOString(),
   395→        });
   396→      }
   397→    });
   398→
   399→    const t = teacherFor([batch.batch_id, tpl.subject]);
   400→    const teacher = { name: t.name, qualification: t.qualification, experience: t.experience };
   401→    for (const l of lectures) l.subjectTeacher = teacher.name;
   402→
   403→    return {
   404→      id: subjectId,
   405→      name: tpl.subject,
   406→      icon: tpl.icon,
   407→      chapters,
   408→      lectures,
   409→      notes,
   410→      dpps,
   411→      teacher,
   412→    };
   413→  });
   414→
   415→  // Part tests & full mocks
   416→  const rngTests = makeRng('tests', batch.batch_id);
   417→  const bigTests: BigTestRef[] = [];
   418→  const partCount = short ? rngTests.int(1, 2) : rngTests.int(2, 4);
   419→  for (let i = 0; i < partCount; i++) {
   420→    const subjectName = subjects.length > 0 ? subjects[i % subjects.length].name : 'General';
   421→    const qs = rngTests.int(12, 18);
   422→    bigTests.push({
   423→      testId: stableId('tst', batch.batch_id, 'part', i),
   424→      name: `Part Test ${pad(i + 1)} : ${subjectName}`,
   425→      subjectName,
   426→      totalQuestions: qs,
   427→      totalMarks: qs * 4,
   428→      duration: Math.max(30, Math.round((qs * 1.5) / 5) * 5),
   429→      kind: 'PART',
   430→    });
   431→  }
   432→  const mockCount = short ? rngTests.int(1, 2) : rngTests.int(2, 3);
   433→  for (let i = 0; i < mockCount; i++) {
   434→    const qs = rngTests.int(20, 28);
   435→    bigTests.push({
   436→      testId: stableId('tst', batch.batch_id, 'mock', i),
   437→      name: `Full Syllabus Mock ${pad(i + 1)}`,
   438→      subjectName: subjects.length > 0 ? subjects[i % subjects.length].name : 'General',
   439→      totalQuestions: qs,
   440→      totalMarks: qs * 4,
   441→      duration: Math.max(45, Math.round((qs * 1.5) / 5) * 5),
   442→      kind: 'MOCK',
   443→    });
   444→  }
   445→
   446→  const bundle: BatchContent = { batch, short, subjects, bigTests };
   447→  if (bundleCache.size >= BUNDLE_CACHE_MAX) {
   448→    const first = bundleCache.keys().next().value;
   449→    if (first) bundleCache.delete(first);
   450→  }
   451→  bundleCache.set(batch.batch_id, bundle);
   452→  return bundle;
   453→}

   463→export interface ChapterItem {
   464→  _id: string;
   465→  name: string;
   466→  typeId: string;

   472→export function getChapters(
   473→  batchId: string,
   474→  subjectId: string,
   475→  tagType: string,
   476→  page = 1,
   477→  limit = 50,
   478→): ChapterItem[] {
   479→  const bundle = getBatchContent(batchId);
   480→  if (!bundle) return [];
   481→  const subject = bundle.subjects.find((s) => s.id === subjectId);
   482→  if (!subject) return [];
   483→
   484→  let items: ChapterItem[];
   485→  if (tagType === 'STUDY_MATERIAL') {
   486→    items = subject.notes.map((n) => ({
   487→      _id: n.nodeId,
   488→      name: n.topic,
   489→      typeId: subject.id,
   490→      notes: 1,
   491→      videos: 0,
   492→      order: 0,
   493→    }));
   494→  } else {
   495→    items = subject.chapters.map((c, i) => ({
   496→      _id: c.id,
   497→      name: c.name,
   498→      typeId: subject.id,
   499→      notes: c.notesCount + c.dppCount,
   500→      videos: c.lectureCount,
   501→      order: i,
   502→    }));
   503→  }
   504→  const start = (Math.max(1, page) - 1) * limit;
   505→  return items.slice(start, start + Math.max(1, limit));
   506→}

   508→export function getResourceTopics(batchId: string, subjectIds: string): ChapterItem[] {
   509→  const bundle = getBatchContent(batchId);
   510→  if (!bundle) return [];
   511→  const wanted = (subjectIds || '').split(',').map((s) => s.trim()).filter(Boolean);
   512→  const subjects = wanted.length > 0 ? bundle.subjects.filter((s) => wanted.includes(s.id)) : bundle.subjects;
   513→  return subjects.map((s) => ({
   514→    _id: stableId('res', bundle.batch.batch_id, s.name),
   515→    name: `${s.name} — All Resources`,
   516→    typeId: s.id,
   517→    notes: s.notes.length + s.dpps.length,
   518→    videos: s.lectures.length,
   519→    order: 0,
   520→  }));
   521→}

   536→function lectureNode(l: LectureRef): ContentNode {
   537→  return {
   538→    _id: l.nodeId,
   539→    type: 'LECTURE',

   541→      _id: l.videoId,

   623→export function getContentNodes(
   624→  batchId: string,
   625→  subjectId: string,
   626→  tagId: string,
   627→  contentType = 'ALL',
   628→  skip = 0,
   629→  limit = 50,
   630→): ContentNode[] {
   631→  const bundle = getBatchContent(batchId);
   632→  if (!bundle) return [];
   633→
   634→  let nodes: ContentNode[] = [];
   635→
   636→  if (tagId.startsWith('res_')) {
   637→    // Resource folder — everything for the subject.
   638→    const subject = bundle.subjects.find((s) => s.id === subjectId) ?? bundle.subjects[0];
   639→    if (!subject) return [];
   640→    nodes = [
   641→      ...subject.notes.map(notesNode),
   642→      ...subject.dpps.map(dppNode),
   643→      ...subject.lectures.map(lectureNode),
   644→    ];
   645→  } else {
   646→    // Chapter / study-material topic.
   647→    for (const subject of bundle.subjects) {
   648→      const chapter = subject.chapters.find((c) => c.id === tagId);
   649→      if (chapter) {
   650→        const lectures = subject.lectures.filter((l) => l.chapterId === chapter.id);
   651→        const notes = subject.notes.filter((n) => n.chapterId === chapter.id);
   652→        const dpps = subject.dpps.filter((d) => d.chapterId === chapter.id);
   653→        const type = (contentType || 'ALL').toUpperCase();
   654→        if (type === 'LECTURE') nodes = lectures.map(lectureNode);
   655→        else if (type === 'NOTES') nodes = notes.map(notesNode);
   656→        else if (type === 'DPP_PDF') nodes = dpps.map(dppNode);
   657→        else if (type === 'DPP_VIDEOS') nodes = dpps.map(dppVideoNode);
   658→        else nodes = [...lectures.map(lectureNode), ...notes.map(notesNode), ...dpps.map(dppNode)];
   659→        break;
   660→      }
   661→      const sm = subject.notes.find((n) => n.nodeId === tagId);
   662→      if (sm) {
   663→        nodes = [notesNode(sm)];
   664→        break;
   665→      }
   666→    }
   667→  }
   668→
   669→  return nodes.slice(Math.max(0, skip), Math.max(0, skip) + Math.max(1, limit));
   670→}

   743→export function getTodaySchedule(batchId: string): ContentNode[] {
   744→  const bundle = getBatchContent(batchId);
   745→  if (!bundle || bundle.subjects.length === 0) return [];
   746→
   747→  const rng = makeRng('today', bundle.batch.batch_id, new Date().toISOString().slice(0, 10));
   748→  const now = Date.now();
   749→
   750→  const allLectures = bundle.subjects.flatMap((s) => s.lectures);
   751→  if (allLectures.length === 0) return [];
   752→
   753→  const nodes: ContentNode[] = [];
   754→
   755→  const liveLecture = rng.pick(allLectures);
   756→  nodes.push(
   757→    eventNode(
   758→      liveLecture,
   759→      new Date(now - 35 * 60000).toISOString(),
   760→      new Date(now + 25 * 60000).toISOString(),
   761→      'LIVE',
   762→    ),
   763→  );
   764→
   765→  const upcomingCount = rng.int(1, 2);
   766→  for (let i = 0; i < upcomingCount; i++) {
   767→    const lec = rng.pick(allLectures);
   768→    const start = new Date(now + (90 + i * 150) * 60000);
   769→    nodes.push(eventNode(lec, start.toISOString(), new Date(start.getTime() + 3600000).toISOString(), 'UPCOMING'));
   770→  }
   771→
   772→  const tomorrow = new Date(now + 86400000);
   773→  tomorrow.setHours(9, 30, 0, 0);
   774→  nodes.push(
   775→    eventNode(
   776→      rng.pick(allLectures),
   777→      tomorrow.toISOString(),
   778→      new Date(tomorrow.getTime() + 3600000).toISOString(),
   779→      'UPCOMING',
   780→    ),
   781→  );
   782→
   783→  return nodes;
   784→}

   816→export function getScheduleDetail(batchId: string, scheduleId: string): Record<string, unknown> | null {
   817→  const bundle = getBatchContent(batchId);
   818→  if (!bundle) return null;
   819→
   820→  // Today-schedule events resolve within the same day.
   821→  const events = getTodaySchedule(batchId);
   822→  const evt = events.find((e) => e._id === scheduleId);
   823→  if (evt) return evt.data;
   824→
   825→  for (const subject of bundle.subjects) {
   826→    const lecture = subject.lectures.find((l) => l.nodeId === scheduleId);
   827→    if (lecture) return lectureNode(lecture).data;
   828→
   829→    const note = subject.notes.find((n) => n.nodeId === scheduleId);
   830→    if (note) return notesNode(note).data;
   831→
   832→    const dpp = subject.dpps.find((d) => d.testId === scheduleId);
   833→    if (dpp) return dppNode(dpp).data;
   834→  }
   835→  return null;
   836→}

   849→export function getNotifications(batchId: string, page = 1, limit = 20): { data: NotificationItem[]; hasMore: boolean } {
   850→  const bundle = getBatchContent(batchId);
   851→  if (!bundle) return { data: [], hasMore: false };
   852→
   853→  const rng = makeRng('notifs', bundle.batch.batch_id);
   854→  const allLectures = bundle.subjects.flatMap((s) => s.lectures);
   855→  const allDpps = bundle.subjects.flatMap((s) => s.dpps);
   856→  const allTests = bundle.bigTests;
   857→  const now = Date.now();
   858→
   859→  const total = rng.int(22, 30);
   860→  const items: NotificationItem[] = [];
   861→  for (let i = 0; i < total; i++) {
   862→    const tpl = rng.pick(NOTIF_TEMPLATES);
   863→    let topic = '';
   864→    if (tpl.type === 'LECTURE' && allLectures.length > 0) topic = rng.pick(allLectures).chapterName;
   865→    else if (tpl.type === 'DPP' && allDpps.length > 0) topic = rng.pick(allDpps).name;
   866→    else if (tpl.type === 'TEST' && allTests.length > 0) topic = rng.pick(allTests).name;
   867→    else topic = bundle.batch.name;
   868→
   869→    const ageHours = i * rng.int(9, 26) + rng.int(1, 8);
   870→    items.push({
   871→      _id: stableId('ntf', bundle.batch.batch_id, i),
   872→      type: tpl.type,
   873→      heading: tpl.heading(topic),
   874→      announcement: tpl.announcement(topic),
   875→      scheduleTime: new Date(now - ageHours * 3600000).toISOString(),
   876→      isRead: i > 5,
   877→    });
   878→  }
   879→
   880→  const start = (Math.max(1, page) - 1) * limit;
   881→  const slice = items.slice(start, start + limit);
   882→  return { data: slice, hasMore: start + limit < items.length };
   883→}

   924→export interface QuizData {
   925→  _id: string;
   926→  name: string;
   927→  duration: number; // minutes

   930→
   931→export function getQuiz(batchId: string, testId: string): QuizData | null {
   932→  const bundle = getBatchContent(batchId);
   933→  if (!bundle) return null;
   934→
   935→  const dpp = bundle.subjects.flatMap((s) => s.dpps).find((d) => d.testId === testId);
   936→
   937→  let name = 'Practice Quiz';
   938→  let subjectName = bundle.subjects[0]?.name ?? 'General';
   939→  let questionCount = 8;
   940→
   941→  if (dpp) {
   942→    name = dpp.name;
   943→    subjectName = dpp.subjectName;
   944→    questionCount = dpp.totalQuestions;
   945→  } else {
   946→    const big = bundle.bigTests.find((t) => t.testId === testId);
   947→    if (big) {
   948→      name = big.name;
   949→      subjectName = big.subjectName;
   950→      questionCount = big.totalQuestions;
   951→    }
   952→  }
   953→
   954→  // Build the question pool from the subject bank + general banks.
   955→  const pool = [
   956→    ...quizBankForSubject(subjectName),
   957→    ...quizBankForSubject('Quantitative Aptitude'),
   958→    ...quizBankForSubject('General Awareness'),
   959→    ...quizBankForSubject('English Language'),
   960→  ];
   961→  const rng = makeRng('quiz', batchId, testId);
   962→  const shuffled = rng.shuffle(pool).slice(0, Math.max(1, Math.min(questionCount, pool.length)));
   963→  const questions = shuffled.map((qq, i) => {
   964→    const optionRng = makeRng('opts', batchId, testId, i);
   965→    const indices = optionRng.shuffle(qq.options.map((_, idx) => idx));
   966→    const options = indices.map((idx) => qq.options[idx]);
   967→    return {
   968→      _id: stableId('q', testId, i),
   969→      question: qq.q,
   970→      options,
   971→      answer: indices.indexOf(qq.answer),
   972→      marks: 4,
   973→    };
   974→  });
   975→
   976→  const duration = Math.max(10, Math.round((questions.length * 1.5) / 5) * 5);
   977→  return { _id: testId, name, duration, questions };
   978→}
