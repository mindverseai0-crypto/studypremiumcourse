    45→const asParams = (v: unknown): Params =>
    46→  v && typeof v === "object" && !Array.isArray(v) ? (v as Params) : {};

    48→const str = (v: unknown, fallback = ""): string =>
    49→  v === undefined || v === null ? fallback : String(v);

    51→const num = (v: unknown, fallback: number): number => {
    52→  const n = Number(v);

    56→/** Payload may arrive as object or JSON string. */
    57→function parsePayload(payload: unknown): unknown {
    58→  if (typeof payload === "string") {
    59→    try {
    60→      return JSON.parse(payload);
    61→    } catch {
    62→      return payload;
    63→    }
    64→  }
    65→  return payload;
    66→}

    76→/** PDF attachments are baked as /api/lxpdf links that resolve the real file. */
    77→function pdfAttachmentRef(batchId: string, subjectId: string, pdfId: string, attachmentId: string) {
    78→  const q = `batchId=${encodeURIComponent(batchId)}&subjectId=${encodeURIComponent(
    79→    subjectId,
    80→  )}&pdfId=${encodeURIComponent(pdfId)}&attachmentId=${encodeURIComponent(attachmentId)}`;
    81→  return { baseUrl: `/api/lxpdf?${q}`, key: "", _id: attachmentId };
    82→}

    84→function subjectRef(s: unknown): { _id: string; slug: string; name: string } {
    85→  if (s && typeof s === "object") {
    86→    const o = s as { _id?: string; slug?: string; name?: string };
    87→    return { _id: o._id || "", slug: o.slug || o._id || "", name: o.name || "" };
    88→  }
    89→  return { _id: String(s || ""), slug: String(s || ""), name: "" };
    90→}

    92→function eventStatus(item: LxScheduleItem, startIso: string, endIso: string): string {
    93→  const tag = String(item.tag || "").toLowerCase();
    94→  if (tag === "live" || tag === "live now") return "LIVE";
    95→  if (tag === "upcoming") return "UPCOMING";
    96→  if (tag === "ended" || tag === "cancelled") return tag === "cancelled" ? "CANCELLED" : "ENDED";
    97→  const now = Date.now();
    98→  const s = new Date(startIso || "").getTime();
    99→  const e = new Date(endIso || "").getTime();
   100→  if (Number.isFinite(s) && Number.isFinite(e)) {
   101→    if (now >= s && now <= e) return "LIVE";
   102→    if (now < s) return "UPCOMING";
   103→  }
   104→  return "ENDED";
   105→}

   135→function contentNodesFromItems(
   136→  batchId: string,
   137→  subjectId: string,
   138→  items: LxScheduleItem[],
   139→  kind: "LECTURE" | "DPP_VIDEOS" | "NOTES",
   140→): Array<{ _id: string; type: string; data: Record<string, unknown> }> {
   141→  const nodes: Array<{ _id: string; type: string; data: Record<string, unknown> }> = [];
   142→  for (const item of items) {
   143→    const pdfId = item._id || "";
   144→
   145→    if (kind === "LECTURE" || kind === "DPP_VIDEOS") {
   146→      const vd = item.videoDetails || {};
   147→      nodes.push({
   148→        _id: item._id || "",
   149→        type: kind,
   150→        data: {
   151→          _id: item._id || "",
   152→          topic: item.topic || vd.name || item.name || "Lecture",
   153→          date: item.date || item.startTime || "",
   154→          urlType: item.urlType || "",
   155→          url: item.url || "",
   156→          videoDetails: {
   157→            _id: item._id || "",
   158→            image: vd.image || item.previewImageUrl || "",
   159→            videoUrl: "",
   160→            duration: vd.duration || "",
   161→          },
   162→          homeworkIds: asArray(item.homeworkIds).flatMap((hw) => {
   163→            const h = hw as { _id?: string; topic?: string; attachmentIds?: unknown[] };
   164→            return asArray(h.attachmentIds).map((att) => {
   165→              const a = att as { _id?: string };
   166→              return {
   167→                _id: h._id || "",
   168→                topic: h.topic || "Class Note",
   169→                attachmentIds: [pdfAttachmentRef(batchId, subjectId, pdfId, a._id || "")],
   170→              };
   171→            });
   172→          }),
   173→          exerciseIds: [],
   174→        },
   175→      });
   176→      continue;
   177→    }
   178→
   179→    // NOTES / DppNotes → PDF cards
   180→    const atts: Array<Record<string, unknown>> = [];
   181→    let title = item.topic || item.name || "";
   182→    const homeworkSources = [
   183→      ...asArray(item.homeworkIds),
   184→      ...asArray(item.dpp?.homeworkIds),
   185→    ];
   186→    for (const hw of homeworkSources) {
   187→      const h = hw as { _id?: string; topic?: string; attachmentIds?: unknown[] };
   188→      if (!title && h.topic) title = h.topic;
   189→      for (const att of asArray(h.attachmentIds)) {
   190→        const a = att as { _id?: string; key?: string; baseUrl?: string; name?: string };
   191→        if (!a?._id) continue;
   192→        if (!title) title = a.name || "";
   193→        // some attachments already carry a real static.pw.live key
   194→        const direct = a.baseUrl && a.key ? `${a.baseUrl}${a.key}` : "";
   195→        const ref = direct
   196→          ? { baseUrl: direct, key: "", _id: a._id }
   197→          : pdfAttachmentRef(batchId, subjectId, pdfId, a._id);
   198→        atts.push({
   199→          _id: a._id,
   200→          topic: h.topic || a.name || item.topic || item.name || "Study Material",
   201→          attachmentIds: [ref],
   202→        });
   203→      }
   204→    }
   205→    if (atts.length === 0) continue;
   206→    nodes.push({
   207→      _id: item._id || "",
   208→      type: "NOTES",
   209→      data: {
   210→        _id: item._id || "",
   211→        topic: title || "Study Material",
   212→        date: item.date || item.startTime || "",
   213→        homeworkIds: atts,
   214→      },
   215→    });
   216→  }
   217→  return nodes;
   218→}

   220→function resolveContentKind(contentType: string): "LECTURE" | "DPP_VIDEOS" | "NOTES" | null {
   221→  switch ((contentType || "ALL").toUpperCase()) {
   222→    case "ALL":
   223→    case "LECTURE":
   224→      return "LECTURE";
   225→    case "DPP_VIDEOS":
   226→      return "DPP_VIDEOS";
   227→    case "NOTES":
   228→    case "DPP_PDF":
   229→      return "NOTES";
   230→    default:
   231→      return null;
   232→  }
   233→}

   240→      return "DppNotes";

   248→/** base64url-encode a signed query for the stateless /s/<sig>/ relay path. */
   249→function sigPathSegment(signedQuery: string): string {
   250→  const q = signedQuery.startsWith("?") ? signedQuery.slice(1) : signedQuery;
   251→  if (!q) return "";
   252→  return Buffer.from(q, "utf8").toString("base64url");
   253→}

   255→/** Resolve a video → same-origin playable payload (dash/hls/youtube/mp4). */
   256→async function resolveVideoPayload(childId: string, batchId: string, subjectId: string) {
   257→  const vid = await lxVideoUrl(batchId, subjectId, childId);
   258→  if (vid?.url) {
   259→    const url = vid.url.trim();
   260→    if (/\.mpd(\?|$)/i.test(url)) {
   261→      const folder = folderFromUrl(url);
   262→      const sig = sigPathSegment(vid.signedUrl || "");
   263→      if (folder && sig) registerSignedFolder(folder, vid.signedUrl || "");
   264→      let clearKeys: Record<string, string> | null = null;
   265→      try {
   266→        const mpdUrl = `${url}${vid.signedUrl || ""}`;
   267→        const controller = new AbortController();
   268→        const timer = setTimeout(() => controller.abort(), 30000);
   269→        const mpdRes = await fetch(mpdUrl, {
   270→          cache: "no-store",
   271→          signal: controller.signal,
   272→          headers: { "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64)" },
   273→        });
   274→        clearTimeout(timer);
   275→        if (mpdRes.ok) {
   276→          const text = await mpdRes.text();
   277→          const kidMatch = text.match(/default_KID="([0-9a-fA-F-]+)"/);
   278→          if (kidMatch) {
   279→            clearKeys = await lxOtp(kidMatch[1]);
   280→          }
   281→        }
   282→      } catch {
   283→        /* keys unavailable — player will surface a decrypt error */
   284→      }
   285→      // Stateless relay URL: relative DASH segments inherit the signature.
   286→      const manifestUrl = folder
   287→        ? `/api/pw/${folder}${sig ? `/s/${sig}` : ""}/master.mpd`
   288→        : "";
   289→      if (!manifestUrl) return null;
   290→      return {
   291→        kind: "dash",
   292→        url: manifestUrl,
   293→        signedUrl: "",
   294→        clearKeys,
   295→        poster: "",
   296→        drmDetails: null,
   297→      };
   298→    }
   299→    if (/\.m3u8(\?|$)/i.test(url)) {
   300→      const folder = folderFromUrl(url);
   301→      const sig = sigPathSegment(vid.signedUrl || "");
   302→      if (!folder) return null;
   303→      if (sig) registerSignedFolder(folder, vid.signedUrl || "");
   304→      // Root-level LIVE manifest (…/index.m3u8?Signature=… — no uuid folder):
   305→      // each live class gets its own channel like this, on its own live host.
   306→      // The host is embedded in the relay path (validated against an
   307→      // allowlist in pw-cdn); the manifest itself is the relayed asset and
   308→      // segments resolve as its siblings through the same /s/<sig>/ path.
   309→      const isUuidFolder = /^[a-f0-9-]{16,}$/i.test(folder);
   310→      if (!isUuidFolder && /^[A-Za-z0-9._-]+\.m3u8$/i.test(folder)) {
   311→        const liveHost = hostFromUrl(url);
   312→        return {
   313→          kind: "hls",
   314→          url: `/api/pw/h/${liveHost}/${folder}${sig ? `/s/${sig}` : ""}`,
   315→          signedUrl: "",
   316→          clearKeys: null,
   317→          poster: "",
   318→          drmDetails: null,
   319→        };
   320→      }
   321→      return {
   322→        kind: "hls",
   323→        url: `/api/pw/${folder}${sig ? `/s/${sig}` : ""}/master.m3u8`,
   324→        signedUrl: "",
   325→        clearKeys: null,
   326→        poster: "",
   327→        drmDetails: null,
   328→      };
   329→    }
   330→    if (/youtube\.com|youtu\.be/i.test(url)) {
   331→      return { kind: "youtube", url, signedUrl: "", clearKeys: null, poster: "", drmDetails: null };
   332→    }
   333→    // plain file (mp4 etc.) inside the signed folder
   334→    const folder = folderFromUrl(url);
   335→    const sig = sigPathSegment(vid.signedUrl || "");
   336→    if (folder) {
   337→      if (sig) registerSignedFolder(folder, vid.signedUrl || "");
   338→      const fileName = url.split("/").pop() || "video.mp4";
   339→      return {
   340→        kind: "mp4",
   341→        url: `/api/pw/${folder}${sig ? `/s/${sig}` : ""}/${fileName}`,
   342→        signedUrl: "",
   343→        clearKeys: null,
   344→        poster: "",
   345→        drmDetails: null,
   346→      };
   347→    }
   348→    return { kind: "mp4", url, signedUrl: "", clearKeys: null, poster: "", drmDetails: null };
   349→  }
   350→
   351→  // No signed stream — maybe a youtube lecture.
   352→  if (subjectId) {
   353→    const detail = await lxScheduleDetail(batchId, subjectId, childId);
   354→    const url = detail?.url || "";
   355→    if (url && /youtube\.com|youtu\.be/i.test(url)) {
   356→      return { kind: "youtube", url, signedUrl: "", clearKeys: null, poster: "", drmDetails: null };
   357→    }
   358→  }
   359→  return null;
   360→}

   362→// ---------------------------------------------------------------------------

   365→export async function POST(request: NextRequest) {
   366→  try {
   367→    const body = (await request.json().catch(() => ({}))) as ActionBody;
   368→    const action = typeof body.action === "string" ? body.action : "";
   369→    const params = asParams(body.params);
   370→    const payload = parsePayload(body.payload);
   371→
   372→    switch (action) {
   373→      case "pw_btch_dtl": {
   374→        const batchId = str(params.batchId).trim();
   375→        if (!batchId) return NextResponse.json({ success: false }, { headers: CORS_HEADERS });
   376→        const d = await lxBatchDetails(batchId);
   377→        if (!d) return NextResponse.json({ success: false }, { headers: CORS_HEADERS });
   378→
   379→        const exams = Array.isArray(d.exam)
   380→          ? (d.exam as unknown[])
   381→              .map((e) => (typeof e === "string" ? e : (e as { name?: string })?.name || ""))
   382→              .filter(Boolean)
   383→          : [];
   384→        const subjects = Array.isArray(d.subjects) ? d.subjects : [];
   385→        return NextResponse.json(
   386→          {
   387→            success: true,
   388→            data: {
   389→              _id: d._id || batchId,
   390→              id: d.slug || batchId,
   391→              name: d.name || "Batch",
   392→              byName: d.byName || "",
   393→              description: typeof d.description === "string" ? d.description : "",
   394→              imageUrl: d.previewImage || d.previewImageUrl || "",
   395→              exams,
   396→              language: d.language || "",
   397→              className:
   398→                typeof d.class === "string"
   399→                  ? d.class
   400→                  : Array.isArray(d.class)
   401→                    ? (d.class as unknown[]).map((c) => String(c)).join(", ")
   402→                    : "",
   403→              amount: typeof d.feeTotal === "number" ? d.feeTotal : null,
   404→              startDate: d.startDate || "",
   405→              endDate: d.endDate || "",
   406→              // subject mapping — subjectId = SLUG (SubjectInfo/TopicInfo are
   407→              // slug-addressed), batchSubjectId & masterId for dpp/schedule apis
   408→              subjects: subjects.map((s) => ({
   409→                _id: s.slug || s.subjectId || s._id || "",
   410→                subject: s.subject || "",
   411→                subjectId: s.slug || s.subjectId || "",
   412→                batchSubjectId: s._id || "",
   413→                masterId: s.subjectId || "",
   414→              })),
   415→            },
   416→          },
   417→          { headers: CORS_HEADERS },
   418→        );
   419→      }
   420→
   421→      case "pw_sub_topics": {
   422→        const batchId = str(params.batchId).trim();
   423→        const subjectId = str(params.subjectId).trim();
   424→        const page = num(params.page, 1);
   425→        if (!batchId || !subjectId) {
   426→          return NextResponse.json({ success: true, data: { data: [] } }, { headers: CORS_HEADERS });
   427→        }
   428→        try {
   429→          const chapters = await lxChapters(batchId, subjectId, page);
   430→          const data = chapters.map((c, i) => ({
   431→            _id: c._id || "",
   432→            name: c.name || "Chapter",
   433→            typeId: subjectId,
   434→            notes: Number(c.notes || 0),
   435→            videos: Number(c.videos || c.lectureVideos || 0),
   436→            exercises: Number(c.exercises || 0),
   437→            order: Number.isFinite(c.displayOrder) ? Number(c.displayOrder) : i,
   438→          }));
   439→          return NextResponse.json({ success: true, data: { data } }, { headers: CORS_HEADERS });
   440→        } catch {
   441→          return NextResponse.json({ success: true, data: { data: [] } }, { headers: CORS_HEADERS });
   442→        }
   443→      }
   444→
   445→      case "pw_res_topics":
   446→        // upstream has no separate resource folders
   447→        return NextResponse.json({ success: true, data: { data: [] } }, { headers: CORS_HEADERS });
   448→
   449→      case "pw_sch_cntnt": {
   450→        const batchId = str(params.batchId).trim();
   451→        const subjectId = str(params.subjectId).trim();
   452→        const tagId = str(params.tagId).trim();
   453→        const contentType = str(params.contentType, "ALL");
   454→        const page = num(params.page, 1);
   455→        if (!batchId || !subjectId || !tagId) {
   456→          return NextResponse.json({ success: true, data: [] }, { headers: CORS_HEADERS });
   457→        }
   458→        const kind = resolveContentKind(contentType);
   459→        if (!kind) return NextResponse.json({ success: true, data: [] }, { headers: CORS_HEADERS });
   460→        try {
   461→          const items = await lxTopicContent(batchId, subjectId, tagId, upstreamContentType(contentType), page);
   462→          const nodes = contentNodesFromItems(batchId, subjectId, items, kind);
   463→          return NextResponse.json({ success: true, data: nodes }, { headers: CORS_HEADERS });
   464→        } catch {
   465→          return NextResponse.json({ success: true, data: [] }, { headers: CORS_HEADERS });
   466→        }
   467→      }
   468→
   469→      case "pw_dpp_lst": {
   470→        const batchId = str(params.batchId).trim();
   471→        const batchSubjectId = str(params.batchSubjectId).trim();
   472→        const chapterId = str(params.chapterId).trim();
   473→        const page = num(params.page, 1);
   474→        const limit = num(params.limit, 20);
   475→        if (!batchId || !batchSubjectId || !chapterId) {
   476→          return NextResponse.json({ success: true, data: { data: [] } }, { headers: CORS_HEADERS });
   477→        }
   478→        try {
   479→          const items = await lxDppList(batchId, batchSubjectId, chapterId, page, limit);
   480→          const data = items
   481→            .map((it) => {
   482→              const q = it.dppQuizDetails;
   483→              const t = q?.test;
   484→              if (!t?._id) return null;
   485→              return {
   486→                _id: t._id,
   487→                name: t.name || "DPP",
   488→                totalQuestions: Number(t.totalQuestions || 0),
   489→                totalMarks: Number(t.totalMarks || 0),
   490→                duration: Number(t.maxDuration || 0) * 60 || 1800,
   491→                testActivityStatus: "Not Started",
   492→                testStudentMapping: null,
   493→                dppQuizDetails: {
   494→                  contentId: q?.contentId || t._id,
   495→                  scheduleId: q?.scheduleId || "",
   496→                  test: {
   497→                    _id: t._id,
   498→                    name: t.name || "DPP",
   499→                    totalQuestions: Number(t.totalQuestions || 0),
   500→                    totalMarks: Number(t.totalMarks || 0),
   501→                  },
   502→                  testStudentMapping: { testActivityStatus: q?.tag || "Start" },
   503→                },
   504→              };
   505→            })
   506→            .filter(Boolean);
   507→          return NextResponse.json({ success: true, data: { data } }, { headers: CORS_HEADERS });
   508→        } catch {
   509→          return NextResponse.json({ success: true, data: { data: [] } }, { headers: CORS_HEADERS });
   510→        }
   511→      }
   512→
   513→      case "pw_dpp_start": {
   514→        // Auth-gated upstream: anonymous calls fail → surfaced honestly.
   515→        const testId = str(params.testId).trim();
   516→        const batchId = str(params.batchId).trim();
   517→        if (!testId || !batchId) {
   518→          return NextResponse.json(
   519→            { success: false, error: "testId and batchId are required" },
   520→            { headers: CORS_HEADERS },
   521→          );
   522→        }
   523→        try {
   524→          const data = await lxStartDpp({
   525→            testId,
   526→            batchId,
   527→            type: str(params.type, "Start"),
   528→            scheduleId: str(params.scheduleId),
   529→            batchScheduleId: str(params.batchScheduleId) || str(params.scheduleId),
   530→            subjectId: str(params.subjectId),
   531→          });
   532→          return NextResponse.json({ success: true, data }, { headers: CORS_HEADERS });
   533→        } catch (e) {
   534→          return NextResponse.json(
   535→            { success: false, error: e instanceof Error ? e.message : "Failed to start DPP" },
   536→            { headers: CORS_HEADERS },
   537→          );
   538→        }
   539→      }
   540→
   541→      case "pw_test_lst": {

   702→
   703→/** Flatten test ids from scalars / arrays / objects ({testId}|{_id}|{id}). */
   704→function collectTestIds(source: unknown, out: string[], seen: Set<string>): void {
   705→  if (source === undefined || source === null) return;
   706→  if (Array.isArray(source)) {
   707→    source.forEach((item) => collectTestIds(item, out, seen));
   708→    return;
   709→  }
   710→  if (typeof source === "object") {
   711→    const o = source as Record<string, unknown>;
   712→    collectTestIds(o.testId ?? o._id ?? o.id, out, seen);
   713→    return;
   714→  }
   715→  const s = String(source).trim();
   716→  if (s && !seen.has(s)) {
   717→    seen.add(s);
   718→    out.push(s);
   719→  }
   720→}
