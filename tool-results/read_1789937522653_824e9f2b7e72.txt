    43→function formatTime(t: number): string {
    44→  if (!isFinite(t) || t < 0) return '00:00';
    45→  const s = Math.floor(t % 60);
    46→  const m = Math.floor((t / 60) % 60);
    47→  const h = Math.floor(t / 3600);
    48→  const mm = String(m).padStart(2, '0');
    49→  const ss = String(s).padStart(2, '0');
    50→  return h > 0 ? `${h}:${mm}:${ss}` : `${mm}:${ss}`;
    51→}

    53→function timeAgo(iso: string): string {
    54→  const then = new Date(iso).getTime();
    55→  if (!isFinite(then)) return '';
    56→  const secs = Math.max(0, Math.floor((Date.now() - then) / 1000));
    57→  if (secs < 60) return 'just now';
    58→  const mins = Math.floor(secs / 60);
    59→  if (mins < 60) return `${mins}m ago`;
    60→  const hrs = Math.floor(mins / 60);
    61→  if (hrs < 24) return `${hrs}h ago`;
    62→  const days = Math.floor(hrs / 24);
    63→  if (days < 30) return `${days}d ago`;
    64→  return new Date(then).toLocaleDateString();
    65→}

    69→function AuthorAvatar({ name, small = false }: { name: string; small?: boolean }) {
    70→  const cls = `cm-avatar${small ? ' sm' : ''}${isBotName(name) ? ' bot' : ''}`;
    71→  if (isBotName(name)) {
    72→    return (
    73→      <span className={cls} aria-hidden="true">
    74→        <img src={avatarUrlFor(name)} alt="" width={34} height={34} loading="lazy" />
    75→      </span>
    76→    );
    77→  }
    78→  return <span className={cls}>{(name || 'S').charAt(0)}</span>;
    79→}
