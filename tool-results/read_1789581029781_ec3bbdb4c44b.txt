    47→interface ActionBody {
    48→  action?: unknown;

    54→const asParams = (v: unknown): Params =>
    55→  v && typeof v === "object" && !Array.isArray(v) ? (v as Params) : {};
    56→
    57→const str = (v: unknown, fallback = ""): string =>
    58→  v === undefined || v === null ? fallback : String(v);
    59→
    60→const num = (v: unknown, fallback: number): number => {
    61→  const n = Number(v);
    62→  return Number.isFinite(n) ? n : fallback;

    65→/** Payload may arrive as object or JSON string. */
    66→function parsePayload(payload: unknown): unknown {
    67→  if (typeof payload === "string") {
    68→    try {
    69→      return JSON.parse(payload);
    70→    } catch {
    71→      return payload;
    72→    }
    73→  }
    74→  return payload;
    75→}

    86→function pdfAttachmentRef(batchId: string, subjectId: string, pdfId: string, attachmentId: string) {
    87→  const q = `batchId=${encodeURIComponent(batchId)}&subjectId=${encodeURIComponent(
    88→    subjectId,
    89→  )}&pdfId=${encodeURIComponent(pdfId)}&attachmentId=${encodeURIComponent(attachmentId)}`;
    90→  return { baseUrl: `/api/lxpdf?${q}`, key: "", _id: attachmentId };

    94→  if (s && typeof s === "object") {

    96→    return { _id: o._id || "", slug: o.slug || o._id || "", name: o.name || "" };

   101→function eventStatus(item: LxScheduleItem, startIso: string, endIso: string): string {
   102→  const tag = String(item.tag || "").toLowerCase();
   103→  if (tag === "live" || tag === "live now") return "LIVE";
   104→  if (tag === "upcoming") return "UPCOMING";
   105→  if (tag === "ended" || tag === "cancelled") return tag === "cancelled" ? "CANCELLED" : "ENDED";
   106→  const now = Date.now();
   107→  const s = new Date(startIso || "").getTime();
   108→  const e = new Date(endIso || "").getTime();
   109→  if (Number.isFinite(s) && Number.isFinite(e)) {
   110→    if (now >= s && now <= e) return "LIVE";
   111→    if (now < s) return "UPCOMING";
   112→  }
   113→  return "ENDED";
   114→}

   144→function contentNodesFromItems(
   145→  batchId: string,
   146→  subjectId: string,
   147→  items: LxScheduleItem[],
   148→  kind: "LECTURE" | "DPP_VIDEOS" | "NOTES",
   149→): Array<{ _id: string; type: string; data: Record<string, unknown> }> {
   150→  const nodes: Array<{ _id: string; type: string; data: Record<string, unknown> }> = [];
   151→  for (const item of items) {
   152→    const pdfId = item._id || "";
   153→
   154→    if (kind === "LECTURE" || kind === "DPP_VIDEOS") {
   155→      const vd = item.videoDetails || {};
   156→      nodes.push({
   157→        _id: item._id || "",
   158→        type: kind,
   159→        data: {
   160→          _id: item._id || "",
   161→          topic: item.topic || vd.name || item.name || "Lecture",
   162→          date: item.date || item.startTime || "",
   163→          urlType: item.urlType || "",
   164→          url: item.url || "",
   165→          videoDetails: {
   166→            _id: item._id || "",
   167→            image: vd.image || item.previewImageUrl || "",
   168→            videoUrl: "",
   169→            duration: vd.duration || "",
   170→          },
   171→          homeworkIds: asArray(item.homeworkIds).flatMap((hw) => {
   172→            const h = hw as { _id?: string; topic?: string; attachmentIds?: unknown[] };
   173→            return asArray(h.attachmentIds).map((att) => {
   174→              const a = att as { _id?: string };
   175→              return {
   176→                _id: h._id || "",
   177→                topic: h.topic || "Class Note",
   178→                attachmentIds: [pdfAttachmentRef(batchId, subjectId, pdfId, a._id || "")],
   179→              };
   180→            });
   181→          }),
   182→          exerciseIds: [],
   183→        },
   184→      });
   185→      continue;
   186→    }
   187→
   188→    // NOTES / DppNotes → PDF cards
   189→    const atts: Array<Record<string, unknown>> = [];
   190→    let title = item.topic || item.name || "";
   191→    const homeworkSources = [
   192→      ...asArray(item.homeworkIds),
   193→      ...asArray(item.dpp?.homeworkIds),
   194→    ];
   195→    for (const hw of homeworkSources) {
   196→      const h = hw as { _id?: string; topic?: string; attachmentIds?: unknown[] };
   197→      if (!title && h.topic) title = h.topic;
   198→      for (const att of asArray(h.attachmentIds)) {
   199→        const a = att as { _id?: string; key?: string; baseUrl?: string; name?: string };
   200→        if (!a?._id) continue;
   201→        if (!title) title = a.name || "";
   202→        // some attachments already carry a real static.pw.live key
   203→        const direct = a.baseUrl && a.key ? `${a.baseUrl}${a.key}` : "";
   204→        const ref = direct
   205→          ? { baseUrl: direct, key: "", _id: a._id }
   206→          : pdfAttachmentRef(batchId, subjectId, pdfId, a._id);
   207→        atts.push({
   208→          _id: a._id,
   209→          topic: h.topic || a.name || item.topic || item.name || "Study Material",
   210→          attachmentIds: [ref],
   211→        });
   212→      }
   213→    }
   214→    if (atts.length === 0) continue;
   215→    nodes.push({
   216→      _id: item._id || "",
   217→      type: "NOTES",
   218→      data: {
   219→        _id: item._id || "",
   220→        topic: title || "Study Material",
   221→        date: item.date || item.startTime || "",
   222→        homeworkIds: atts,
   223→      },
   224→    });
   225→  }
   226→  return nodes;
   227→}

   257→/**
   258→ * Direct (unsigned) CloudFront urls per schedule item from the owner's
   259→ * V2 listing API — the same source the production site merges into its

   262→async function v2DirectUrls(
   263→  batchId: string,
   264→  subjectId: string,
   265→  tagId: string,
   266→): Promise<Map<string, { url: string; videoId: string }>> {
   267→  const out = new Map<string, { url: string; videoId: string }>();
   268→  try {
   269→    const controller = new AbortController();
   270→    const timer = setTimeout(() => controller.abort(), 9000);
   271→    const res = await fetch("https://pw.studyparcham.in/api/data", {
   272→      method: "POST",
   273→      headers: { "Content-Type": "application/json" },
   274→      cache: "no-store",
   275→      signal: controller.signal,
   276→      body: JSON.stringify({
   277→        action: "pw_v2_list",
   278→        params: { batchId, subjectId, tagId, contentType: "videos", page: 1 },
   279→        method: "GET",
   280→        payload: null,
   281→      }),
   282→    });
   283→    clearTimeout(timer);
   284→    if (!res.ok) return out;
   285→    const body = (JSON.parse(await res.text()) as {
   286→      data?: Array<{
   287→        _id?: string;
   288→        url?: string;
   289→        videoDetails?: { videoUrl?: string; _id?: string };
   290→      }>;
   291→    });
   292→    for (const it of body.data || []) {
   293→      const u = it.url || it.videoDetails?.videoUrl || "";
   294→      if (it._id && u) out.set(it._id, { url: u, videoId: it.videoDetails?._id || "" });
   295→    }
   296→  } catch {
   297→    /* best-effort merge */
   298→  }
   299→  return out;
   300→}

   302→/** base64url-encode a signed query for the stateless /s/<sig>/ relay path. */
   303→function sigPathSegment(signedQuery: string): string {
   304→  const q = signedQuery.startsWith("?") ? signedQuery.slice(1) : signedQuery;
   305→  if (!q) return "";
   306→  return Buffer.from(q, "utf8").toString("base64url");
   307→}

   309→// ---------------------------------------------------------------------------
   310→// Lecture stream signing chain (server-side only).
   311→//
   312→// 1. signedUrl from the upstream video-url API (per-video folder policy)
   313→// 2. LECTURE_SIGN_URL env template (owner's signing service, e.g.
   314→//    https://video-detail.studyparcham.in/?batch_id={batch_id}&video_id={video_id}&subject_id={subject_id})
   315→// 3. non-expired fallback constants registered in pw-cdn
   316→// ---------------------------------------------------------------------------
   317→
   318→async function envSignerSignedQuery(
   319→  batchId: string,
   320→  videoId: string,
   321→  subjectId: string,
   322→): Promise<string> {
   323→  const tpl = process.env.LECTURE_SIGN_URL || "";
   324→  if (!tpl) return "";
   325→  const url = tpl
   326→    .replace(/\{batch_id\}/g, encodeURIComponent(batchId))
   327→    .replace(/\{video_id\}/g, encodeURIComponent(videoId))
   328→    .replace(/\{subject_id\}/g, encodeURIComponent(subjectId));
   329→  try {
   330→    const controller = new AbortController();
   331→    const timer = setTimeout(() => controller.abort(), 12000);
   332→    const res = await fetch(url, {
   333→      cache: "no-store",
   334→      signal: controller.signal,
   335→      headers: { "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) Chrome/126.0" },
   336→    });
   337→    clearTimeout(timer);
   338→    if (!res.ok) return "";
   339→    const body = (await res.json()) as {
   340→      url?: string;
   341→      signedUrl?: string;
   342→      data?: { url?: string; signedUrl?: string };
   343→    };
   344→    const signed = body.signedUrl || body.data?.signedUrl || "";
   345→    if (signed) return signed.startsWith("?") ? signed : `?${signed}`;
   346→    // a signed absolute URL works too — peel off its query
   347→    const abs = body.url || body.data?.url || "";
   348→    if (abs.includes("Signature=")) {
   349→      const q = abs.slice(abs.indexOf("?"));
   350→      return q;
   351→    }
   352→    return "";
   353→  } catch {
   354→    return "";
   355→  }
   356→}

   358→/** Resolve the folder-wide signed query for a video through the whole chain. */
   359→async function resolveSignedQuery(
   360→  folder: string,
   361→  opts: { batchId: string; videoId: string; subjectId: string; signedUrl?: string },
   362→): Promise<string> {
   363→  if (opts.signedUrl) {
   364→    registerSignedFolder(folder, opts.signedUrl);
   365→    return opts.signedUrl.startsWith("?") ? opts.signedUrl : `?${opts.signedUrl}`;
   366→  }
   367→  const envSig = await envSignerSignedQuery(opts.batchId, opts.videoId, opts.subjectId);
   368→  if (envSig) {
   369→    registerSignedFolder(folder, envSig);
   370→    return envSig;
   371→  }
   372→  const fb = getFallbackSignedQuery(folder);
   373→  if (fb) {
   374→    registerSignedFolder(folder, fb);
   375→    return fb;
   376→  }
   377→  const cached = getSignedQuery(folder);
   378→  return cached || "";
   379→}

   381→/**
   382→ * Same-origin relay URL for a resolved CloudFront asset — keeps the REAL
   383→ * manifest path from the upstream url (e.g. hls/720/main.m3u8, master.mpd)
   384→ * instead of guessing, and carries the signature statelessly in the path.
   385→ */
   386→async function relayUrlFor(
   387→  rawUrl: string,
   388→  opts: { batchId: string; videoId: string; subjectId: string; signedUrl?: string },
   389→): Promise<{ url: string; signedQuery: string } | null> {
   390→  const host = hostFromUrl(rawUrl);
   391→  const folder = folderFromUrl(rawUrl);
   392→  const sub = subPathFromUrl(rawUrl);
   393→  if (!folder || !sub) return null;
   394→  const signedQuery = await resolveSignedQuery(folder, { ...opts, signedUrl: opts.signedUrl || "" });
   395→  if (host) registerFolderHost(folder, host);
   396→  if (!signedQuery) return null;
   397→  const sig = sigPathSegment(signedQuery);
   398→  const hostSeg = host && host !== "d1d34p8vz63oiq.cloudfront.net" ? `/h/${host}` : "";
   399→  return { url: `/api/pw${hostSeg}/${folder}${sig ? `/s/${sig}` : ""}/${sub}`, signedQuery };
   400→}

   402→/** Resolve a video → same-origin playable payload (dash/hls/youtube/mp4). */
   403→async function resolveVideoPayload(
   404→  childId: string,
   405→  batchId: string,
   406→  subjectId: string,
   407→  vUrlParam = "",
   408→  realVideoId = "",
   409→) {
   410→  let vid = await lxVideoUrl(batchId, subjectId, childId).catch(() => null);
   411→  if (!vid && realVideoId && realVideoId !== childId) {
   412→    vid = await lxVideoUrl(batchId, subjectId, realVideoId).catch(() => null);
   413→  }
   414→  const sigOpts = { batchId, videoId: realVideoId || childId, subjectId };
   415→  const rawUrl = vid?.url?.trim() || vUrlParam.trim();
   416→  if (rawUrl) {
   417→    const url = rawUrl;
   418→    if (/\.mpd(\?|$)/i.test(url)) {
   419→      const relay = await relayUrlFor(url, { ...sigOpts, signedUrl: vid?.signedUrl || "" });
   420→      if (!relay) return null;
   421→      let clearKeys: Record<string, string> | null = null;
   422→      try {
   423→        // Probe the manifest through the same upstream lane chain the relay
   424→        // uses (delta → pearl → direct) to extract the ClearKey KID.
   425→        const u = new URL(url);
   426→        const mpdRes = await fetchSignedAsset(
   427→          u.host,
   428→          u.pathname.replace(/^\/+/, ""),
   429→          relay.signedQuery,
   430→          { timeoutMs: 30000 },
   431→        );
   432→        if (mpdRes.ok) {
   433→          const text = await mpdRes.text();
   434→          const kidMatch = text.match(/default_KID="([0-9a-fA-F-]+)"/);
   435→          if (kidMatch) {
   436→            clearKeys = await lxOtp(kidMatch[1]);
   437→          }
   438→        }
   439→      } catch {
   440→        /* keys unavailable — player will surface a decrypt error */
   441→      }
   442→      return {
   443→        kind: "dash",
   444→        url: relay.url,
   445→        signedUrl: "",
   446→        clearKeys,
   447→        poster: "",
   448→        drmDetails: null,
   449→      };
   450→    }
   451→    if (/\.m3u8(\?|$)/i.test(url)) {
   452→      const relay = await relayUrlFor(url, { ...sigOpts, signedUrl: vid?.signedUrl || "" });
   453→      if (!relay) return null;
   454→      return {
   455→        kind: "hls",
   456→        url: relay.url,
   457→        signedUrl: "",
   458→        clearKeys: null,
   459→        poster: "",
   460→        drmDetails: null,
   461→      };
   462→    }
   463→    if (/youtube\.com|youtu\.be/i.test(url)) {
   464→      return { kind: "youtube", url, signedUrl: "", clearKeys: null, poster: "", drmDetails: null };
   465→    }
   466→    // plain file (mp4 etc.) inside the signed folder
   467→    const relay = await relayUrlFor(url, { ...sigOpts, signedUrl: vid?.signedUrl || "" });
   468→    if (!relay) return null;
   469→    return {
   470→      kind: "mp4",
   471→      url: relay.url,
   472→      signedUrl: "",
   473→      clearKeys: null,
   474→      poster: "",
   475→      drmDetails: null,
   476→    };
   477→  }
   478→
   479→  // No signed stream — maybe a youtube lecture.
   480→  if (subjectId) {
   481→    const detail = await lxScheduleDetail(batchId, subjectId, childId).catch(() => null);
   482→    const url = (detail as { url?: string } | null)?.url || "";
   483→    if (url && /youtube\.com|youtu\.be/i.test(url)) {
   484→      return { kind: "youtube", url, signedUrl: "", clearKeys: null, poster: "", drmDetails: null };
   485→    }
   486→  }
   487→  return null;
   488→}

   490→// ---------------------------------------------------------------------------
   491→// POST /api/data — action API (real upstream data, transformed)
   492→// ---------------------------------------------------------------------------
   493→export async function POST(request: NextRequest) {
   494→  try {
   495→    const body = (await request.json().catch(() => ({}))) as ActionBody;
   496→    const action = typeof body.action === "string" ? body.action : "";
   497→    const params = asParams(body.params);
   498→    const payload = parsePayload(body.payload);
   499→
   500→    switch (action) {
   501→      case "pw_btch_dtl": {
   502→        const batchId = str(params.batchId).trim();
   503→        if (!batchId) return NextResponse.json({ success: false }, { headers: CORS_HEADERS });
   504→        const d = await lxBatchDetails(batchId);
   505→        if (!d) return NextResponse.json({ success: false }, { headers: CORS_HEADERS });
   506→
   507→        const exams = Array.isArray(d.exam)
   508→          ? (d.exam as unknown[])
   509→              .map((e) => (typeof e === "string" ? e : (e as { name?: string })?.name || ""))
   510→              .filter(Boolean)
   511→          : [];
   512→        const subjects = Array.isArray(d.subjects) ? d.subjects : [];
   513→        return NextResponse.json(
   514→          {
   515→            success: true,
   516→            data: {
   517→              _id: d._id || batchId,
   518→              id: d.slug || batchId,
   519→              name: d.name || "Batch",
   520→              byName: d.byName || "",
   521→              description: typeof d.description === "string" ? d.description : "",
   522→              imageUrl: d.previewImage || d.previewImageUrl || "",
   523→              exams,
   524→              language: d.language || "",
   525→              className:
   526→                typeof d.class === "string"
   527→                  ? d.class
   528→                  : Array.isArray(d.class)
   529→                    ? (d.class as unknown[]).map((c) => String(c)).join(", ")
   530→                    : "",
   531→              amount: typeof d.feeTotal === "number" ? d.feeTotal : null,
   532→              startDate: d.startDate || "",
   533→              endDate: d.endDate || "",
   534→              // subject mapping — subjectId = SLUG (SubjectInfo/TopicInfo are
   535→              // slug-addressed), batchSubjectId & masterId for dpp/schedule apis
   536→              subjects: subjects.map((s) => ({
   537→                _id: s.slug || s.subjectId || s._id || "",
   538→                subject: s.subject || "",
   539→                subjectId: s.slug || s.subjectId || "",
   540→                batchSubjectId: s._id || "",
   541→                masterId: s.subjectId || "",
   542→              })),
   543→            },
   544→          },
   545→          { headers: CORS_HEADERS },
   546→        );
   547→      }
   548→
   549→      case "pw_sub_topics": {
   550→        const batchId = str(params.batchId).trim();
   551→        const subjectId = str(params.subjectId).trim();
   552→        const page = num(params.page, 1);
   553→        if (!batchId || !subjectId) {
   554→          return NextResponse.json({ success: true, data: { data: [] } }, { headers: CORS_HEADERS });
   555→        }
   556→        try {
   557→          const chapters = await lxChapters(batchId, subjectId, page);
   558→          const data = chapters.map((c, i) => ({

   851→/** Flatten test ids from scalars / arrays / objects ({testId}|{_id}|{id}). */
   852→function collectTestIds(source: unknown, out: string[], seen: Set<string>): void {
   853→  if (source === undefined || source === null) return;
   854→  if (Array.isArray(source)) {
   855→    source.forEach((item) => collectTestIds(item, out, seen));
   856→    return;
   857→  }
   858→  if (typeof source === "object") {
   859→    const o = source as Record<string, unknown>;
   860→    collectTestIds(o.testId ?? o._id ?? o.id, out, seen);
   861→    return;
   862→  }
   863→  const s = String(source).trim();
   864→  if (s && !seen.has(s)) {
   865→    seen.add(s);
   866→    out.push(s);
   867→  }
   868→}
