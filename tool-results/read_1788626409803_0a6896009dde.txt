    18→export function hashString(str: string): number {
    19→  let h = 2166136261;
    20→  for (let i = 0; i < str.length; i++) {
    21→    h ^= str.charCodeAt(i);
    22→    h = Math.imul(h, 16777619);
    23→  }
    24→  return h >>> 0;
    25→}
    26→
    27→export function makeRng(seed: string): () => number {
    28→  let a = hashString(seed);
    29→  return function () {
    30→    a |= 0;
    31→    a = (a + 0x6d2b79f5) | 0;
    32→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    33→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    34→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    35→  };
    36→}
    37→
    38→function pick<T>(rng: () => number, arr: T[]): T {
    39→  return arr[Math.floor(rng() * arr.length)];
    40→}

   156→export function searchCatalog(query: string, page: number, limit: number) {
   157→  const q = (query || '').toLowerCase().trim();
   158→  const filtered = BATCH_CATALOG.filter((b) => {
   159→    const bName = (b.name || '').toLowerCase().trim();
   160→    const bId = (b._id || '').toLowerCase();
   161→    if (BLOCKED_BATCH_NAMES.includes(bName)) return false;
   162→    if (!q) return true;
   163→    return bName.includes(q) || bId === q;
   164→  });
   165→  const offset = (page - 1) * limit;
   166→  return {
   167→    total: filtered.length,
   168→    page,
   169→    limit,
   170→    data: filtered.slice(offset, offset + limit),
   171→    hasMore: (page * limit) < filtered.length,
   172→  };
   173→}

   180→function imageRefFromUrl(url?: string | null): ImageRef {
   181→  if (!url || url.trim() === '') {
   182→    return { baseUrl: 'https://store.pw.live/', key: 'footer-logo.svg' };
   183→  }
   184→  if (url.startsWith(PW_STATIC)) {
   185→    return { baseUrl: PW_STATIC, key: url.slice(PW_STATIC.length) };
   186→  }
   187→  // Non-PW urls: keep absolute in key, empty base
   188→  return { baseUrl: '', key: url };
   189→}

   191→export function imageUrlOf(ref?: ImageRef | string | null): string {
   192→  if (!ref) return 'https://store.pw.live/footer-logo.svg';
   193→  if (typeof ref === 'string') return ref;
   194→  const url = (ref.baseUrl || '') + (ref.key || '');
   195→  if (!url) return 'https://store.pw.live/footer-logo.svg';
   196→  if (url.startsWith('http')) return url;
   197→  return PW_STATIC + url;
   198→}

   212→function pickTeachersFor(subject: string, batchId: string): TeacherPublic[] {
   213→  const pool = TEACHER_POOL[subject] || TEACHER_POOL.General;
   214→  const rng = makeRng(`teachers_${batchId}_${subject}`);
   215→  const count = Math.min(pool.length, 1 + Math.floor(rng() * 2));
   216→  const chosen: TeacherSeedLike[] = [];
   217→  const used = new Set<number>();
   218→  while (chosen.length < count && used.size < pool.length) {
   219→    const idx = Math.floor(rng() * pool.length);
   220→    if (used.has(idx)) continue;
   221→    used.add(idx);
   222→    chosen.push(pool[idx]);
   223→  }
   224→  return chosen.map((t, i) => {
   225→    const tid = `tchr_${hashString(batchId + subject + t.name).toString(36)}`;
   226→    return {
   227→      _id: tid,
   228→      firstName: t.name.split(' ')[0],
   229→      lastName: t.name.split(' ').slice(1).join(' '),
   230→      subject,
   231→      experience: t.exp,
   232→      qualification: t.qual,
   233→      featuredLine: t.line,
   234→      tags: t.tags,
   235→      about: `${t.name} is one of the most loved ${subject} faculty, known for making concepts crystal clear with in-depth explanations, exam-oriented problem solving and regular mentoring sessions. Over ${t.exp} years of teaching experience has helped thousands of students crack competitive & board exams.`,
   236→      imageId: teacherAvatar(tid + i),
   237→    };
   238→  });
   239→}

   249→export function getSubjects(batchId: string, batchName: string): BatchSubject[] {
   250→  const cacheKey = batchId + '|' + batchName;
   251→  if (subjectCache.has(cacheKey)) return subjectCache.get(cacheKey)!;
   252→  const cls = classifyBatch(batchName, (s) => hashString(s));
   253→  const subjects: BatchSubject[] = cls.subjects.map((sub, i) => {
   254→    const sid = `sbj_${hashString(batchId + sub).toString(36)}`;
   255→    return {
   256→      _id: sid,
   257→      subject: sub,
   258→      subjectId: `sint_${hashString('internal_' + batchId + sub).toString(36)}`,
   259→      teacherIds: pickTeachersFor(sub, batchId),
   260→      // index used by callers
   261→      ...(i >= 0 ? {} : {}),
   262→    };
   263→  });
   264→  subjectCache.set(cacheKey, subjects);
   265→  return subjects;
   266→}

   292→export function getBatchDetail(batchId: string): BatchDetail | null {
   293→  if (detailCache.has(batchId)) return detailCache.get(batchId)!;
   294→  const cat = BATCH_CATALOG.find((b) => b._id === batchId);
   295→  if (!cat) return null;
   296→  const rng = makeRng('detail_' + batchId);
   297→  const cls = classifyBatch(cat.name, (s) => hashString(s));
   298→  const yearMatch = cat.name.match(/20\d{2}/);
   299→  const year = yearMatch ? parseInt(yearMatch[0]) : new Date().getFullYear() + 1;
   300→  const startYear = year - 1;
   301→  const price = Math.floor(rng() * 45) * 100 + 1200;
   302→  const discount = price > 1500 ? Math.floor(rng() * 20) * 100 + 500 : 0;
   303→
   304→  const descFn = EXAM_DESC[cls.exam] || EXAM_DESC.GENERAL;
   305→  const detail: BatchDetail = {
   306→    _id: cat._id,
   307→    id: cat._id,
   308→    name: cat.name,
   309→    previewImage: imageRefFromUrl(cat.imageUrl || cat.previewImage),
   310→    description: descFn(cat.name, cls.classLevel || 'exam'),
   311→    shortDescription: descFn(cat.name, cls.classLevel || 'exam').slice(0, 120) + '…',
   312→    price,
   313→    discount,
   314→    validity: cls.exam === 'BANKING' ? '6 Months' : 'Till Exams',
   315→    startDate: new Date(`${startYear}-04-20T00:00:00Z`).toISOString(),
   316→    endDate: new Date(`${year}-06-15T00:00:00Z`).toISOString(),
   317→    class: {
   318→      name: cls.classLevel || CLASS_LEVELS[hashString(batchId) % CLASS_LEVELS.length],
   319→      _id: `cls_${hashString(batchId).toString(36)}`,
   320→    },
   321→    subjects: getSubjects(batchId, cat.name),
   322→    channelId: `ch_${hashString('ch' + batchId).toString(36)}`,
   323→    facultyChannelId: `fch_${hashString('fch' + batchId).toString(36)}`,
   324→    cohortId: `coh_${hashString('coh' + batchId).toString(36)}`,
   325→  };
   326→  detailCache.set(batchId, detail);
   327→  return detail;
   328→}

   383→export function getContentNodes(
   384→  batchId: string,
   385→  subjectId: string,
   386→  tagId: string,
   387→  contentType: string,
   388→): ContentNode[] {
   389→  const detail = getBatchDetail(batchId);
   390→  if (!detail) return [];
   391→  const subject = detail.subjects.find((s) => s._id === subjectId || s.subjectId === subjectId);
   392→  const bank =
   393→    subject
   394→      ? classifyBatch(detail.name, (s) => hashString(s)).chaptersFor(subject.subject)
   395→      : CHAPTER_BANKS.physics12;
   396→  const chapter = bank.find((c) => `chp_${hashString(batchId + (subject ? subject._id : '') + c).toString(36)}` === tagId)
   397→    || bank.find((c) => `rtag_${hashString(batchId + c).toString(36)}` === tagId)
   398→    || null;
   399→
   400→  const chapterName = chapter || 'Complete Chapter';
   401→  const rng = makeRng(`content_${batchId}_${subjectId}_${tagId}_${contentType}`);
   402→  const lectureCount = 9 + Math.floor(rng() * 5);
   403→  const lectureStyle = pick(rng, LECTURE_STYLES);
   404→  const now = Date.now();
   405→
   406→  const nodes: ContentNode[] = [];
   407→
   408→  const buildLecture = (i: number, type: 'LECTURE' | 'DPP_VIDEOS', title: string): ContentNode => {
   409→    const vid = `vid_${hashString(`${batchId}_${tagId}_${type}_${i}_${title}`).toString(36)}`;
   410→    const daysAgo = Math.max(0, lectureCount - i) * 2 + Math.floor(rng() * 2);
   411→    const date = new Date(now - daysAgo * 86400000).toISOString();
   412→    const durH = 1 + Math.floor(rng() * 2);
   413→    const durM = Math.floor(rng() * 60);
   414→    return {
   415→      _id: `sch_${hashString(`sch_${batchId}_${tagId}_${type}_${i}_${title}`).toString(36)}`,
   416→      type,
   417→      data: {
   418→        _id: vid,
   419→        topic: title,
   420→        name: title,
   421→        date,
   422→        videoDetails: {
   423→          _id: vid,
   424→          image: `https://picsum.photos/seed/${vid}/480/270`,
   425→          videoUrl: DEMO_VIDEOS[hashString(vid) % DEMO_VIDEOS.length],
   426→          duration: `${durH}h ${durM}m`,
   427→        },
   428→        homeworkIds: [{ topic: `${title} - Class Notes`, attachmentIds: [noteRef(vid, 'notes')] }],
   429→        exerciseIds: [],
   430→        subjectId: subject ? { _id: subject.subjectId } : undefined,
   431→      },
   432→    };
   433→  };
   434→
   435→  if (contentType === 'ALL' || contentType === 'LECTURE' || contentType === 'LECTURES') {
   436→    for (let i = 1; i <= lectureCount; i++) {
   437→      const title = lectureStyle(chapterName, i);
   438→      nodes.push(buildLecture(i, 'LECTURE', title));
   439→    }
   440→  }
   441→  if (contentType === 'ALL' || contentType === 'NOTES') {
   442→    const noteTitles = [
   443→      `${chapterName} - Complete Class Notes`,
   444→      `${chapterName} - Short Revision Notes`,
   445→      `${chapterName} - Formula & Concept Sheet`,
   446→    ];
   447→    noteTitles.forEach((t, i) => {
   448→      const nid = `note_${hashString(`${batchId}_${tagId}_notes_${i}_${t}`).toString(36)}`;
   449→      nodes.push({
   450→        _id: `sch_${hashString(`sch_${batchId}_${tagId}_NOTES_${i}`).toString(36)}`,
   451→        type: 'NOTES',
   452→        data: {
   453→          _id: nid,
   454→          topic: t,
   455→          name: t,
   456→          homeworkIds: [{ topic: t, attachmentIds: [noteRef(nid, 'notes')] }],
   457→          exerciseIds: [],
   458→        },
   459→      });
   460→    });
   461→  }
   462→  if (contentType === 'ALL' || contentType === 'DPP_PDF') {
   463→    const dppCount = 3 + Math.floor(rng() * 3);
   464→    for (let i = 1; i <= dppCount; i++) {
   465→      const t = `${chapterName} - DPP ${String(i).padStart(2, '0')} (PDF)`;
   466→      const did = `dppdf_${hashString(`${batchId}_${tagId}_dppdf_${i}`).toString(36)}`;
   467→      nodes.push({
   468→        _id: `sch_${hashString(`sch_${batchId}_${tagId}_DPP_PDF_${i}`).toString(36)}`,
   469→        type: 'DPP_PDF',
   470→        data: {
   471→          _id: did,
   472→          topic: t,
   473→          name: t,
   474→          homeworkIds: [{ topic: t, attachmentIds: [noteRef(did, 'dpp')] }],
   475→          exerciseIds: [],
   476→        },
   477→      });
   478→    }
   479→  }
   480→  if (contentType === 'ALL' || contentType === 'DPP_VIDEOS') {
   481→    const dvCount = 3 + Math.floor(rng() * 3);
   482→    for (let i = 1; i <= dvCount; i++) {
   483→      const t = `${chapterName} - DPP ${String(i).padStart(2, '0')} Video Discussion`;
   484→      nodes.push(buildLecture(i, 'DPP_VIDEOS', t));
   485→    }
   486→  }
   487→  return nodes;
   488→}

   498→export function resolveAttachmentUrl(ref: ImageRef | { baseUrl?: string; key?: string } | undefined): string {
   499→  if (!ref) return '/api/notes/generic';
   500→  const url = (ref.baseUrl || '') + (ref.key || '');
   501→  if (url.startsWith('__LOCALPDF__')) {
   502→    // key format: __LOCALPDF__<kind>_<id>  ->  /api/notes/<kind>_<id>
   503→    return `/api/notes/${url.replace('__LOCALPDF__', '')}`;
   504→  }
   505→  if (url.startsWith('http')) return url;
   506→  return PW_STATIC + url;
   507→}

   637→export function getTodaySchedule(batchId: string): ScheduleEvent[] {
   638→  const detail = getBatchDetail(batchId);
   639→  if (!detail) return [];
   640→  const rng = makeRng(`tdy_${batchId}_${new Date().toISOString().slice(0, 10)}`);
   641→  const now = new Date();
   642→  const events: ScheduleEvent[] = [];
   643→  const count = 2 + Math.floor(rng() * 3);
   644→  for (let i = 0; i < count; i++) {
   645→    const subject = detail.subjects[Math.floor(rng() * detail.subjects.length)];
   646→    const teacher = subject.teacherIds[0];
   647→    const startOffsetMin = Math.floor((rng() - 0.5) * 480); // -4h to +4h
   648→    const start = new Date(now.getTime() + startOffsetMin * 60000);
   649→    start.setSeconds(0, 0);
   650→    const end = new Date(start.getTime() + (60 + Math.floor(rng() * 60)) * 60000);
   651→    const vid = `live_${hashString(`${batchId}_today_${i}_${start.toISOString().slice(0, 13)}`).toString(36)}`;
   652→    const isLive = start <= now && end >= now;
   653→    const titleBank = [
   654→      `${subject.subject} | ${teacher ? teacher.firstName + ' Sir' : 'Expert Faculty'} Special Session`,
   655→      `${subject.subject} Marathon - Rapid Revision`,
   656→      `${subject.subject} Doubt Destroyer Session`,
   657→      `${subject.subject} Important Questions Sprint`,
   658→    ];
   659→    const title = titleBank[hashString(vid) % titleBank.length];
   660→    events.push({
   661→      _id: `tsch_${hashString(`tsch_${batchId}_${i}_${start.toISOString().slice(0, 13)}`).toString(36)}`,
   662→      type: 'LECTURE',
   663→      data: {
   664→        _id: vid,
   665→        topic: title,
   666→        startTime: start.toISOString(),
   667→        endTime: end.toISOString(),
   668→        status: isLive ? 'LIVE' : 'SCHEDULED',
   669→        videoDetails: { image: `https://picsum.photos/seed/${vid}/480/270` },
   670→        teachers: teacher ? [teacher._id] : [],
   671→        subjectId: { _id: subject._id },
   672→        homeworkIds: rng() > 0.5 ? [{ topic: 'Session Notes', attachmentIds: [noteRef(vid + '_hw', 'notes')] }] : [],
   673→        exerciseIds: [],
   674→      },
   675→    });
   676→  }
   677→  // Sort: live first, then upcoming
   678→  events.sort((a, b) => new Date(a.data.startTime).getTime() - new Date(b.data.startTime).getTime());
   679→  return events;
   680→}

   694→export function getNotifications(batchId: string, page: number, limit: number): { data: NotificationItem[]; hasMore: boolean } {
   695→  const detail = getBatchDetail(batchId);

   697→  const total = 66;

   700→  for (let i = offset; i < Math.min(offset + limit, total); i++) {
   701→    const tpl = NOTIF_TEMPLATES[hashString(batchId + 'n' + i) % NOTIF_TEMPLATES.length];
   702→    const hoursAgo = i * 7 + (hashString(batchId + 'h' + i) % 6);

   705→      type: tpl.type,
   706→      heading: tpl.heading,
   707→      announcement: tpl.body,

   709→      isRead: hoursAgo > 48,
   710→      attachment: null,
   711→    });
   712→  }
   713→  return { data: items, hasMore: (page * limit) < total };

   837→export function getQuiz(batchId: string, testId: string): QuizPayload | null {
   838→  const detail = getBatchDetail(batchId);
   839→  const title = detail ? findTestName(batchId, testId) : 'Practice Quiz';
   840→  const rng = makeRng(`quiz_${testId}`);
   841→  const count = 5 + Math.floor(rng() * 4); // 5-8 questions
   842→  const topic = title.replace(/\s*-\s*DPP\s*\d+/i, '').replace(/_\d+$/, '');
   843→  const questions: QuizQuestion[] = [];
   844→  const used = new Set<number>();
   845→  for (let i = 0; i < count; i++) {
   846→    let tIdx = hashString(testId + '_' + i) % QUESTION_TEMPLATES.length;
   847→    let guard = 0;
   848→    while (used.has(tIdx) && guard < QUESTION_TEMPLATES.length) {
   849→      tIdx = (tIdx + 1) % QUESTION_TEMPLATES.length;
   850→      guard++;
   851→    }
   852→    used.add(tIdx);
   853→    const tpl = QUESTION_TEMPLATES[tIdx](topic);
   854→    // Deterministic shuffle of options
   855→    const order = [0, 1, 2, 3].sort(() => rng() - 0.5);
   856→    questions.push({
   857→      _id: `q_${hashString(`${testId}_q${i}`).toString(36)}`,
   858→      q: tpl.q,
   859→      options: order.map((o) => tpl.opts[o]),
   860→      correct: order.indexOf(tpl.correct),
   861→      explanation: tpl.why,
   862→    });
   863→  }
   864→  return {
   865→    testId,
   866→    title,
   867→    durationMin: 15 + Math.floor(rng() * 3) * 5,
   868→    totalMarks: questions.length * 4,
   869→    questions,
   870→  };
   871→}
