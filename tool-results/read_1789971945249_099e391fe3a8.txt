     1→// ============================================================================
     2→// Study Bison — offline lecture library (PWA downloads).
     3→//
     4→// YouTube-style "Available offline": every lecture can be downloaded once and
     5→// then watched with the network fully off. Media bytes live in the Cache API
     6→// (one cache per lecture folder: `sb-video-<folder>`) and are served by the
     7→// service worker (public/sw.js) even when the device is offline.
     8→//
     9→// Flow:
    10→//   1. The player resolves the playable relay URL (/api/pw/<folder>/s/<sig>/…)
    11→//      — all segment URIs are same-origin and signed statelessly in the PATH,
    12→//      so they stay byte-identical between download time and offline playback.
    13→//   2. collectParts() walks the manifest (shaka headless for DASH, a text
    14→//      parser for HLS) and returns the full part list for ONE good quality.
    15→//   3. startDownload() fetches every part (concurrency 4, resume-safe) and
    16→//      cache.put()s it into the lecture cache. Progress is persisted in
    17→//      IndexedDB so downloads survive reloads and can be paused/resumed.
    18→//   4. Offline the player finds the stored record, re-uses the stored
    19→//      manifest URL + ClearKeys and the SW serves every byte from the cache.
    20→//
    21→// There is no artificial cap on the number of downloads — the only limit is
    22→// the device's own storage quota (we also request persistent storage so the
    23→// browser's janitor does not evict lectures).
    24→// ============================================================================
    25→

    79→/** Extract the media folder id from a relay/CloudFront media URL. */
    80→export function folderFromMediaUrl(url: string): string {
    81→  try {
    82→    const u = new URL(url, typeof location !== 'undefined' ? location.href : 'https://x.invalid');
    83→    if (u.origin === location.origin) {
    84→      // /api/pw/<folder>/…  |  /api/pw/h/<host>/<folder>/…  |  /api/pw/<folder>/s/<sig>/…
    85→      const m = u.pathname.match(/^\/api\/pw\/(?:h\/[^/]+\/)?([^/]+)\//);
    86→      return m ? m[1] : '';
    87→    }
    88→    if (/\.cloudfront\.net$/i.test(u.hostname)) {
    89→      const m = u.pathname.match(/^\/([^/]+)\//);
    90→      return m ? m[1] : '';
    91→    }
    92→  } catch {
    93→    /* ignore */
    94→  }
    95→  return '';
    96→}

   102→let dbPromise: Promise<IDBDatabase> | null = null;
   103→
   104→function openDb(): Promise<IDBDatabase> {
   105→  if (dbPromise) return dbPromise;
   106→  dbPromise = new Promise((resolve, reject) => {
   107→    try {
   108→      const req = indexedDB.open(DB_NAME, DB_VERSION);
   109→      req.onupgradeneeded = () => {
   110→        const db = req.result;
   111→        if (!db.objectStoreNames.contains(STORE)) {
   112→          db.createObjectStore(STORE, { keyPath: 'id' });
   113→        }
   114→      };
   115→      req.onsuccess = () => resolve(req.result);
   116→      req.onerror = () => reject(req.error || new Error('IndexedDB unavailable'));
   117→    } catch (e) {
   118→      reject(e instanceof Error ? e : new Error('IndexedDB unavailable'));
   119→    }
   120→  });
   121→  return dbPromise;
   122→}
   123→
   124→async function tx<T>(mode: IDBTransactionMode, fn: (store: IDBObjectStore) => IDBRequest<T>): Promise<T> {
   125→  const db = await openDb();
   126→  return new Promise<T>((resolve, reject) => {
   127→    const t = db.transaction(STORE, mode);
   128→    const req = fn(t.objectStore(STORE));
   129→    req.onsuccess = () => resolve(req.result);
   130→    req.onerror = () => reject(req.error || new Error('IndexedDB error'));
   131→  });
   132→}

   143→export function subscribeDownloads(cb: () => void): () => void {
   144→  window.addEventListener(CHANGE_EVENT, cb);
   145→  window.addEventListener('online', cb);
   146→  window.addEventListener('offline', cb);

   148→    window.removeEventListener(CHANGE_EVENT, cb);
   149→    window.removeEventListener('online', cb);

   154→export async function listDownloads(): Promise<SbDownload[]> {
   155→  try {
   156→    const all = await tx<SbDownload[]>('readonly', (s) => s.getAll() as IDBRequest<SbDownload[]>);
   157→    return (all || []).sort((a, b) => (a.createdAt < b.createdAt ? 1 : -1));
   158→  } catch {
   159→    return [];
   160→  }

   172→async function putDownload(rec: SbDownload): Promise<void> {
   173→  rec.updatedAt = new Date().toISOString();
   174→  // Persist WITHOUT the parts array: a 2-hour lecture carries thousands of
   175→  // ~100-char URLs (~0.5MB per save × every 5 parts ≈ 100s of MB of IndexedDB
   176→  // writes per download — a real throughput killer). The part list lives in
   177→  // memory while running and is re-collected from the stored manifest on
   178→  // resume (same sig → same URIs → cached parts still match).
   179→  const slim: SbDownload = { ...rec, parts: [] };
   180→  await tx('readwrite', (s) => s.put(slim) as IDBRequest<IDBValidKey>);
   181→}

   187→/** Find the download record matching a lecture (by id candidates or vUrl). */
   188→export async function findDownloadFor(ids: {
   189→  videoId?: string;
   190→  realVideoId?: string;
   191→  vUrl?: string;
   192→}): Promise<SbDownload | null> {
   193→  const cands = new Set<string>();
   194→  if (ids.videoId) cands.add(ids.videoId);
   195→  if (ids.realVideoId) cands.add(ids.realVideoId);
   196→  if (ids.vUrl) {
   197→    cands.add(`url:${ids.vUrl.split('?')[0]}`);
   198→    cands.add(ids.vUrl);
   199→  }
   200→  if (cands.size === 0) return null;
   201→  const all = await listDownloads();
   202→  for (const rec of all) {
   203→    if (cands.has(rec.id) || (ids.vUrl && rec.vUrl && rec.vUrl === ids.vUrl)) return rec;
   204→  }
   205→  return null;
   206→}

   242→function uniqKeepOrder(items: string[]): string[] {
   243→  const seen = new Set<string>();
   244→  const out: string[] = [];
   245→  for (const u of items) {
   246→    if (!u || seen.has(u)) continue;
   247→    seen.add(u);
   248→    out.push(u);
   249→  }
   250→  return out;
   251→}

   253→async function fetchText(url: string): Promise<string> {
   254→  const res = await fetch(url);
   255→  if (!res.ok) throw new Error(`HTTP ${res.status} for playlist`);
   256→  return res.text();
   257→}

   259→/**
   260→ * HLS — relay playlists are fully rewritten (absolute same-origin URIs), so a
   261→ * plain text walk is exact. For a master playlist we pick ONE rendition
   262→ * (≤720p, highest bitrate) and download only that media playlist + its
   263→ * segments — the offline player opens the variant playlist directly.
   264→ */
   265→async function collectHlsParts(url: string, depth = 0): Promise<CollectedParts> {
   266→  // absolutize (input may be a same-origin relative path like /e2ehls/main.m3u8)
   267→  const absUrl = new URL(url, location.href).href;
   268→  const text = await fetchText(absUrl);
   269→  const lines = text.split('\n').map((l) => l.trim());
   270→
   271→  // master playlist? → choose one variant
   272→  const variants: Array<{ url: string; bandwidth: number; height: number }> = [];
   273→  for (let i = 0; i < lines.length; i++) {
   274→    if (!/^#EXT-X-STREAM-INF/i.test(lines[i])) continue;
   275→    const attrs = lines[i].slice(lines[i].indexOf(':') + 1);
   276→    const res = /RESOLUTION=(\d+)x(\d+)/i.exec(attrs);
   277→    const bw = /BANDWIDTH=(\d+)/i.exec(attrs);
   278→    let j = i + 1;
   279→    while (j < lines.length && (!lines[j] || lines[j].startsWith('#'))) j++;
   280→    if (j < lines.length) {
   281→      variants.push({
   282→        url: new URL(lines[j], absUrl).href,
   283→        bandwidth: bw ? Number(bw[1]) || 0 : 0,
   284→        height: res ? Number(res[2]) || 0 : 0,
   285→      });
   286→    }
   287→  }
   288→
   289→  let playlistUrl = absUrl;
   290→  let height: number | null = null;
   291→  if (variants.length > 0) {
   292→    const capped = variants.filter((v) => v.height > 0 && v.height <= 720);
   293→    const best =
   294→      capped.sort((a, b) => b.bandwidth - a.bandwidth)[0] ||
   295→      variants.slice().sort((a, b) => a.bandwidth - b.bandwidth)[0];
   296→    playlistUrl = best.url;
   297→    height = best.height || null;
   298→  }
   299→
   300→  const mediaText = playlistUrl === absUrl ? text : await fetchText(playlistUrl);
   301→  const mediaLines = mediaText.split('\n').map((l) => l.trim());
   302→
   303→  // defensive: a variant that is itself still a master (depth guard = 2)
   304→  if (depth < 2 && mediaLines.some((l) => /^#EXT-X-STREAM-INF/i.test(l))) {
   305→    return collectHlsParts(playlistUrl, depth + 1);
   306→  }
   307→
   308→  const parts: string[] = [playlistUrl];
   309→  for (const l of mediaLines) {
   310→    if (!l) continue;
   311→    if (l.startsWith('#')) {
   312→      if (/^#EXT-X-(KEY|MAP|MEDIA|SESSION-KEY)/i.test(l)) {
   313→        const m = /URI="([^"]+)"/i.exec(l);
   314→        if (m) parts.push(new URL(m[1], playlistUrl).href);
   315→      }
   316→      continue;
   317→    }
   318→    parts.push(new URL(l, playlistUrl).href);
   319→  }
   320→  return { parts: uniqKeepOrder(parts), height, playbackUrl: playlistUrl };
   321→}

   323→/** Cheap ranged probe — verifies a URL is actually fetchable (200/206). */
   324→async function probePart(u: string, timeoutMs = 8000): Promise<boolean> {
   325→  try {
   326→    const ctrl = new AbortController();
   327→    const t = setTimeout(() => ctrl.abort(), timeoutMs);
   328→    const res = await fetch(u, { headers: { Range: 'bytes=0-1' }, signal: ctrl.signal });
   329→    clearTimeout(t);
   330→    return res.status < 400;
   331→  } catch {
   332→    return false;
   333→  }
   334→}

   336→/** Verify the collected list: probe the manifest + the first media segment. */
   337→async function probeCollected(parts: string[]): Promise<boolean> {
   338→  if (parts.length === 0) return false;
   339→  if (!(await probePart(parts[0]))) return false;
   340→  // find the first non-init part (the first real media segment)
   341→  const seg = parts.slice(1).find((u) => !/init\.(mp4|m4s)$/i.test(u.split('?')[0]));
   342→  if (!seg) return true; // playlist-only download
   343→  return probePart(seg);
   344→}

   346→/**
   347→ * DASH via shaka headless (no <video>): parse the MPD and walk the chosen
   348→ * variant's SegmentReferences. NOTE: some shaka versions substitute $Number$
   349→ * with time values when a SegmentTimeline is present — every collected list
   350→ * is therefore probe-verified by the caller before use.
   351→ */
   352→async function collectDashShaka(url: string, clearKeys: Record<string, string> | null): Promise<CollectedParts> {
   353→  const shaka = (await import('shaka-player')).default;
   354→  shaka.polyfill.installAll();
   355→  const player = new shaka.Player();
   356→  player.configure({
   357→    manifest: {
   358→      dash: { ignoreDrmInfo: true }, // we store encrypted bytes; keys only matter at playback
   359→      retryParameters: { timeout: 20000, maxAttempts: 3 },
   360→    },
   361→  });
   362→  try {
   363→    await player.load(url, 0, 'application/dash+xml');
   364→    const manifest = player.getManifest();
   365→    if (!manifest) throw new Error('Manifest did not parse');
   366→    const variants = manifest.variants.filter((v) => v.video);
   367→    if (variants.length === 0) throw new Error('No video variants in manifest');
   368→
   369→    // pick ≤720p with the highest bitrate (fallback: the smallest available)
   370→    const capped = variants.filter((v) => (v.video?.height || 0) > 0 && (v.video?.height || 0) <= 720);
   371→    const chosen =
   372→      capped.sort((a, b) => (b.bandwidth || 0) - (a.bandwidth || 0))[0] ||
   373→      variants.slice().sort((a, b) => (a.video?.height || 4320) - (b.video?.height || 4320))[0] ||
   374→      variants[0];
   375→
   376→    const height = chosen.video?.height || null;
   377→    const parts: string[] = [];
   378→    const seenStreams = new Set<number>();
   379→    const streams = [chosen.video, chosen.audio].filter(Boolean) as NonNullable<typeof chosen.video | typeof chosen.audio>[];
   380→    for (const stream of streams) {
   381→      if (seenStreams.has(stream.id)) continue;
   382→      seenStreams.add(stream.id);
   383→      try {
   384→        await stream.createSegmentIndex();
   385→      } catch {
   386→        /* already created */
   387→      }
   388→      const segIndex = stream.segmentIndex;
   389→      if (!segIndex) continue;
   390→      for (const ref of segIndex) {
   391→        if (!ref) continue;
   392→        try {
   393→          const init = ref.initSegmentReference;
   394→          if (init) {
   395→            for (const u of init.getUris()) parts.push(u);
   396→          }
   397→          for (const u of ref.getUris()) parts.push(u);
   398→        } catch {
   399→          /* skip broken reference */
   400→        }
   401→      }
   402→    }
   403→    if (parts.length === 0) throw new Error('No segments found in manifest');
   404→    return { parts: uniqKeepOrder(parts), height, playbackUrl: url };
   405→  } finally {
   406→    try {
   407→      await player.destroy();
   408→    } catch {
   409→      /* ignore */
   410→    }
   411→  }
   412→}

   414→/**
   415→ * MPD parser (primary DASH path) — handles SegmentTemplate with $Number$
   416→ * (+ SegmentTimeline or @duration) and $Time$, SegmentList and plain BaseURL
   417→ * representations, per the DASH spec ($Number$ = segment sequence number).
   418→ */
   419→async function collectDashText(url: string): Promise<CollectedParts> {
   420→  const xml = new DOMParser().parseFromString(await fetchText(url), 'application/xml');
   421→  const q = (el: Element | Document, name: string): Element | null => {
   422→    const list = (el as Element).getElementsByTagName(name);
   423→    return list.length ? list[0] : null;
   424→  };
   425→  const txt = (el: Element, name: string): string => {
   426→    const n = q(el, name);
   427→    return n && n.textContent ? n.textContent.trim() : '';
   428→  };
   429→  const num = (el: Element, name: string): number => Number(el.getAttribute(name) || 0);
   430→  const attr = (el: Element, name: string): string => el.getAttribute(name) || '';
   431→
   432→  const resolve = (base: string, uri: string): string => new URL(uri, new URL(base, location.href)).href;
   433→
   434→  const mpdBase = txt(xml, 'BaseURL') ? resolve(url, txt(xml, 'BaseURL')) : url;
   435→  const isoDuration = attr(xml.getElementsByTagName('MPD')[0] || xml.documentElement, 'mediaPresentationDuration');
   436→  const totalSec = (() => {
   437→    const m = /^P(?:(\d+(?:\.\d+)?)Y)?(?:(\d+(?:\.\d+)?)M)?(?:(\d+(?:\.\d+)?)D)?(?:T(?:(\d+(?:\.\d+)?)H)?(?:(\d+(?:\.\d+)?)M)?(?:(\d+(?:\.\d+)?)S)?)?$/.exec(isoDuration);
   438→    if (!m) return 0;
   439→    return (Number(m[4]) || 0) * 3600 + (Number(m[5]) || 0) * 60 + (Number(m[6]) || 0);
   440→  })();
   441→
   442→  const videoReps: Array<{ el: Element; base: string; height: number; bandwidth: number }> = [];
   443→  const audioReps: Array<{ el: Element; base: string }> = [];
   444→  for (const aset of Array.from(xml.getElementsByTagName('AdaptationSet'))) {
   445→    const asBase = txt(aset, 'BaseURL') ? resolve(mpdBase, txt(aset, 'BaseURL')) : mpdBase;
   446→    const contentType = (attr(aset, 'contentType') || attr(aset, 'mimeType')).toLowerCase();
   447→    const isAudio = /audio/.test(contentType);
   448→    for (const rep of Array.from(aset.getElementsByTagName('Representation'))) {
   449→      const repBase = txt(rep, 'BaseURL') ? resolve(asBase, txt(rep, 'BaseURL')) : asBase;
   450→      if (isAudio || /audio/.test(attr(rep, 'mimeType').toLowerCase())) {
   451→        audioReps.push({ el: rep, base: repBase });
   452→      } else {
   453→        videoReps.push({ el: rep, base: repBase, height: num(rep, 'height'), bandwidth: num(rep, 'bandwidth') });
   454→      }
   455→    }
   456→  }
   457→  if (videoReps.length === 0) throw new Error('No representations found');
   458→
   459→  const capped = videoReps.filter((r) => r.height > 0 && r.height <= 720);
   460→  const chosen =
   461→    capped.sort((a, b) => b.bandwidth - a.bandwidth)[0] ||
   462→    videoReps.slice().sort((a, b) => a.height - b.height)[0];
   463→  const height = chosen.height || null;
   464→
   465→  const expandRep = (rep: Element, base: string, out: string[]) => {
   466→    const repBase = txt(rep, 'BaseURL') ? resolve(base, txt(rep, 'BaseURL')) : base;
   467→    const aset = rep.parentElement as Element | null;
   468→    const tmpl = (q(rep, 'SegmentTemplate') as Element | null) || (aset ? q(aset, 'SegmentTemplate') : null);
   469→    if (tmpl) {
   470→      const mediaT = attr(tmpl, 'media');
   471→      const initT = attr(tmpl, 'initialization');
   472→      const timescale = num(tmpl, 'timescale') || 1;
   473→      const segDur = num(tmpl, 'duration') / timescale;
   474→      const startNumber = num(tmpl, 'startNumber') || (/\$Number%0(\d+)d\$/.test(mediaT) ? 1 : 0);
   475→      const repId = attr(rep, 'id');
   476→      const bandwidth = attr(rep, 'bandwidth');
   477→      const fill = (t: string): string =>
   478→        t
   479→          .replace(/\$RepresentationID\$/g, repId)
   480→          .replace(/\$Bandwidth\$/g, bandwidth)
   481→          .replace(/\$\$/g, '$');
   482→      if (initT) out.push(resolve(repBase, fill(initT)));
   483→      const timeline = q(tmpl, 'SegmentTimeline');
   484→      if (timeline) {
   485→        let t = 0;
   486→        let segIdx = 0;
   487→        for (const s of Array.from(timeline.getElementsByTagName('S'))) {
   488→          const st = s.getAttribute('t');
   489→          if (st !== null) t = Number(st);
   490→          const d = Number(s.getAttribute('d') || 0);
   491→          const r = Number(s.getAttribute('r') || 0);
   492→          for (let i = 0; i <= Math.max(0, r); i++) {
   493→            // $Number$ = sequence number (startNumber + index), $Time$ = media time
   494→            const filled = fill(mediaT)
   495→              .replace(/\$Number(?:%0(\d+)d)?\$/g, (_m, pad?: string) => String(startNumber + segIdx).padStart(pad ? Number(pad) : 0, '0'))
   496→              .replace(/\$Time\$/g, String(t));
   497→            out.push(resolve(repBase, filled));
   498→            segIdx++;
   499→            t += d;
   500→          }
   501→        }
   502→      } else if (segDur > 0 && totalSec > 0) {
   503→        const count = Math.ceil(totalSec / segDur);
   504→        for (let i = 0; i < count; i++) {
   505→          out.push(resolve(repBase, fill(mediaT).replace(/\$Number(?:%0(\d+)d)?\$/g, (_m, pad?: string) => String(startNumber + i).padStart(pad ? Number(pad) : 0, '0'))));
   506→        }
   507→      }
   508→      return;
   509→    }
   510→    const list = q(rep, 'SegmentList') || (aset ? q(aset, 'SegmentList') : null);
   511→    if (list) {
   512→      const initEl = q(list, 'Initialization');
   513→      if (initEl) {
   514→        const src = attr(initEl, 'sourceURL');
   515→        if (src) out.push(resolve(repBase, src));
   516→      }
   517→      for (const su of Array.from(list.getElementsByTagName('SegmentURL'))) {
   518→        const media = attr(su, 'media');
   519→        if (media) out.push(resolve(repBase, media));
   520→      }
   521→      return;
   522→    }
   523→    // plain <BaseURL> representation (single self-contained file)
   524→    const single = txt(rep, 'BaseURL');
   525→    if (single) out.push(resolve(base, single));
   526→  };
   527→
   528→  const parts: string[] = [url];
   529→  expandRep(chosen.el, chosen.base, parts);
   530→  for (const a of audioReps.slice(0, 2)) expandRep(a.el, a.base, parts);
   531→  return { parts: uniqKeepOrder(parts), height, playbackUrl: url };
   532→}

   534→async function collectParts(kind: 'dash' | 'hls' | 'mp4', url: string, clearKeys: Record<string, string> | null): Promise<CollectedParts> {
   535→  if (kind === 'mp4') return { parts: [url], height: null, playbackUrl: url };
   536→  if (kind === 'hls') return collectHlsParts(url);
   537→  // DASH: spec-correct MPD parser first, then probe-verify. Some shaka
   538→  // builds mis-substitute $Number$ on timelines — only trust a part list
   539→  // whose first segment actually fetches. If the parser fails, try the
   540→  // shaka walk (also probe-verified), then fall back to whichever parsed.
   541→  let textParts: CollectedParts | null = null;
   542→  try {
   543→    textParts = await collectDashText(url);
   544→  } catch {
   545→    textParts = null;
   546→  }
   547→  if (textParts && textParts.parts.length > 1 && (await probeCollected(textParts.parts))) {
   548→    return textParts;
   549→  }
   550→  try {
   551→    const shakaParts = await collectDashShaka(url, clearKeys);
   552→    if (shakaParts.parts.length > 1 && (await probeCollected(shakaParts.parts))) {
   553→      return shakaParts;
   554→    }
   555→  } catch {
   556→    /* fall through */
   557→  }
   558→  if (textParts && textParts.parts.length > 1) return textParts;
   559→  throw new Error('Could not read the lecture manifest (segments unreachable).');
   560→}

   573→function pickHeaders(res: Response): Headers {
   574→  const h = new Headers();
   575→  const ct = res.headers.get('content-type');
   576→  if (ct) h.set('Content-Type', ct);
   577→  h.set('Cache-Control', 'immutable');
   578→  return h;

   581→/**
   582→ * Start (or resume) a lecture download. Resolves once the download pipeline
   583→ * has been kicked off (progress continues in the background — observe via
   584→ * subscribeDownloads / listDownloads).
   585→ */
   586→export async function startDownload(spec: DownloadStartSpec): Promise<SbDownload> {
   587→  const now = new Date().toISOString();
   588→  const existing = await getDownload(spec.id);
   589→  // resumable = a healthy in-flight download — keep its progress; the part
   590→  // list is re-collected on resume (parts are not persisted), which yields
   591→  // byte-identical URLs for the already-downloaded parts (same stored sig).
   592→  const resumable = Boolean(
   593→    existing &&
   594→      (existing.parts.length > 0 || existing.partsTotal > 0) &&
   595→      (existing.status === 'paused' || existing.status === 'downloading' || existing.status === 'collecting'),
   596→  );
   597→  let rec: SbDownload;
   598→  if (!existing || existing.status === 'done') {
   599→    // brand-new download (or full re-download)
   600→    rec = {
   601→      ...spec,
   602→      folder: folderFromMediaUrl(spec.url) || spec.id.replace(/[^a-zA-Z0-9_-]/g, '_').slice(0, 40),
   603→      height: null,
   604→      parts: [],
   605→      partsTotal: 0,
   606→      partsDone: 0,
   607→      sizeBytes: 0,
   608→      status: 'collecting',
   609→      error: undefined,
   610→      createdAt: existing?.createdAt || now,
   611→      updatedAt: now,
   612→    };
   613→  } else if (resumable && existing) {
   614→    rec = {
   615→      ...existing,
   616→      ...spec,
   617→      url: existing.url,
   618→      folder: existing.folder,
   619→      parts: [],
   620→      partsTotal: existing.partsTotal || existing.parts.length,
   621→      partsDone: existing.partsDone || 0,
   622→      updatedAt: now,
   623→    };
   624→  } else if (existing) {
   625→    // error state — drop the stale part list and re-collect from the source
   626→    rec = {
   627→      ...existing,
   628→      ...spec,
   629→      folder: folderFromMediaUrl(spec.url) || existing.folder,
   630→      parts: [],
   631→      partsTotal: 0,
   632→      partsDone: 0,
   633→      sizeBytes: 0,
   634→      height: null,
   635→      status: 'collecting',
   636→      error: undefined,
   637→      updatedAt: now,
   638→    };
   639→  } else {
   640→    throw new Error('unreachable');
   641→  }
   642→
   643→  if (activeControllers.has(spec.id)) return rec; // already running
   644→  const ctrl = new AbortController();
   645→  activeControllers.set(spec.id, ctrl);
   646→
   647→  await putDownload(rec);
   648→  notifyChange();
   649→
   650→  // fire-and-forget the heavy lifting
   651→  void runDownload(rec, ctrl).finally(() => activeControllers.delete(spec.id));
   652→  return rec;
   653→}
   654→
   655→async function runDownload(rec: SbDownload, ctrl: AbortController): Promise<void> {
   656→  try {
   657→    // ---- 1. collect parts (re-collect on resume if we never finished) ----
   658→    if (rec.parts.length === 0) {
   659→      rec.status = 'collecting';
   660→      await putDownload(rec);
   661→      notifyChange();
   662→      const collected = await collectParts(rec.kind, rec.url, rec.clearKeys);
   663→      if (ctrl.signal.aborted) return;
   664→      rec.parts = collected.parts;
   665→      rec.partsTotal = collected.parts.length;
   666→      rec.height = collected.height;
   667→      rec.url = collected.playbackUrl;
   668→      // resume: keep progress (cached parts are re-verified cheaply below);
   669→      // fresh/error re-collect: start from zero
   670→      if (!rec.partsDone) {
   671→        rec.partsDone = 0;
   672→        rec.sizeBytes = 0;
   673→      }
   674→      await putDownload(rec);
   675→      notifyChange();
   676→    }
   677→    if (!rec.partsTotal) rec.partsTotal = rec.parts.length;
   678→
   679→    if (!('caches' in window)) throw new Error('Cache API unavailable in this browser');
   680→    const cache = await caches.open(videoCacheName(rec.folder));
   681→
   682→    // ---- 2. download pool ----
   683→    rec.status = 'downloading';
   684→    await putDownload(rec);
   685→    notifyChange();
   686→
   687→    let next = 0;
   688→    let done = rec.partsDone;
   689→    let size = rec.sizeBytes;
   690→    let failure: Error | null = null;
   691→    // fresh start: nothing can be cached yet — skip the per-part resume probe
   692→    const skipResumeMatch = rec.partsDone === 0;
   693→
   694→    const grab = async (): Promise<void> => {
   695→      for (;;) {
   696→        if (ctrl.signal.aborted || failure) return;
   697→        const i = next++;
   698→        if (i >= rec.parts.length) return;
   699→        const partUrl = rec.parts[i];
   700→        try {
   701→          const hit = skipResumeMatch ? null : await cache.match(partUrl); // resume: keep already-cached parts
   702→          if (hit) {
   703→            size += Number(hit.headers.get('content-length') || 0);
   704→          } else {
   705→            let stored = false;
   706→            for (let attempt = 0; attempt < 2 && !stored; attempt++) {
   707→              try {
   708→                const res = await fetch(partUrl, { signal: ctrl.signal });
   709→                if (!res.ok) throw new Error(`HTTP ${res.status}`);
   710→                const buf = await res.arrayBuffer();
   711→                size += buf.byteLength;
   712→                await cache.put(partUrl, new Response(buf, { headers: pickHeaders(res) }));
   713→                stored = true;
   714→              } catch (e) {
   715→                if (ctrl.signal.aborted) return;
   716→                if (attempt === 1) throw e;
   717→                await sleep(1500); // one soft retry on flaky network
   718→              }
   719→            }
   720→            if (!stored) throw new Error('Fetch failed twice');
   721→          }
   722→          done++;
   723→          if (done % 5 === 0 || done === rec.parts.length) {
   724→            rec.partsDone = done;
   725→            rec.sizeBytes = size;
   726→            await putDownload(rec);
   727→            notifyChange();
   728→          }
   729→        } catch (e) {
   730→          if (ctrl.signal.aborted) return;
   731→          failure = e instanceof Error ? e : new Error('Download failed');
   732→        }
   733→      }
   734→    };
   735→
   736→    await Promise.all(Array.from({ length: Math.min(CONCURRENCY, rec.parts.length) }, grab));
   737→
   738→    if (ctrl.signal.aborted) return;
   739→    if (failure) throw failure;
   740→
   741→    rec.partsDone = rec.parts.length;
   742→    rec.sizeBytes = size;
   743→    rec.partsTotal = rec.parts.length;
   744→    rec.status = 'done';
   745→    rec.error = undefined;
   746→    await putDownload(rec);
   747→    notifyChange();
   748→    await ensurePersistentStorage();
   749→  } catch (e) {
   750→    if (ctrl.signal.aborted) return; // paused — record keeps its state
   751→    rec.status = 'error';
   752→    rec.error = e instanceof Error ? e.message : 'Download failed';
   753→    await putDownload(rec);
   754→    notifyChange();
   755→  }
   756→}

   758→/** Pause an in-flight download (resume later with startDownload). */
   759→export async function pauseDownload(id: string): Promise<void> {
   760→  const ctrl = activeControllers.get(id);
   761→  if (ctrl) {
   762→    activeControllers.delete(id);
   763→    ctrl.abort();
   764→  }
   765→  const rec = await getDownload(id);
   766→  if (rec && (rec.status === 'downloading' || rec.status === 'collecting')) {
   767→    rec.status = 'paused';
   768→    await putDownload(rec);
   769→    notifyChange();
   770→  }
   771→}
   772→
   773→export async function isDownloading(id: string): Promise<boolean> {
   774→  return activeControllers.has(id);
   775→}

   777→/** Delete a lecture's offline copy (cache + record). */
   778→export async function deleteDownload(id: string): Promise<void> {
   779→  await pauseDownload(id);
   780→  const rec = await getDownload(id);
   781→  if (rec) {
   782→    try {
   783→      await caches.delete(videoCacheName(rec.folder));
   784→    } catch {
   785→      /* ignore */
   786→    }
   787→    await tx('readwrite', (s) => s.delete(rec.id) as unknown as IDBRequest<undefined>);
   788→    notifyChange();
   789→  }
   790→}

   792→/** Delete every offline lecture (used by "Clear all"). */
   793→export async function deleteAllDownloads(): Promise<void> {
   794→  const all = await listDownloads();
   795→  for (const rec of all) await deleteDownload(rec.id);
   796→}

   802→export function formatBytes(n: number): string {
   803→  if (!n || n < 0) return '—';
   804→  if (n < 1024) return `${n} B`;
   805→  const kb = n / 1024;
   806→  if (kb < 1024) return `${kb.toFixed(0)} KB`;
   807→  const mb = kb / 1024;
   808→  if (mb < 1024) return `${mb.toFixed(mb < 10 ? 1 : 0)} MB`;
   809→  return `${(mb / 1024).toFixed(2)} GB`;
   810→}
