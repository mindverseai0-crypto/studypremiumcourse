    72→  const [eventsLoading, setEventsLoading] = useState(false);
    73→
    74→  // ---------------- classes / chapters / content ----------------
    75→  const [subjects, setSubjects] = useState<BatchSubjectPublic[]>([]);
    76→  const [subjectsLoading, setSubjectsLoading] = useState(false);
    77→  const [currentSubject, setCurrentSubject] = useState<SubjectRef | null>(null);
    78→  const [chaptersTab, setChaptersTab] = useState<'UNITS' | 'STUDY_MATERIAL'>('UNITS');
    79→  const [chapters, setChapters] = useState<ChapterItemPublic[]>([]);

    85→  const [contentLoading, setContentLoading] = useState(false);
    86→
    87→  // ---------------- tests ----------------

    90→
    91→  // ---------------- resources tab ----------------

    93→  const [resourceTopics, setResourceTopics] = useState<ChapterItemPublic[]>([]);
    94→  const [resourcesLoading, setResourcesLoading] = useState(false);

    96→  // ---------------- modals ----------------

   100→
   101→  const [notesOpen, setNotesOpen] = useState(false);
   102→  const [notesLoading, setNotesLoading] = useState(false);
   103→  const [noteRows, setNoteRows] = useState<AttachmentRow[]>([]);
   104→  const [noteDppRows, setNoteDppRows] = useState<DppRow[]>([]);

   106→  const [dppOpen, setDppOpen] = useState(false);
   107→  const [dppMeta, setDppMeta] = useState<DppMeta | null>(null);
   108→
   109→  const [testFrame, setTestFrame] = useState<{ open: boolean; url: string; title: string }>({ open: false, url: '', title: 'Test Attempt' });
   110→  const [testPreparing, setTestPreparing] = useState<string | null>(null);
   111→
   112→  // ==================================================================
   113→  // NAV
   114→  // ==================================================================
   115→  const nav = useCallback((v: ViewId) => {
   116→    setView(v);
   117→    setSidebarOpen(false);
   118→    window.scrollTo(0, 0);

   121→  // ==================================================================
   122→  // CATALOG (home view)
   123→  // ==================================================================
   124→  const loadCatalog = useCallback(async (q: string, p: number, isNew: boolean) => {
   125→    try {
   126→      if (isNew) {
   127→        setSearching(true);
   128→        setBatches([]);
   129→      }
   130→      setCatalogError('');
   131→      const res = await fetchCatalog(q, p, 20);
   132→      setBatches((prev) => (isNew ? res.data : [...prev, ...res.data]));
   133→      setHasMore(res.hasMore);
   134→      setPage(p);
   135→    } catch {
   136→      setCatalogError('Failed to load batches.');
   137→    } finally {
   138→      setSearching(false);
   139→    }
   140→  }, []);
   141→
   142→  useEffect(() => {
   143→    loadCatalog('', 1, true);
   144→  }, [loadCatalog]);
   145→
   146→  const handleSearchInput = (value: string) => {
   147→    setSearchTerm(value);
   148→    debounceTerm.current = value;
   149→    if (searchTimeout.current) clearTimeout(searchTimeout.current);
   150→    searchTimeout.current = setTimeout(() => {
   151→      loadCatalog(debounceTerm.current, 1, true);
   152→    }, 500);
   153→  };
   154→
   155→  const loadNextPage = () => {
   156→    loadCatalog(searchTerm, page + 1, false);
   157→  };
   158→
   159→  /** Catalog card click — open the real-data batch detail sheet. */
   160→  const openCatalogInfo = async (b: BatchCatalogItem) => {
   161→    setBatchInfoOpen(true);
   162→    setBatchInfoLoading(true);
   163→    setBatchInfo(b);
   164→    const res = await api.batchDetail(b._id);
   165→    if (res.success && res.data) setBatchInfo(res.data);
   166→    setBatchInfoLoading(false);
   167→  };
   168→
   169→  const enrollInBatch = (batchId: string) => {
   170→    enroll(batchId);
   171→    setCurrentBatchId(batchId);
   172→    setBatchInfoOpen(false);
   173→    nav('study-view');
   174→  };
   175→
   176→  const studyBatch = (batchId: string) => {
   177→    setBatchInfoOpen(false);
   178→    setCurrentBatchId(batchId);
   179→    nav('study-view');
   180→  };
   181→
   182→  // ==================================================================
   183→  // STUDY DASHBOARD
   184→  // ==================================================================
   185→  const refreshEnrolledDetails = useCallback(async () => {
   186→    const results: BatchDetailPublic[] = [];
   187→    for (const id of enrolled) {
   188→      const cached = batchDetailCache.current.get(id);
   189→      if (cached) {
   190→        results.push(cached);
   191→        continue;
   192→      }
   193→      const res = await api.batchDetail(id);
   194→      if (res.success && res.data) {
   195→        batchDetailCache.current.set(id, res.data);
   196→        results.push(res.data);
   197→      }
   198→    }
   199→    setEnrolledDetails(results);
   200→    setCurrentBatchId((cur) => {
   201→      if (cur && results.some((r) => r._id === cur)) return cur;
   202→      return results.length > 0 ? results[0]._id : null;
   203→    });
   204→  }, [enrolled]);
   205→
   206→  useEffect(() => {
   207→    refreshEnrolledDetails();
   208→  }, [refreshEnrolledDetails]);
   209→
   210→  const loadEvents = useCallback(async (batchId: string) => {
   211→    setEventsLoading(true);
   212→    const res = await api.todaySchedule(batchId);
   213→    setEvents(res.success && res.data ? res.data : []);
   214→    setEventsLoading(false);
   215→  }, []);
   216→
   217→  useEffect(() => {
   218→    if (currentBatchId) loadEvents(currentBatchId);
   219→    else setEvents([]);
   220→  }, [currentBatchId, loadEvents]);
   221→
   222→  const currentBatch = enrolledDetails.find((b) => b._id === currentBatchId) || null;
   223→
   224→  // ==================================================================
   225→  // ALL CLASSES — SUBJECTS & RESOURCES
   226→  // ==================================================================
   227→  const loadSubjects = useCallback(async (batchId: string) => {
   228→    setSubjectsLoading(true);
   229→    // Batch detail carries the subject mapping (curriculum engine).
   230→    const res = await api.batchDetail(batchId);
   231→    if (res.success && res.data) {
   232→      batchDetailCache.current.set(batchId, res.data);
   233→      setSubjects(res.data.subjects || []);
   234→    } else {
   235→      setSubjects([]);
   236→    }
   237→    setSubjectsLoading(false);
   238→  }, []);
   239→
   240→  const loadResources = useCallback(async (batchId: string) => {
   241→    setResourcesLoading(true);
   242→    const res = await api.resourceTopics(batchId, '');
   243→    const topics = res.success && res.data ? (res.data as unknown as { data?: ChapterItemPublic[] }).data || [] : [];
   244→    setResourceTopics(topics);
   245→    setResourcesLoading(false);
   246→  }, []);
   247→
   248→  const openAllClasses = () => {
   249→    setResourcesTab('subjects');
   250→    nav('all-classes-view');
   251→    if (currentBatchId) loadSubjects(currentBatchId);
   252→  };
   253→
   254→  const switchClassesTab = (tab: 'subjects' | 'resources') => {
   255→    setResourcesTab(tab);
   256→    if (tab === 'subjects' && currentBatchId) loadSubjects(currentBatchId);
   257→    if (tab === 'resources' && currentBatchId) loadResources(currentBatchId);
   258→  };
   259→
   260→  // ==================================================================
   261→  // CHAPTERS
   262→  // ==================================================================
   263→  const loadChaptersList = useCallback(async (subjectId: string, tagType: 'UNITS' | 'STUDY_MATERIAL') => {
   264→    if (!currentBatchId) return;
   265→    setChaptersLoading(true);
   266→    const res = await api.subjectTopics(currentBatchId, subjectId, tagType);
   267→    const list = res.success && res.data ? (res.data as unknown as { data?: ChapterItemPublic[] }).data || [] : [];
   268→    setChapters(list);
   269→    setChaptersLoading(false);
   270→  }, [currentBatchId]);
   271→
   272→  const openSubject = (sub: BatchSubjectPublic) => {
   273→    setCurrentSubject({ id: sub._id, internalId: sub.subjectId, name: sub.subject });
   274→    setChaptersTab('UNITS');
   275→    nav('chapters-view');
   276→    loadChaptersList(sub._id, 'UNITS');
   277→  };
   278→
   279→  const switchChaptersTab = (tagType: 'UNITS' | 'STUDY_MATERIAL') => {
   280→    setChaptersTab(tagType);
   281→    if (currentSubject) loadChaptersList(currentSubject.id, tagType);
   282→  };
   283→
   284→  // ==================================================================
   285→  // CONTENT
   286→  // ==================================================================
   287→  const loadContent = useCallback(async (subjectId: string, tagId: string, filter: string) => {
   288→    if (!currentBatchId) return;
   289→    setContentLoading(true);
   290→    if (filter === 'DPP_TEST') {
   291→      const res = await api.dppList(currentBatchId, subjectId || '', tagId);
   292→      const list = res.success && res.data ? (res.data as unknown as { data?: TestItemPublic[] }).data || [] : [];
   293→      setDppItems(list);
   294→      setContentNodes([]);
   295→    } else {
   296→      const res = await api.scheduleContent(currentBatchId, subjectId || '', tagId, filter);
   297→      const nodes = res.success && res.data ? (res.data as ContentNodePublic[]) : [];
   298→      setContentNodes(nodes);
   299→      setDppItems([]);
   300→    }
   301→    setContentLoading(false);
   302→  }, [currentBatchId]);
   303→
   304→  const openTopic = (t: ChapterItemPublic) => {
   305→    setCurrentTopic({ id: t._id, name: t.name });
   306→    setContentTab('ALL');
   307→    nav('content-view');
   308→    if (currentSubject) loadContent(currentSubject.id, t._id, 'ALL');
   309→  };
   310→
   311→  const openResourceTopic = (t: ChapterItemPublic) => {
   312→    setCurrentSubject({ id: t.typeId, internalId: t.typeId, name: 'Resources' });
   313→    setCurrentTopic({ id: t._id, name: t.name });
   314→    setContentTab('ALL');
   315→    nav('content-view');
   316→    loadContent(t.typeId, t._id, 'ALL');
   317→  };
   318→
   319→  const filterContentTab = (filter: string) => {
   320→    setContentTab(filter);
   321→    if (currentSubject && currentTopic) loadContent(currentSubject.id, currentTopic.id, filter);
   322→  };
   323→
   324→  // ==================================================================
   325→  // TESTS
   326→  // ==================================================================
   327→  const openAllTests = () => {
   328→    nav('all-tests-view');
   329→    if (!currentBatchId) return;
   330→    setTestsLoading(true);
   331→    api.testList(currentBatchId).then((res) => {
   332→      setTests(res.success && res.data ? res.data : []);
   333→      setTestsLoading(false);
   334→    }).catch(() => {
   335→      setTests([]);
   336→      setTestsLoading(false);
   337→    });
   338→  };
   339→
   340→  const launchTestGateway = async (testId: string) => {
   341→    setTestPreparing(testId);
   342→    const res = await api.testMap(testId);
   343→    let actionType = 'Start';
   344→    let testMapId = '';
   345→    if (res.success && res.data && res.data.length > 0) {
   346→      actionType = res.data[0].tag2 || 'Start';
   347→      testMapId = res.data[0].testStudentMapping?._id || '';
   348→    }
   349→    const url = `/quiz?testId=${testId}&batchId=${currentBatchId || ''}&mode=TEST&type=${actionType}&mapId=${testMapId}`;
   350→    setTestPreparing(null);
   351→    setTestFrame({ open: true, url, title: 'Test Attempt' });

   357→  const openBatchInfo = async () => {
   358→    if (!currentBatchId) return;
   359→    setBatchInfoOpen(true);
   360→    setBatchInfoLoading(true);
   361→    const res = await api.batchDetail(currentBatchId);
   362→    setBatchInfo(res.success && res.data ? res.data : null);
   363→    setBatchInfoLoading(false);

   365→
   366→  const openNotesModal = async (scheduleId: string) => {
   367→    if (!currentBatchId) return;
   368→    setNotesOpen(true);
   369→    setNotesLoading(true);
   370→    setNoteRows([]);
   371→    setNoteDppRows([]);
   372→    const res = await api.scheduleDetail(currentBatchId, scheduleId);
   373→    const item = res.data as {
   374→      topic?: string;
   375→      name?: string;
   376→      attachmentIds?: Array<{ baseUrl?: string; key?: string }>;
   377→      homeworkIds?: Array<{ topic?: string; attachmentIds?: Array<{ baseUrl?: string; key?: string }> }>;
   378→      exerciseIds?: Array<Record<string, unknown>>;
   379→    } | undefined;
   380→
   381→    const rows: AttachmentRow[] = [];
   382→    if (item) {
   383→      (item.homeworkIds || []).forEach((hw) => {
   384→        (hw.attachmentIds || []).forEach((att) => {
   385→          rows.push({ title: hw.topic || 'Class Note', url: imageUrl(att) });
   386→        });
   387→      });
   388→      (item.attachmentIds || []).forEach((att) => {
   389→        rows.push({ title: item.topic || item.name || 'Document', url: imageUrl(att) });
   390→      });
   391→    }
   392→    setNoteRows(rows);
   393→
   394→    // DPP matches from exercises
   395→    const testRes = await api.testList(currentBatchId);
   396→    const testList = testRes.success && testRes.data ? testRes.data : [];
   397→    const dppRows: DppRow[] = [];
   398→    (item?.exerciseIds || []).forEach((ex) => {
   399→      const exDppId = String((ex as { dppId?: string }).dppId || (ex as { _id?: string })._id || '');
   400→      const match = testList.find((t) => t.dppQuizDetails && t.dppQuizDetails.contentId === exDppId);
   401→      if (match && match.dppQuizDetails) {
   402→        const status = String((match.dppQuizDetails.testStudentMapping as { testActivityStatus?: string })?.testActivityStatus || '');
   403→        dppRows.push({
   404→          testId: match.dppQuizDetails.test._id,
   405→          scheduleId: match.dppQuizDetails.scheduleId,
   406→          name: (ex as { title?: string }).title || match.dppQuizDetails.test.name,
   407→          isAttempted: status === 'Finished' || status === 'COMPLETED',
   408→          actionType: status === 'Finished' ? 'Reattempt' : status === 'Started' ? 'Resume' : 'Start',
   409→          testMapId: '',
   410→        });
   411→      }
   412→    });
   413→    setNoteDppRows(dppRows);
   414→    setNotesLoading(false);
   415→  };
   416→
   417→  const openDirectPdf = async (scheduleId: string) => {
   418→    if (!currentBatchId) return;
   419→    const res = await api.scheduleDetail(currentBatchId, scheduleId);
   420→    const item = res.data as {
   421→      topic?: string;
   422→      name?: string;
   423→      attachmentIds?: Array<{ baseUrl?: string; key?: string }>;
   424→      homeworkIds?: Array<{ topic?: string; attachmentIds?: Array<{ baseUrl?: string; key?: string }> }>;
   425→      exerciseIds?: Array<{ title?: string; attachmentIds?: Array<{ baseUrl?: string; key?: string }> }>;
   426→    } | undefined;
   427→    const rows: AttachmentRow[] = [];
   428→    if (item) {
   429→      (item.attachmentIds || []).forEach((att) => rows.push({ title: item.topic || item.name || 'Document', url: imageUrl(att) }));
   430→      (item.homeworkIds || []).forEach((hw) => (hw.attachmentIds || []).forEach((att) => rows.push({ title: hw.topic || 'Class Note', url: imageUrl(att) })));
   431→      (item.exerciseIds || []).forEach((ex) => (ex.attachmentIds || []).forEach((att) => rows.push({ title: ex.title || 'DPP', url: imageUrl(att) })));
   432→    }
   433→    const deduped = rows.filter((r, i, arr) => arr.findIndex((x) => x.url === r.url) === i);
   434→    if (deduped.length === 0) return;
   435→    if (deduped.length === 1) {
   436→      window.open(deduped[0].url, '_blank');
   437→      addXP(5, 'Viewed Document');
   438→    } else {
   439→      setNoteRows(deduped);
   440→      setNoteDppRows([]);
   441→      setNotesOpen(true);
   442→    }
   443→  };
   444→
   445→  const openDppModal = (meta: DppMeta) => {
   446→    setDppMeta(meta);
   447→    setDppOpen(true);
   448→  };
   449→
   450→  const launchDpp = (mode: 'BATCH_QUIZ' | 'DPP_QUIZ_MODE') => {
   451→    if (!dppMeta || !currentBatchId) return;
   452→    const subjectId = currentSubject?.internalId || currentSubject?.id || '';
   453→    const url = `/quiz?testId=${dppMeta.testId}&batchId=${currentBatchId}&mode=${mode}&scheduleId=${dppMeta.scheduleId}&subjectId=${subjectId}&type=${dppMeta.actionType}`;
   454→    addXP(10, 'Attempted Test/DPP');
   455→    setDppOpen(false);
   456→    setTestFrame({ open: true, url, title: 'DPP Attempt' });
   457→  };
   458→
   459→  const handleUnenroll = (batchId: string) => {
   460→    if (!window.confirm('Are you sure you want to unenroll from this batch?')) return;
   461→    unenroll(batchId);
   462→    setBatchInfoOpen(false);
   463→    if (currentBatchId === batchId) {
   464→      const remaining = enrolled.filter((id) => id !== batchId);
   465→      if (remaining.length > 0) setCurrentBatchId(remaining[0]);
   466→      else {
   467→        setCurrentBatchId(null);
   468→        nav('home-view');
   469→      }
   470→    }
   471→  };

   475→  // ==================================================================
   476→  const renderLectureCard = (node: ContentNodePublic) => {
   477→    const d = node.data;
   478→    const thumb = d.videoDetails?.image || DEFAULT_THUMB;
   479→    const title = d.topic || d.name || 'Lecture';
   480→    const date = d.date ? new Date(d.date).toLocaleDateString('en-GB', { day: 'numeric', month: 'short', year: 'numeric' }) : '';
   481→    const vId = d.videoDetails?._id || d._id;
   482→    const vUrl = d.videoDetails?.videoUrl || '';
   483→    const isWatched = watched.includes(vId);
   484→    const bName = currentBatch?.name || 'PW Batch';
   485→    const playerUrl = `/player?videoId=${vId}&schId=${node._id}&batchId=${currentBatchId || ''}&subjectId=${currentSubject?.id || ''}&title=${encodeURIComponent(title)}&batchName=${encodeURIComponent(bName)}&vUrl=${encodeURIComponent(vUrl)}`;
   486→    const hasAttachments = (d.homeworkIds && d.homeworkIds.length > 0) || (d.exerciseIds && d.exerciseIds.length > 0);
   487→
   488→    return (
   489→      <div className="content-card" key={node._id} style={{ flexDirection: 'column', alignItems: 'stretch' }}>
   490→        <div className="d-flex gap-3 mb-3">
   491→          <img src={thumb} className="lecture-thumb" alt={title} />
   492→          <div className="flex-grow-1">
   493→            <span style={{ background: '#eff6ff', color: '#2563eb', fontSize: '0.7rem', fontWeight: 800, padding: '2px 6px', borderRadius: 4, display: 'inline-block', marginBottom: 5 }}>
   494→              {node.type === 'DPP_VIDEOS' ? 'DPP VIDEO' : 'VIDEO'}
   495→            </span>
   496→            <div style={{ fontWeight: 700, fontSize: '1rem', color: 'var(--sb-text-dark)' }}>{title}</div>
   497→            <div style={{ color: 'var(--sb-text-muted)', fontSize: '0.8rem', marginTop: 5 }}>
   498→              <i className="fas fa-clock"></i> {date} &bull; {d.videoDetails?.duration || '1h'}
   499→            </div>
   500→          </div>
   501→        </div>
   502→        <div className="d-flex gap-2">
   503→          <button
   504→            className={`btn-check-done ${isWatched ? 'active' : ''}`}
   505→            title="Mark as Watched"
   506→            onClick={() => {
   507→              const added = toggleWatched(vId);
   508→              if (added) addXP(10, 'Watched Video');
   509→            }}
   510→          >
