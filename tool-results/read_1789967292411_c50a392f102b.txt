     1→// ============================================================================
     2→// Study Bison — offline lecture library (PWA downloads).
     3→//
     4→// YouTube-style "Available offline": every lecture can be downloaded once and
     5→// then watched with the network fully off. Media bytes live in the Cache API
     6→// (one cache per lecture folder: `sb-video-<folder>`) and are served by the
     7→// service worker (public/sw.js) even when the device is offline.
     8→//
     9→// Flow:
    10→//   1. The player resolves the playable relay URL (/api/pw/<folder>/s/<sig>/…)
    11→//      — all segment URIs are same-origin and signed statelessly in the PATH,
    12→//      so they stay byte-identical between download time and offline playback.
    13→//   2. collectParts() walks the manifest (shaka headless for DASH, a text
    14→//      parser for HLS) and returns the full part list for ONE good quality.
    15→//   3. startDownload() fetches every part (concurrency 4, resume-safe) and
    16→//      cache.put()s it into the lecture cache. Progress is persisted in
    17→//      IndexedDB so downloads survive reloads and can be paused/resumed.
    18→//   4. Offline the player finds the stored record, re-uses the stored
    19→//      manifest URL + ClearKeys and the SW serves every byte from the cache.
    20→//
    21→// There is no artificial cap on the number of downloads — the only limit is
    22→// the device's own storage quota (we also request persistent storage so the
    23→// browser's janitor does not evict lectures).
    24→// ============================================================================
    25→

    78→/** Extract the media folder id from a relay/CloudFront media URL. */
    79→export function folderFromMediaUrl(url: string): string {
    80→  try {
    81→    const u = new URL(url, typeof location !== 'undefined' ? location.href : 'https://x.invalid');
    82→    if (u.origin === location.origin) {
    83→      // /api/pw/<folder>/…  |  /api/pw/h/<host>/<folder>/…  |  /api/pw/<folder>/s/<sig>/…
    84→      const m = u.pathname.match(/^\/api\/pw\/(?:h\/[^/]+\/)?([^/]+)\//);
    85→      return m ? m[1] : '';
    86→    }
    87→    if (/\.cloudfront\.net$/i.test(u.hostname)) {
    88→      const m = u.pathname.match(/^\/([^/]+)\//);
    89→      return m ? m[1] : '';
    90→    }
    91→  } catch {
    92→    /* ignore */
    93→  }
    94→  return '';
    95→}

   101→let dbPromise: Promise<IDBDatabase> | null = null;
   102→
   103→function openDb(): Promise<IDBDatabase> {
   104→  if (dbPromise) return dbPromise;
   105→  dbPromise = new Promise((resolve, reject) => {
   106→    try {
   107→      const req = indexedDB.open(DB_NAME, DB_VERSION);
   108→      req.onupgradeneeded = () => {
   109→        const db = req.result;
   110→        if (!db.objectStoreNames.contains(STORE)) {
   111→          db.createObjectStore(STORE, { keyPath: 'id' });
   112→        }
   113→      };
   114→      req.onsuccess = () => resolve(req.result);
   115→      req.onerror = () => reject(req.error || new Error('IndexedDB unavailable'));
   116→    } catch (e) {
   117→      reject(e instanceof Error ? e : new Error('IndexedDB unavailable'));
   118→    }
   119→  });
   120→  return dbPromise;
   121→}
   122→
   123→async function tx<T>(mode: IDBTransactionMode, fn: (store: IDBObjectStore) => IDBRequest<T>): Promise<T> {
   124→  const db = await openDb();
   125→  return new Promise<T>((resolve, reject) => {
   126→    const t = db.transaction(STORE, mode);
   127→    const req = fn(t.objectStore(STORE));
   128→    req.onsuccess = () => resolve(req.result);
   129→    req.onerror = () => reject(req.error || new Error('IndexedDB error'));
   130→  });
   131→}

   142→export function subscribeDownloads(cb: () => void): () => void {
   143→  window.addEventListener(CHANGE_EVENT, cb);
   144→  window.addEventListener('online', cb);
   145→  window.addEventListener('offline', cb);
   146→  return () => {
   147→    window.removeEventListener(CHANGE_EVENT, cb);
   148→    window.removeEventListener('online', cb);
   149→    window.removeEventListener('offline', cb);

   180→/** Find the download record matching a lecture (by id candidates or vUrl). */
   181→export async function findDownloadFor(ids: {
   182→  videoId?: string;
   183→  realVideoId?: string;
   184→  vUrl?: string;
   185→}): Promise<SbDownload | null> {
   186→  const cands = new Set<string>();
   187→  if (ids.videoId) cands.add(ids.videoId);
   188→  if (ids.realVideoId) cands.add(ids.realVideoId);
   189→  if (ids.vUrl) {
   190→    cands.add(`url:${ids.vUrl.split('?')[0]}`);
   191→    cands.add(ids.vUrl);
   192→  }
   193→  if (cands.size === 0) return null;
   194→  const all = await listDownloads();
   195→  for (const rec of all) {
   196→    if (cands.has(rec.id) || (ids.vUrl && rec.vUrl && rec.vUrl === ids.vUrl)) return rec;
   197→  }
   198→  return null;
   199→}

   235→function uniqKeepOrder(items: string[]): string[] {
   236→  const seen = new Set<string>();
   237→  const out: string[] = [];
   238→  for (const u of items) {
   239→    if (!u || seen.has(u)) continue;
   240→    seen.add(u);
   241→    out.push(u);
   242→  }
   243→  return out;
   244→}

   246→async function fetchText(url: string): Promise<string> {
   247→  const res = await fetch(url);
   248→  if (!res.ok) throw new Error(`HTTP ${res.status} for playlist`);
   249→  return res.text();
   250→}
   251→
   252→/**
   253→ * HLS — relay playlists are fully rewritten (absolute same-origin URIs), so a
   254→ * plain text walk is exact. For a master playlist we pick ONE rendition
   255→ * (≤720p, highest bitrate) and download only that media playlist + its
   256→ * segments — the offline player opens the variant playlist directly.
   257→ */
   258→async function collectHlsParts(url: string, depth = 0): Promise<CollectedParts> {
   259→  // absolutize (input may be a same-origin relative path like /e2ehls/main.m3u8)
   260→  const absUrl = new URL(url, location.href).href;
   261→  const text = await fetchText(absUrl);
   262→  const lines = text.split('\n').map((l) => l.trim());
   263→
   264→  // master playlist? → choose one variant
   265→  const variants: Array<{ url: string; bandwidth: number; height: number }> = [];
   266→  for (let i = 0; i < lines.length; i++) {
   267→    if (!/^#EXT-X-STREAM-INF/i.test(lines[i])) continue;
   268→    const attrs = lines[i].slice(lines[i].indexOf(':') + 1);
   269→    const res = /RESOLUTION=(\d+)x(\d+)/i.exec(attrs);
   270→    const bw = /BANDWIDTH=(\d+)/i.exec(attrs);
   271→    let j = i + 1;
   272→    while (j < lines.length && (!lines[j] || lines[j].startsWith('#'))) j++;
   273→    if (j < lines.length) {
   274→      variants.push({
   275→        url: new URL(lines[j], absUrl).href,
   276→        bandwidth: bw ? Number(bw[1]) || 0 : 0,
   277→        height: res ? Number(res[2]) || 0 : 0,
   278→      });
   279→    }
   280→  }
   281→
   282→  let playlistUrl = absUrl;
   283→  let height: number | null = null;
   284→  if (variants.length > 0) {
   285→    const capped = variants.filter((v) => v.height > 0 && v.height <= 720);
   286→    const best =
   287→      capped.sort((a, b) => b.bandwidth - a.bandwidth)[0] ||
   288→      variants.slice().sort((a, b) => a.bandwidth - b.bandwidth)[0];
   289→    playlistUrl = best.url;
   290→    height = best.height || null;
   291→  }
   292→
   293→  const mediaText = playlistUrl === absUrl ? text : await fetchText(playlistUrl);
   294→  const mediaLines = mediaText.split('\n').map((l) => l.trim());
   295→
   296→  // defensive: a variant that is itself still a master (depth guard = 2)
   297→  if (depth < 2 && mediaLines.some((l) => /^#EXT-X-STREAM-INF/i.test(l))) {
   298→    return collectHlsParts(playlistUrl, depth + 1);
   299→  }
   300→
   301→  const parts: string[] = [playlistUrl];
   302→  for (const l of mediaLines) {
   303→    if (!l) continue;
   304→    if (l.startsWith('#')) {
   305→      if (/^#EXT-X-(KEY|MAP|MEDIA|SESSION-KEY)/i.test(l)) {
   306→        const m = /URI="([^"]+)"/i.exec(l);
   307→        if (m) parts.push(new URL(m[1], playlistUrl).href);
   308→      }
   309→      continue;
   310→    }
   311→    parts.push(new URL(l, playlistUrl).href);
   312→  }
   313→  return { parts: uniqKeepOrder(parts), height, playbackUrl: playlistUrl };
   314→}

   316→/** Cheap ranged probe — verifies a URL is actually fetchable (200/206). */
   317→async function probePart(u: string, timeoutMs = 8000): Promise<boolean> {
   318→  try {
   319→    const ctrl = new AbortController();
   320→    const t = setTimeout(() => ctrl.abort(), timeoutMs);
   321→    const res = await fetch(u, { headers: { Range: 'bytes=0-1' }, signal: ctrl.signal });
   322→    clearTimeout(t);
   323→    return res.status < 400;
   324→  } catch {
   325→    return false;
   326→  }
   327→}

   329→/** Verify the collected list: probe the manifest + the first media segment. */
   330→async function probeCollected(parts: string[]): Promise<boolean> {
   331→  if (parts.length === 0) return false;
   332→  if (!(await probePart(parts[0]))) return false;
   333→  // find the first non-init part (the first real media segment)
   334→  const seg = parts.slice(1).find((u) => !/init\.(mp4|m4s)$/i.test(u.split('?')[0]));
   335→  if (!seg) return true; // playlist-only download
   336→  return probePart(seg);
   337→}

   339→/**
   340→ * DASH via shaka headless (no <video>): parse the MPD and walk the chosen
   341→ * variant's SegmentReferences. NOTE: some shaka versions substitute $Number$
   342→ * with time values when a SegmentTimeline is present — every collected list
   343→ * is therefore probe-verified by the caller before use.
   344→ */
   345→async function collectDashShaka(url: string, clearKeys: Record<string, string> | null): Promise<CollectedParts> {
   346→  const shaka = (await import('shaka-player')).default;
   347→  shaka.polyfill.installAll();
   348→  const player = new shaka.Player();
   349→  player.configure({
   350→    manifest: {
   351→      dash: { ignoreDrmInfo: true }, // we store encrypted bytes; keys only matter at playback
   352→      retryParameters: { timeout: 20000, maxAttempts: 3 },
   353→    },
   354→  });
   355→  try {
   356→    await player.load(url, 0, 'application/dash+xml');
   357→    const manifest = player.getManifest();
   358→    if (!manifest) throw new Error('Manifest did not parse');
   359→    const variants = manifest.variants.filter((v) => v.video);
   360→    if (variants.length === 0) throw new Error('No video variants in manifest');
   361→
   362→    // pick ≤720p with the highest bitrate (fallback: the smallest available)
   363→    const capped = variants.filter((v) => (v.video?.height || 0) > 0 && (v.video?.height || 0) <= 720);
   364→    const chosen =
   365→      capped.sort((a, b) => (b.bandwidth || 0) - (a.bandwidth || 0))[0] ||
   366→      variants.slice().sort((a, b) => (a.video?.height || 4320) - (b.video?.height || 4320))[0] ||
   367→      variants[0];
   368→
   369→    const height = chosen.video?.height || null;
   370→    const parts: string[] = [];
   371→    const seenStreams = new Set<number>();
   372→    const streams = [chosen.video, chosen.audio].filter(Boolean) as NonNullable<typeof chosen.video | typeof chosen.audio>[];
   373→    for (const stream of streams) {
   374→      if (seenStreams.has(stream.id)) continue;
   375→      seenStreams.add(stream.id);
   376→      try {
   377→        await stream.createSegmentIndex();
   378→      } catch {
   379→        /* already created */
   380→      }
   381→      const segIndex = stream.segmentIndex;
   382→      if (!segIndex) continue;
   383→      for (const ref of segIndex) {
   384→        if (!ref) continue;
   385→        try {
   386→          const init = ref.initSegmentReference;
   387→          if (init) {
   388→            for (const u of init.getUris()) parts.push(u);
   389→          }
   390→          for (const u of ref.getUris()) parts.push(u);
   391→        } catch {
   392→          /* skip broken reference */
   393→        }
   394→      }
   395→    }
   396→    if (parts.length === 0) throw new Error('No segments found in manifest');
   397→    return { parts: uniqKeepOrder(parts), height, playbackUrl: url };
   398→  } finally {
   399→    try {
   400→      await player.destroy();
   401→    } catch {
   402→      /* ignore */
   403→    }
   404→  }
   405→}

   407→/**
   408→ * MPD parser (primary DASH path) — handles SegmentTemplate with $Number$
   409→ * (+ SegmentTimeline or @duration) and $Time$, SegmentList and plain BaseURL
   410→ * representations, per the DASH spec ($Number$ = segment sequence number).
   411→ */
   412→async function collectDashText(url: string): Promise<CollectedParts> {
   413→  const xml = new DOMParser().parseFromString(await fetchText(url), 'application/xml');
   414→  const q = (el: Element | Document, name: string): Element | null => {
   415→    const list = (el as Element).getElementsByTagName(name);
   416→    return list.length ? list[0] : null;
   417→  };
   418→  const txt = (el: Element, name: string): string => {
   419→    const n = q(el, name);
   420→    return n && n.textContent ? n.textContent.trim() : '';
   421→  };
   422→  const num = (el: Element, name: string): number => Number(el.getAttribute(name) || 0);
   423→  const attr = (el: Element, name: string): string => el.getAttribute(name) || '';
   424→
   425→  const resolve = (base: string, uri: string): string => new URL(uri, new URL(base, location.href)).href;
   426→
   427→  const mpdBase = txt(xml, 'BaseURL') ? resolve(url, txt(xml, 'BaseURL')) : url;
   428→  const isoDuration = attr(xml.getElementsByTagName('MPD')[0] || xml.documentElement, 'mediaPresentationDuration');
   429→  const totalSec = (() => {
   430→    const m = /^P(?:(\d+(?:\.\d+)?)Y)?(?:(\d+(?:\.\d+)?)M)?(?:(\d+(?:\.\d+)?)D)?(?:T(?:(\d+(?:\.\d+)?)H)?(?:(\d+(?:\.\d+)?)M)?(?:(\d+(?:\.\d+)?)S)?)?$/.exec(isoDuration);
   431→    if (!m) return 0;
   432→    return (Number(m[4]) || 0) * 3600 + (Number(m[5]) || 0) * 60 + (Number(m[6]) || 0);
   433→  })();
   434→
   435→  const videoReps: Array<{ el: Element; base: string; height: number; bandwidth: number }> = [];
   436→  const audioReps: Array<{ el: Element; base: string }> = [];
   437→  for (const aset of Array.from(xml.getElementsByTagName('AdaptationSet'))) {
   438→    const asBase = txt(aset, 'BaseURL') ? resolve(mpdBase, txt(aset, 'BaseURL')) : mpdBase;
   439→    const contentType = (attr(aset, 'contentType') || attr(aset, 'mimeType')).toLowerCase();
   440→    const isAudio = /audio/.test(contentType);
   441→    for (const rep of Array.from(aset.getElementsByTagName('Representation'))) {
   442→      const repBase = txt(rep, 'BaseURL') ? resolve(asBase, txt(rep, 'BaseURL')) : asBase;
   443→      if (isAudio || /audio/.test(attr(rep, 'mimeType').toLowerCase())) {
   444→        audioReps.push({ el: rep, base: repBase });
   445→      } else {
   446→        videoReps.push({ el: rep, base: repBase, height: num(rep, 'height'), bandwidth: num(rep, 'bandwidth') });
   447→      }
   448→    }
   449→  }
   450→  if (videoReps.length === 0) throw new Error('No representations found');
   451→
   452→  const capped = videoReps.filter((r) => r.height > 0 && r.height <= 720);
   453→  const chosen =
   454→    capped.sort((a, b) => b.bandwidth - a.bandwidth)[0] ||
   455→    videoReps.slice().sort((a, b) => a.height - b.height)[0];
   456→  const height = chosen.height || null;
   457→
   458→  const expandRep = (rep: Element, base: string, out: string[]) => {
   459→    const repBase = txt(rep, 'BaseURL') ? resolve(base, txt(rep, 'BaseURL')) : base;
   460→    const aset = rep.parentElement as Element | null;
   461→    const tmpl = (q(rep, 'SegmentTemplate') as Element | null) || (aset ? q(aset, 'SegmentTemplate') : null);
   462→    if (tmpl) {
   463→      const mediaT = attr(tmpl, 'media');
   464→      const initT = attr(tmpl, 'initialization');
   465→      const timescale = num(tmpl, 'timescale') || 1;
   466→      const segDur = num(tmpl, 'duration') / timescale;
   467→      const startNumber = num(tmpl, 'startNumber') || (/\$Number%0(\d+)d\$/.test(mediaT) ? 1 : 0);
   468→      const repId = attr(rep, 'id');
   469→      const bandwidth = attr(rep, 'bandwidth');
   470→      const fill = (t: string): string =>
   471→        t
   472→          .replace(/\$RepresentationID\$/g, repId)
   473→          .replace(/\$Bandwidth\$/g, bandwidth)
   474→          .replace(/\$\$/g, '$');
   475→      if (initT) out.push(resolve(repBase, fill(initT)));
   476→      const timeline = q(tmpl, 'SegmentTimeline');
   477→      if (timeline) {
   478→        let t = 0;
   479→        let segIdx = 0;
   480→        for (const s of Array.from(timeline.getElementsByTagName('S'))) {
   481→          const st = s.getAttribute('t');
   482→          if (st !== null) t = Number(st);
   483→          const d = Number(s.getAttribute('d') || 0);
   484→          const r = Number(s.getAttribute('r') || 0);
   485→          for (let i = 0; i <= Math.max(0, r); i++) {
   486→            // $Number$ = sequence number (startNumber + index), $Time$ = media time
   487→            const filled = fill(mediaT)
   488→              .replace(/\$Number(?:%0(\d+)d)?\$/g, (_m, pad?: string) => String(startNumber + segIdx).padStart(pad ? Number(pad) : 0, '0'))
   489→              .replace(/\$Time\$/g, String(t));
   490→            out.push(resolve(repBase, filled));
   491→            segIdx++;
   492→            t += d;
   493→          }
   494→        }
   495→      } else if (segDur > 0 && totalSec > 0) {
   496→        const count = Math.ceil(totalSec / segDur);
   497→        for (let i = 0; i < count; i++) {
   498→          out.push(resolve(repBase, fill(mediaT).replace(/\$Number(?:%0(\d+)d)?\$/g, (_m, pad?: string) => String(startNumber + i).padStart(pad ? Number(pad) : 0, '0'))));
   499→        }
   500→      }
   501→      return;
   502→    }
   503→    const list = q(rep, 'SegmentList') || (aset ? q(aset, 'SegmentList') : null);
   504→    if (list) {
   505→      const initEl = q(list, 'Initialization');
   506→      if (initEl) {
   507→        const src = attr(initEl, 'sourceURL');
   508→        if (src) out.push(resolve(repBase, src));
   509→      }
   510→      for (const su of Array.from(list.getElementsByTagName('SegmentURL'))) {
   511→        const media = attr(su, 'media');
   512→        if (media) out.push(resolve(repBase, media));
   513→      }
   514→      return;
   515→    }
   516→    // plain <BaseURL> representation (single self-contained file)
   517→    const single = txt(rep, 'BaseURL');
   518→    if (single) out.push(resolve(base, single));
   519→  };
   520→
   521→  const parts: string[] = [url];
   522→  expandRep(chosen.el, chosen.base, parts);
   523→  for (const a of audioReps.slice(0, 2)) expandRep(a.el, a.base, parts);
   524→  return { parts: uniqKeepOrder(parts), height, playbackUrl: url };
   525→}

   527→async function collectParts(kind: 'dash' | 'hls' | 'mp4', url: string, clearKeys: Record<string, string> | null): Promise<CollectedParts> {
   528→  if (kind === 'mp4') return { parts: [url], height: null, playbackUrl: url };
   529→  if (kind === 'hls') return collectHlsParts(url);
   530→  // DASH: spec-correct MPD parser first, then probe-verify. Some shaka
   531→  // builds mis-substitute $Number$ on timelines — only trust a part list
   532→  // whose first segment actually fetches. If the parser fails, try the
   533→  // shaka walk (also probe-verified), then fall back to whichever parsed.
   534→  let textParts: CollectedParts | null = null;
   535→  try {
   536→    textParts = await collectDashText(url);
   537→  } catch {
   538→    textParts = null;
   539→  }
   540→  if (textParts && textParts.parts.length > 1 && (await probeCollected(textParts.parts))) {
   541→    return textParts;
   542→  }
   543→  try {
   544→    const shakaParts = await collectDashShaka(url, clearKeys);
   545→    if (shakaParts.parts.length > 1 && (await probeCollected(shakaParts.parts))) {
   546→      return shakaParts;
   547→    }
   548→  } catch {
   549→    /* fall through */
   550→  }
   551→  if (textParts && textParts.parts.length > 1) return textParts;
   552→  throw new Error('Could not read the lecture manifest (segments unreachable).');
   553→}

   566→function pickHeaders(res: Response): Headers {
   567→  const h = new Headers();
   568→  const ct = res.headers.get('content-type');
   569→  if (ct) h.set('Content-Type', ct);
   570→  h.set('Cache-Control', 'immutable');
   571→  return h;

   574→/**
   575→ * Start (or resume) a lecture download. Resolves once the download pipeline
   576→ * has been kicked off (progress continues in the background — observe via
   577→ * subscribeDownloads / listDownloads).
   578→ */
   579→export async function startDownload(spec: DownloadStartSpec): Promise<SbDownload> {
   580→  const now = new Date().toISOString();
   581→  const existing = await getDownload(spec.id);
   582→  // resumable = a healthy in-flight download whose part list matches the
   583→  // signed source it was collected against — keep that URL so the
   584→  // already-downloaded parts (absolute URLs with that signature) still match.
   585→  const resumable = Boolean(
   586→    existing &&
   587→      existing.parts.length > 0 &&
   588→      (existing.status === 'paused' || existing.status === 'downloading' || existing.status === 'collecting'),
   589→  );
   590→  let rec: SbDownload;
   591→  if (!existing || existing.status === 'done') {
   592→    // brand-new download (or full re-download)
   593→    rec = {
   594→      ...spec,
   595→      folder: folderFromMediaUrl(spec.url) || spec.id.replace(/[^a-zA-Z0-9_-]/g, '_').slice(0, 40),
   596→      height: null,
   597→      parts: [],
   598→      partsDone: 0,
   599→      sizeBytes: 0,
   600→      status: 'collecting',
   601→      error: undefined,
   602→      createdAt: existing?.createdAt || now,
   603→      updatedAt: now,
   604→    };
   605→  } else if (resumable && existing) {
   606→    rec = { ...existing, ...spec, url: existing.url, folder: existing.folder, updatedAt: now };
   607→  } else if (existing) {
   608→    // error state — drop the stale part list and re-collect from the source
   609→    rec = {
   610→      ...existing,
   611→      ...spec,
   612→      folder: folderFromMediaUrl(spec.url) || existing.folder,
   613→      parts: [],
   614→      partsDone: 0,
   615→      sizeBytes: 0,
   616→      height: null,
   617→      status: 'collecting',
   618→      error: undefined,
   619→      updatedAt: now,
   620→    };
   621→  } else {
   622→    throw new Error('unreachable');
   623→  }
   624→
   625→  if (activeControllers.has(spec.id)) return rec; // already running
   626→  const ctrl = new AbortController();
   627→  activeControllers.set(spec.id, ctrl);
   628→
   629→  await putDownload(rec);
   630→  notifyChange();
   631→
   632→  // fire-and-forget the heavy lifting
   633→  void runDownload(rec, ctrl).finally(() => activeControllers.delete(spec.id));
   634→  return rec;
   635→}
   636→
   637→async function runDownload(rec: SbDownload, ctrl: AbortController): Promise<void> {
   638→  try {
   639→    // ---- 1. collect parts (re-collect on resume if we never finished) ----
   640→    if (rec.parts.length === 0) {
   641→      rec.status = 'collecting';
   642→      await putDownload(rec);
   643→      notifyChange();
   644→      const collected = await collectParts(rec.kind, rec.url, rec.clearKeys);
   645→      if (ctrl.signal.aborted) return;
   646→      rec.parts = collected.parts;
   647→      rec.height = collected.height;
   648→      rec.url = collected.playbackUrl;
   649→      rec.partsDone = 0;
   650→      rec.sizeBytes = 0;
   651→      await putDownload(rec);
   652→      notifyChange();
   653→    }
   654→
   655→    if (!('caches' in window)) throw new Error('Cache API unavailable in this browser');
   656→    const cache = await caches.open(videoCacheName(rec.folder));
   657→
   658→    // ---- 2. download pool ----
   659→    rec.status = 'downloading';
   660→    await putDownload(rec);
   661→    notifyChange();
   662→
   663→    let next = 0;
   664→    let done = rec.partsDone;
   665→    let size = rec.sizeBytes;
   666→    let failure: Error | null = null;
   667→
   668→    const grab = async (): Promise<void> => {
   669→      for (;;) {
   670→        if (ctrl.signal.aborted || failure) return;
   671→        const i = next++;
   672→        if (i >= rec.parts.length) return;
   673→        const partUrl = rec.parts[i];
   674→        try {
   675→          const hit = await cache.match(partUrl); // resume: keep already-cached parts
   676→          if (hit) {
   677→            size += Number(hit.headers.get('content-length') || 0);
   678→          } else {
   679→            let stored = false;
   680→            for (let attempt = 0; attempt < 2 && !stored; attempt++) {
   681→              try {
   682→                const res = await fetch(partUrl, { signal: ctrl.signal });
   683→                if (!res.ok) throw new Error(`HTTP ${res.status}`);
   684→                const buf = await res.arrayBuffer();
   685→                size += buf.byteLength;
   686→                await cache.put(partUrl, new Response(buf, { headers: pickHeaders(res) }));
   687→                stored = true;
   688→              } catch (e) {
   689→                if (ctrl.signal.aborted) return;
   690→                if (attempt === 1) throw e;
   691→                await sleep(1500); // one soft retry on flaky network
   692→              }
   693→            }
   694→            if (!stored) throw new Error('Fetch failed twice');
   695→          }
   696→          done++;
   697→          if (done % 5 === 0 || done === rec.parts.length) {
   698→            rec.partsDone = done;
   699→            rec.sizeBytes = size;
   700→            await putDownload(rec);
   701→            notifyChange();
   702→          }
   703→        } catch (e) {
   704→          if (ctrl.signal.aborted) return;
   705→          failure = e instanceof Error ? e : new Error('Download failed');
   706→        }
   707→      }
   708→    };
   709→
   710→    await Promise.all(Array.from({ length: Math.min(CONCURRENCY, rec.parts.length) }, grab));
   711→
   712→    if (ctrl.signal.aborted) return;
   713→    if (failure) throw failure;
   714→
   715→    rec.partsDone = rec.parts.length;
   716→    rec.sizeBytes = size;
   717→    rec.status = 'done';
   718→    rec.error = undefined;
   719→    await putDownload(rec);
   720→    notifyChange();
   721→    await ensurePersistentStorage();
   722→  } catch (e) {
   723→    if (ctrl.signal.aborted) return; // paused — record keeps its state
   724→    rec.status = 'error';
   725→    rec.error = e instanceof Error ? e.message : 'Download failed';
   726→    await putDownload(rec);
   727→    notifyChange();
   728→  }
   729→}

   731→/** Pause an in-flight download (resume later with startDownload). */
   732→export async function pauseDownload(id: string): Promise<void> {
   733→  const ctrl = activeControllers.get(id);
   734→  if (ctrl) {
   735→    activeControllers.delete(id);
   736→    ctrl.abort();
   737→  }
   738→  const rec = await getDownload(id);
   739→  if (rec && (rec.status === 'downloading' || rec.status === 'collecting')) {
   740→    rec.status = 'paused';
   741→    await putDownload(rec);
   742→    notifyChange();
   743→  }
   744→}
   745→
   746→export async function isDownloading(id: string): Promise<boolean> {
   747→  return activeControllers.has(id);
   748→}

   750→/** Delete a lecture's offline copy (cache + record). */
   751→export async function deleteDownload(id: string): Promise<void> {
   752→  await pauseDownload(id);
   753→  const rec = await getDownload(id);
   754→  if (rec) {
   755→    try {
   756→      await caches.delete(videoCacheName(rec.folder));
   757→    } catch {
   758→      /* ignore */
   759→    }
   760→    await tx('readwrite', (s) => s.delete(rec.id) as unknown as IDBRequest<undefined>);
   761→    notifyChange();
   762→  }
   763→}

   765→/** Delete every offline lecture (used by "Clear all"). */
   766→export async function deleteAllDownloads(): Promise<void> {
   767→  const all = await listDownloads();
   768→  for (const rec of all) await deleteDownload(rec.id);
   769→}

   775→export function formatBytes(n: number): string {
   776→  if (!n || n < 0) return '—';
   777→  if (n < 1024) return `${n} B`;
   778→  const kb = n / 1024;
   779→  if (kb < 1024) return `${kb.toFixed(0)} KB`;
   780→  const mb = kb / 1024;
   781→  if (mb < 1024) return `${mb.toFixed(mb < 10 ? 1 : 0)} MB`;
   782→  return `${(mb / 1024).toFixed(2)} GB`;
   783→}
