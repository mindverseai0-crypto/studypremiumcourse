    96→    96→function loadCatalog(): CatalogBatch[] {
    97→    97→  if (catalogCache) return catalogCache;
    98→    98→  const file = path.join(process.cwd(), 'src', 'data', 'all_batches.json');
    99→    99→  try {
   100→   100→    const raw = fs.readFileSync(file, 'utf-8');
   101→   101→    const parsed = JSON.parse(raw);
   102→   102→    catalogCache = Array.isArray(parsed) ? (parsed as CatalogBatch[]) : [];
   103→   103→  } catch {
   104→   104→    catalogCache = [];
   105→   105→  }
   106→   106→  return catalogCache;

   127→   127→// ---------------------------------------------------------------------------
   128→   128→// Catalog search — 20/page, search by name / id / byName.
   129→   129→// (Catalog rows keep subjects empty: building content bundles per row would
   130→   130→// be wasteful; subjects are attached on pw_btch_dtl / single-batch detail.)
   131→   131→// ---------------------------------------------------------------------------
   132→   132→export function searchCatalog(query: string, page: number, limit: number): CatalogResult {
   133→   133→  const catalog = loadCatalog();
   134→   134→  const q = (query || '').trim().toLowerCase();
   135→   135→
   136→   136→  const filtered = q
   137→   137→    ? catalog.filter(
   138→   138→        (b) =>
   139→   139→          b.name.toLowerCase().includes(q) ||
   140→   140→          b.batch_id.toLowerCase().includes(q) ||
   141→   141→          b.id.toLowerCase().includes(q) ||
   142→   142→          b.byName.toLowerCase().includes(q),
   143→   143→      )
   144→   144→    : catalog;
   145→   145→
   146→   146→  const safeLimit = Math.min(100, Math.max(1, limit));
   147→   147→  const safePage = Math.max(1, page);
   148→   148→  const start = (safePage - 1) * safeLimit;
   149→   149→  const slice = filtered.slice(start, start + safeLimit);
   150→   150→
   151→   151→  return {
   152→   152→    total: filtered.length,
   153→   153→    page: safePage,
   154→   154→    limit: safeLimit,
   155→   155→    hasMore: start + safeLimit < filtered.length,
   156→   156→    data: slice.map(toDetail),
   157→   157→  };
   158→   158→}

   163→   163→export function getBatchDetail(batchId: string): BatchDetail | null {
   164→   164→  const catalog = loadCatalog();
   165→   165→  const id = (batchId || '').trim().toLowerCase();
   166→   166→  if (!id) return null;
   167→   167→  const found = catalog.find((b) => b.batch_id.toLowerCase() === id);
   168→   168→  if (!found) return null;
   169→   169→  const detail = toDetail(found);
   170→   170→  detail.subjects = getSubjects(found.batch_id);
   171→   171→  return detail;
   172→   172→}

   262→   262→function docFileId(kind: 'NOTE' | 'DPP', title: string, subject: string, chapter: string): string {
   263→   263→  const payload = JSON.stringify({ k: kind, t: title, s: subject, c: chapter });
   264→   264→  const b64 = Buffer.from(payload, 'utf-8').toString('base64url');
   265→   265→  return `d_${b64}`;

   268→   268→function fmtDuration(rng: ReturnType<typeof makeRng>): string {
   269→   269→  const h = rng.int(0, 2);
   270→   270→  const m = rng.int(8, 55);
   271→   271→  return h > 0 ? `${h}h ${m}m` : `${m}m`;
   272→   272→}

   274→   274→/** Spread n lecture dates evenly over the batch validity window (capped at now + 30d). */
   275→   275→function lectureDates(startDate: string, endDate: string, n: number): string[] {
   276→   276→  const now = Date.now();
   277→   277→  let start = new Date(startDate || '').getTime();
   278→   278→  let end = new Date(endDate || '').getTime();
   279→   279→  if (!Number.isFinite(start)) start = now - 90 * 86400000;
   280→   280→  if (!Number.isFinite(end)) end = start + 120 * 86400000;
   281→   281→  start = Math.max(start, now - 90 * 86400000);
   282→   282→  end = Math.min(end, now + 30 * 86400000);
   283→   283→  if (end <= start) end = start + 86400000;
   284→   284→  const out: string[] = [];
   285→   285→  for (let i = 0; i < n; i++) {
   286→   286→    const frac = (i + 1) / (n + 1);
   287→   287→    out.push(new Date(start + frac * (end - start)).toISOString());
   288→   288→  }
   289→   289→  return out;
   290→   290→}

   293→   293→  return n < 10 ? `0${n}` : String(n);

   296→   296→function getBatchContent(batchId: string): BatchContent | null {
   297→   297→  const batch = findBatch(batchId);
   298→   298→  if (!batch) return null;
   299→   299→
   300→   300→  const cached = bundleCache.get(batch.batch_id);
   301→   301→  if (cached) return cached;
   302→   302→
   303→   303→  const short = isShortCourse(batch.name, batch.byName);
   304→   304→  const templates: SubjectTemplate[] = pickSubjects({
   305→   305→    name: batch.name,
   306→   306→    byName: batch.byName,
   307→   307→    exam: Array.isArray(batch.exam) ? batch.exam : [],
   308→   308→    className: batch.class || '',
   309→   309→  });
   310→   310→
   311→   311→  const subjects: SubjectData[] = templates.map((tpl) => {
   312→   312→    const subjectId = stableId('sub', batch.batch_id, tpl.subject);
   313→   313→    const rngChapters = makeRng('chapters', batch.batch_id, tpl.subject);
   314→   314→
   315→   315→    const bank = tpl.chapters;
   316→   316→    const chapterCount = short ? rngChapters.int(5, 8) : Math.min(bank.length, rngChapters.int(8, 14));
   317→   317→    const chapters: ChapterRef[] = [];
   318→   318→    const lectures: LectureRef[] = [];
   319→   319→    const notes: NotesRef[] = [];
   320→   320→    const dpps: DppRef[] = [];
   321→   321→
   322→   322→    const pickedChapters = short ? rngChapters.sample(bank, chapterCount) : bank.slice(0, chapterCount);
   323→   323→
   324→   324→    pickedChapters.forEach((chapterName, ci) => {
   325→   325→      const chapterId = stableId('ch', batch.batch_id, tpl.subject, chapterName);
   326→   326→      const rng = makeRng('chapter', batch.batch_id, chapterName);
   327→   327→
   328→   328→      const lectureCount = short ? rng.int(3, 5) : rng.int(4, 9);
   329→   329→      const notesCount = short ? 1 : rng.int(1, 2);
   330→   330→      const dppCount = short ? rng.int(1, 3) : rng.int(2, 4);
   331→   331→
   332→   332→      chapters.push({
   333→   333→        id: chapterId,
   334→   334→        name: chapterName,
   335→   335→        subjectId,
   336→   336→        subjectName: tpl.subject,
   337→   337→        lectureCount,
   338→   338→        notesCount,
   339→   339→        dppCount,
   340→   340→      });
   341→   341→
   342→   342→      const dates = lectureDates(batch.start_date, batch.end_date, lectureCount);
   343→   343→      const subtopics = rng.shuffle(LECTURE_SUBTOPICS);
   344→   344→
   345→   345→      // DPP refs for this chapter (shared across its lectures).
   346→   346→      const chapterDpps: DppRef[] = [];
   347→   347→      for (let di = 0; di < dppCount; di++) {
   348→   348→        const dppId = stableId('dpp', batch.batch_id, chapterId, di);
   349→   349→        const questions = short ? rng.int(5, 8) : rng.int(8, 15);
   350→   350→        chapterDpps.push({
   351→   351→          testId: dppId,
   352→   352→          name: `DPP ${pad(di + 1)} : ${chapterName}`,
   353→   353→          chapterId,
   354→   354→          chapterName,
   355→   355→          subjectId,
   356→   356→          subjectName: tpl.subject,
   357→   357→          totalQuestions: questions,
   358→   358→          totalMarks: questions * 4,
   359→   359→          docId: docFileId('DPP', `${chapterName} — DPP ${pad(di + 1)}`, tpl.subject, chapterName),
   360→   360→        });
   361→   361→      }
   362→   362→      dpps.push(...chapterDpps);
   363→   363→
   364→   364→      for (let li = 0; li < lectureCount; li++) {
   365→   365→        const nodeId = stableId('lec', batch.batch_id, chapterId, li);
   366→   366→        const videoId = stableId('vid', batch.batch_id, chapterId, li);
   367→   367→        const subtopic = subtopics[li % subtopics.length];
   368→   368→        const linked = chapterDpps.length > 0 ? [chapterDpps[li % chapterDpps.length]] : [];
   369→   369→        lectures.push({
   370→   370→          nodeId,
   371→   371→          videoId,
   372→   372→          topic: `Lecture ${pad(li + 1)} : ${subtopic}`,
   373→   373→          subtopic,
   374→   374→          date: dates[li],
   375→   375→          duration: fmtDuration(rng),
   376→   376→          chapterId,
   377→   377→          chapterName,
   378→   378→          subjectId,
   379→   379→          subjectName: tpl.subject,
   380→   380→          subjectTeacher: '', // filled after teacher selection below
   381→   381→          notesDocId: docFileId('NOTE', `${chapterName} — Class Notes`, tpl.subject, chapterName),
   382→   382→          dppRefs: linked.map((d) => ({ id: d.testId, docId: d.docId, title: d.name })),
   383→   383→        });
   384→   384→      }
   385→   385→
   386→   386→      for (let ni = 0; ni < notesCount; ni++) {
   387→   387→        const title = STUDY_MATERIAL_TITLES[(ci + ni) % STUDY_MATERIAL_TITLES.length];
   388→   388→        notes.push({
   389→   389→          nodeId: stableId('nt', batch.batch_id, chapterId, ni),
   390→   390→          topic: `${chapterName} : ${title}`,
   391→   391→          chapterId,
   392→   392→          chapterName,
   393→   393→          docId: docFileId('NOTE', `${chapterName} — ${title}`, tpl.subject, chapterName),
   394→   394→          date: dates[Math.min(ni, dates.length - 1)] ?? new Date().toISOString(),
   395→   395→        });
   396→   396→      }
   397→   397→    });
   398→   398→
   399→   399→    const t = teacherFor([batch.batch_id, tpl.subject]);
   400→   400→    const teacher = { name: t.name, qualification: t.qualification, experience: t.experience };
   401→   401→    for (const l of lectures) l.subjectTeacher = teacher.name;
   402→   402→
   403→   403→    return {
   404→   404→      id: subjectId,
   405→   405→      name: tpl.subject,
   406→   406→      icon: tpl.icon,
   407→   407→      chapters,
   408→   408→      lectures,
   409→   409→      notes,
   410→   410→      dpps,
   411→   411→      teacher,
   412→   412→    };
   413→   413→  });
   414→   414→
   415→   415→  // Part tests & full mocks
   416→   416→  const rngTests = makeRng('tests', batch.batch_id);
   417→   417→  const bigTests: BigTestRef[] = [];
   418→   418→  const partCount = short ? rngTests.int(1, 2) : rngTests.int(2, 4);
   419→   419→  for (let i = 0; i < partCount; i++) {
   420→   420→    const subjectName = subjects.length > 0 ? subjects[i % subjects.length].name : 'General';
   421→   421→    const qs = rngTests.int(12, 18);
   422→   422→    bigTests.push({
   423→   423→      testId: stableId('tst', batch.batch_id, 'part', i),
   424→   424→      name: `Part Test ${pad(i + 1)} : ${subjectName}`,
   425→   425→      subjectName,
   426→   426→      totalQuestions: qs,
   427→   427→      totalMarks: qs * 4,
   428→   428→      duration: Math.max(30, Math.round((qs * 1.5) / 5) * 5),
   429→   429→      kind: 'PART',
   430→   430→    });
   431→   431→  }
   432→   432→  const mockCount = short ? rngTests.int(1, 2) : rngTests.int(2, 3);
   433→   433→  for (let i = 0; i < mockCount; i++) {
   434→   434→    const qs = rngTests.int(20, 28);
   435→   435→    bigTests.push({
   436→   436→      testId: stableId('tst', batch.batch_id, 'mock', i),
   437→   437→      name: `Full Syllabus Mock ${pad(i + 1)}`,
   438→   438→      subjectName: subjects.length > 0 ? subjects[i % subjects.length].name : 'General',
   439→   439→      totalQuestions: qs,
   440→   440→      totalMarks: qs * 4,
   441→   441→      duration: Math.max(45, Math.round((qs * 1.5) / 5) * 5),
   442→   442→      kind: 'MOCK',
   443→   443→    });
   444→   444→  }
   445→   445→
   446→   446→  const bundle: BatchContent = { batch, short, subjects, bigTests };
   447→   447→  if (bundleCache.size >= BUNDLE_CACHE_MAX) {
   448→   448→    const first = bundleCache.keys().next().value;
   449→   449→    if (first) bundleCache.delete(first);
   450→   450→  }
   451→   451→  bundleCache.set(batch.batch_id, bundle);
   452→   452→  return bundle;
   453→   453→}

   472→   472→export function getChapters(
   473→   473→  batchId: string,
   474→   474→  subjectId: string,
   475→   475→  tagType: string,
   476→   476→  page = 1,
   477→   477→  limit = 50,
   478→   478→): ChapterItem[] {
   479→   479→  const bundle = getBatchContent(batchId);
   480→   480→  if (!bundle) return [];
   481→   481→  const subject = bundle.subjects.find((s) => s.id === subjectId);
   482→   482→  if (!subject) return [];
   483→   483→
   484→   484→  let items: ChapterItem[];
   485→   485→  if (tagType === 'STUDY_MATERIAL') {
   486→   486→    items = subject.notes.map((n) => ({
   487→   487→      _id: n.nodeId,
   488→   488→      name: n.topic,

   623→   623→export function getContentNodes(
   624→   624→  batchId: string,
   625→   625→  subjectId: string,
   626→   626→  tagId: string,
   627→   627→  contentType = 'ALL',
   628→   628→  skip = 0,
   629→   629→  limit = 50,
   630→   630→): ContentNode[] {
   631→   631→  const bundle = getBatchContent(batchId);
   632→   632→  if (!bundle) return [];
   633→   633→
   634→   634→  let nodes: ContentNode[] = [];
   635→   635→
   636→   636→  if (tagId.startsWith('res_')) {
   637→   637→    // Resource folder — everything for the subject.
   638→   638→    const subject = bundle.subjects.find((s) => s.id === subjectId) ?? bundle.subjects[0];
   639→   639→    if (!subject) return [];
   640→   640→    nodes = [
   641→   641→      ...subject.notes.map(notesNode),
   642→   642→      ...subject.dpps.map(dppNode),
   643→   643→      ...subject.lectures.map(lectureNode),
   644→   644→    ];
   645→   645→  } else {
   646→   646→    // Chapter / study-material topic.
   647→   647→    for (const subject of bundle.subjects) {
   648→   648→      const chapter = subject.chapters.find((c) => c.id === tagId);
   649→   649→      if (chapter) {
   650→   650→        const lectures = subject.lectures.filter((l) => l.chapterId === chapter.id);
   651→   651→        const notes = subject.notes.filter((n) => n.chapterId === chapter.id);
   652→   652→        const dpps = subject.dpps.filter((d) => d.chapterId === chapter.id);
   653→   653→        const type = (contentType || 'ALL').toUpperCase();
   654→   654→        if (type === 'LECTURE') nodes = lectures.map(lectureNode);
   655→   655→        else if (type === 'NOTES') nodes = notes.map(notesNode);
   656→   656→        else if (type === 'DPP_PDF') nodes = dpps.map(dppNode);
   657→   657→        else if (type === 'DPP_VIDEOS') nodes = dpps.map(dppVideoNode);
   658→   658→        else nodes = [...lectures.map(lectureNode), ...notes.map(notesNode), ...dpps.map(dppNode)];
   659→   659→        break;
   660→   660→      }
   661→   661→      const sm = subject.notes.find((n) => n.nodeId === tagId);
   662→   662→      if (sm) {
   663→   663→        nodes = [notesNode(sm)];
   664→   664→        break;
   665→   665→      }
   666→   666→    }
   667→   667→  }
   668→   668→
   669→   669→  return nodes.slice(Math.max(0, skip), Math.max(0, skip) + Math.max(1, limit));
   670→   670→}

   743→   743→export function getTodaySchedule(batchId: string): ContentNode[] {
   744→   744→  const bundle = getBatchContent(batchId);
   745→   745→  if (!bundle || bundle.subjects.length === 0) return [];
   746→   746→
   747→   747→  const rng = makeRng('today', bundle.batch.batch_id, new Date().toISOString().slice(0, 10));
   748→   748→  const now = Date.now();
   749→   749→
   750→   750→  const allLectures = bundle.subjects.flatMap((s) => s.lectures);
   751→   751→  if (allLectures.length === 0) return [];
   752→   752→
   753→   753→  const nodes: ContentNode[] = [];
   754→   754→
   755→   755→  const liveLecture = rng.pick(allLectures);
   756→   756→  nodes.push(
   757→   757→    eventNode(
   758→   758→      liveLecture,
   759→   759→      new Date(now - 35 * 60000).toISOString(),
   760→   760→      new Date(now + 25 * 60000).toISOString(),
   761→   761→      'LIVE',
   762→   762→    ),
   763→   763→  );
   764→   764→
   765→   765→  const upcomingCount = rng.int(1, 2);
   766→   766→  for (let i = 0; i < upcomingCount; i++) {
   767→   767→    const lec = rng.pick(allLectures);
   768→   768→    const start = new Date(now + (90 + i * 150) * 60000);
   769→   769→    nodes.push(eventNode(lec, start.toISOString(), new Date(start.getTime() + 3600000).toISOString(), 'UPCOMING'));
   770→   770→  }
   771→   771→
   772→   772→  const tomorrow = new Date(now + 86400000);
   773→   773→  tomorrow.setHours(9, 30, 0, 0);
   774→   774→  nodes.push(
   775→   775→    eventNode(
   776→   776→      rng.pick(allLectures),
   777→   777→      tomorrow.toISOString(),
   778→   778→      new Date(tomorrow.getTime() + 3600000).toISOString(),
   779→   779→      'UPCOMING',
   780→   780→    ),
   781→   781→  );
   782→   782→
   783→   783→  return nodes;
   784→   784→}

   866→   866→    else if (tpl.type === 'TEST' && allTests.length > 0) topic = rng.pick(allTests).name;
   867→   867→    else topic = bundle.batch.name;

   875→   875→      scheduleTime: new Date(now - ageHours * 3600000).toISOString(),
   876→   876→      isRead: i > 5,
   877→   877→    });
   878→   878→  }
   879→   879→
   880→   880→  const start = (Math.max(1, page) - 1) * limit;
   881→   881→  const slice = items.slice(start, start + limit);

   954→   954→  // Build the question pool from the subject bank + general banks.
   955→   955→  const pool = [
   956→   956→    ...quizBankForSubject(subjectName),
   957→   957→    ...quizBankForSubject('Quantitative Aptitude'),
   958→   958→    ...quizBankForSubject('General Awareness'),
   959→   959→    ...quizBankForSubject('English Language'),
   960→   960→  ];
   961→   961→  const rng = makeRng('quiz', batchId, testId);
   962→   962→  const shuffled = rng.shuffle(pool).slice(0, Math.max(1, Math.min(questionCount, pool.length)));
   963→   963→  const questions = shuffled.map((qq, i) => {

   967→   967→    return {
   968→   968→      _id: stableId('q', testId, i),
   969→   969→      question: qq.q,

   971→   971→      answer: indices.indexOf(qq.answer),
   972→   972→      marks: 4,
   973→   973→    };
   974→   974→  });
   975→   975→
   976→   976→  const duration = Math.max(10, Math.round((questions.length * 1.5) / 5) * 5);
   977→   977→  return { _id: testId, name, duration, questions };
