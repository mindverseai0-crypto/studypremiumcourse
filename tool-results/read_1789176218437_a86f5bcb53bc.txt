    45→const asParams = (v: unknown): Params =>

    56→/** Payload may arrive as object or JSON string. */
    57→function parsePayload(payload: unknown): unknown {
    58→  if (typeof payload === "string") {
    59→    try {
    60→      return JSON.parse(payload);
    61→    } catch {
    62→      return payload;
    63→    }
    64→  }
    65→  return payload;
    66→}

    76→/** PDF attachments are baked as /api/lxpdf links that resolve the real file. */
    77→function pdfAttachmentRef(batchId: string, subjectId: string, pdfId: string, attachmentId: string) {
    78→  const q = `batchId=${encodeURIComponent(batchId)}&subjectId=${encodeURIComponent(
    79→    subjectId,
    80→  )}&pdfId=${encodeURIComponent(pdfId)}&attachmentId=${encodeURIComponent(attachmentId)}`;
    81→  return { baseUrl: `/api/lxpdf?${q}`, key: "", _id: attachmentId };
    82→}

    84→function subjectRef(s: unknown): { _id: string; slug: string; name: string } {
    85→  if (s && typeof s === "object") {
    86→    const o = s as { _id?: string; slug?: string; name?: string };
    87→    return { _id: o._id || "", slug: o.slug || o._id || "", name: o.name || "" };
    88→  }
    89→  return { _id: String(s || ""), slug: String(s || ""), name: "" };
    90→}

    92→function eventStatus(item: LxScheduleItem, startIso: string, endIso: string): string {
    93→  const tag = String(item.tag || "").toLowerCase();
    94→  if (tag === "live" || tag === "live now") return "LIVE";
    95→  if (tag === "upcoming") return "UPCOMING";
    96→  if (tag === "ended" || tag === "cancelled") return tag === "cancelled" ? "CANCELLED" : "ENDED";
    97→  const now = Date.now();
    98→  const s = new Date(startIso || "").getTime();
    99→  const e = new Date(endIso || "").getTime();
   100→  if (Number.isFinite(s) && Number.isFinite(e)) {
   101→    if (now >= s && now <= e) return "LIVE";
   102→    if (now < s) return "UPCOMING";
   103→  }
   104→  return "ENDED";
   105→}

   135→function contentNodesFromItems(
   136→  batchId: string,
   137→  subjectId: string,
   138→  items: LxScheduleItem[],
   139→  kind: "LECTURE" | "DPP_VIDEOS" | "NOTES",
   140→): Array<{ _id: string; type: string; data: Record<string, unknown> }> {
   141→  const nodes: Array<{ _id: string; type: string; data: Record<string, unknown> }> = [];
   142→  for (const item of items) {
   143→    const pdfId = item._id || "";
   144→
   145→    if (kind === "LECTURE" || kind === "DPP_VIDEOS") {
   146→      const vd = item.videoDetails || {};
   147→      nodes.push({
   148→        _id: item._id || "",
   149→        type: kind,
   150→        data: {
   151→          _id: item._id || "",
   152→          topic: item.topic || vd.name || item.name || "Lecture",
   153→          date: item.date || item.startTime || "",
   154→          urlType: item.urlType || "",
   155→          url: item.url || "",
   156→          videoDetails: {
   157→            _id: item._id || "",
   158→            image: vd.image || item.previewImageUrl || "",
   159→            videoUrl: "",
   160→            duration: vd.duration || "",
   161→          },
   162→          homeworkIds: asArray(item.homeworkIds).flatMap((hw) => {
   163→            const h = hw as { _id?: string; topic?: string; attachmentIds?: unknown[] };
   164→            return asArray(h.attachmentIds).map((att) => {
   165→              const a = att as { _id?: string };
   166→              return {
   167→                _id: h._id || "",
   168→                topic: h.topic || "Class Note",
   169→                attachmentIds: [pdfAttachmentRef(batchId, subjectId, pdfId, a._id || "")],
   170→              };
   171→            });
   172→          }),
   173→          exerciseIds: [],
   174→        },
   175→      });
   176→      continue;
   177→    }
   178→
   179→    // NOTES / DppNotes → PDF cards
   180→    const atts: Array<Record<string, unknown>> = [];
   181→    let title = item.topic || item.name || "";
   182→    const homeworkSources = [
   183→      ...asArray(item.homeworkIds),
   184→      ...asArray(item.dpp?.homeworkIds),
   185→    ];
   186→    for (const hw of homeworkSources) {
   187→      const h = hw as { _id?: string; topic?: string; attachmentIds?: unknown[] };
   188→      if (!title && h.topic) title = h.topic;
   189→      for (const att of asArray(h.attachmentIds)) {
   190→        const a = att as { _id?: string; key?: string; baseUrl?: string; name?: string };
   191→        if (!a?._id) continue;
   192→        if (!title) title = a.name || "";
   193→        // some attachments already carry a real static.pw.live key
   194→        const direct = a.baseUrl && a.key ? `${a.baseUrl}${a.key}` : "";
   195→        const ref = direct
   196→          ? { baseUrl: direct, key: "", _id: a._id }
   197→          : pdfAttachmentRef(batchId, subjectId, pdfId, a._id);
   198→        atts.push({
   199→          _id: a._id,
   200→          topic: h.topic || a.name || item.topic || item.name || "Study Material",
   201→          attachmentIds: [ref],
   202→        });
   203→      }
   204→    }
   205→    if (atts.length === 0) continue;
   206→    nodes.push({
   207→      _id: item._id || "",
   208→      type: "NOTES",
   209→      data: {
   210→        _id: item._id || "",
   211→        topic: title || "Study Material",
   212→        date: item.date || item.startTime || "",
   213→        homeworkIds: atts,
   214→      },
   215→    });
   216→  }
   217→  return nodes;
   218→}

   220→function resolveContentKind(contentType: string): "LECTURE" | "DPP_VIDEOS" | "NOTES" | null {
   221→  switch ((contentType || "ALL").toUpperCase()) {
   222→    case "ALL":
   223→    case "LECTURE":
   224→      return "LECTURE";
   225→    case "DPP_VIDEOS":
   226→      return "DPP_VIDEOS";
   227→    case "NOTES":
   228→    case "DPP_PDF":
   229→      return "NOTES";
   230→    default:
   231→      return null;
   232→  }
   233→}

   240→      return "DppNotes";

   248→/** base64url-encode a signed query for the stateless /s/<sig>/ relay path. */
   249→function sigPathSegment(signedQuery: string): string {
   250→  const q = signedQuery.startsWith("?") ? signedQuery.slice(1) : signedQuery;
   251→  if (!q) return "";
   252→  return Buffer.from(q, "utf8").toString("base64url");
   253→}

   255→/** Resolve a video → same-origin playable payload (dash/hls/youtube/mp4). */
   256→async function resolveVideoPayload(childId: string, batchId: string, subjectId: string) {
   257→  const vid = await lxVideoUrl(batchId, subjectId, childId);
   258→  if (vid?.url) {
   259→    const url = vid.url.trim();
   260→    if (/\.mpd(\?|$)/i.test(url)) {
   261→      const folder = folderFromUrl(url);
   262→      const sig = sigPathSegment(vid.signedUrl || "");
   263→      if (folder && sig) registerSignedFolder(folder, vid.signedUrl || "");
   264→      let clearKeys: Record<string, string> | null = null;
   265→      try {
   266→        const mpdUrl = `${url}${vid.signedUrl || ""}`;
   267→        const controller = new AbortController();
   268→        const timer = setTimeout(() => controller.abort(), 30000);
   269→        const mpdRes = await fetch(mpdUrl, {
   270→          cache: "no-store",
   271→          signal: controller.signal,
   272→          headers: { "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64)" },
   273→        });
   274→        clearTimeout(timer);
   275→        if (mpdRes.ok) {
   276→          const text = await mpdRes.text();
   277→          const kidMatch = text.match(/default_KID="([0-9a-fA-F-]+)"/);
   278→          if (kidMatch) {
   279→            clearKeys = await lxOtp(kidMatch[1]);
   280→          }
   281→        }
   282→      } catch {
   283→        /* keys unavailable — player will surface a decrypt error */
   284→      }
   285→      // Stateless relay URL: relative DASH segments inherit the signature.
   286→      const manifestUrl = folder
   287→        ? `/api/pw/${folder}${sig ? `/s/${sig}` : ""}/master.mpd`
   288→        : "";
   289→      if (!manifestUrl) return null;
   290→      return {
   291→        kind: "dash",
   292→        url: manifestUrl,
   293→        signedUrl: "",
   294→        clearKeys,
   295→        poster: "",
   296→        drmDetails: null,
   297→      };
   298→    }
   299→    if (/\.m3u8(\?|$)/i.test(url)) {
   300→      const folder = folderFromUrl(url);
   301→      const sig = sigPathSegment(vid.signedUrl || "");
   302→      if (!folder) return null;
   303→      if (sig) registerSignedFolder(folder, vid.signedUrl || "");
   304→      return {
   305→        kind: "hls",
   306→        url: `/api/pw/${folder}${sig ? `/s/${sig}` : ""}/master.m3u8`,
   307→        signedUrl: "",
   308→        clearKeys: null,
   309→        poster: "",
   310→        drmDetails: null,
   311→      };
   312→    }
   313→    if (/youtube\.com|youtu\.be/i.test(url)) {
   314→      return { kind: "youtube", url, signedUrl: "", clearKeys: null, poster: "", drmDetails: null };
   315→    }
   316→    // plain file (mp4 etc.) inside the signed folder
   317→    const folder = folderFromUrl(url);
   318→    const sig = sigPathSegment(vid.signedUrl || "");
   319→    if (folder) {
   320→      if (sig) registerSignedFolder(folder, vid.signedUrl || "");
   321→      const fileName = url.split("/").pop() || "video.mp4";
   322→      return {
   323→        kind: "mp4",
   324→        url: `/api/pw/${folder}${sig ? `/s/${sig}` : ""}/${fileName}`,
   325→        signedUrl: "",
   326→        clearKeys: null,
   327→        poster: "",
   328→        drmDetails: null,
   329→      };
   330→    }
   331→    return { kind: "mp4", url, signedUrl: "", clearKeys: null, poster: "", drmDetails: null };
   332→  }
   333→
   334→  // No signed stream — maybe a youtube lecture.
   335→  if (subjectId) {
   336→    const detail = await lxScheduleDetail(batchId, subjectId, childId);
   337→    const url = detail?.url || "";
   338→    if (url && /youtube\.com|youtu\.be/i.test(url)) {
   339→      return { kind: "youtube", url, signedUrl: "", clearKeys: null, poster: "", drmDetails: null };
   340→    }
   341→  }
   342→  return null;
   343→}

   345→// ---------------------------------------------------------------------------
   346→// POST /api/data — action API (real upstream data, transformed)
   347→// ---------------------------------------------------------------------------
   348→export async function POST(request: NextRequest) {
   349→  try {
   350→    const body = (await request.json().catch(() => ({}))) as ActionBody;
   351→    const action = typeof body.action === "string" ? body.action : "";
   352→    const params = asParams(body.params);
   353→    const payload = parsePayload(body.payload);
   354→
   355→    switch (action) {
   356→      case "pw_btch_dtl": {
   357→        const batchId = str(params.batchId).trim();
   358→        if (!batchId) return NextResponse.json({ success: false }, { headers: CORS_HEADERS });
   359→        const d = await lxBatchDetails(batchId);
   360→        if (!d) return NextResponse.json({ success: false }, { headers: CORS_HEADERS });
   361→
   362→        const exams = Array.isArray(d.exam)
   363→          ? (d.exam as unknown[])
   364→              .map((e) => (typeof e === "string" ? e : (e as { name?: string })?.name || ""))
   365→              .filter(Boolean)
   366→          : [];
   367→        const subjects = Array.isArray(d.subjects) ? d.subjects : [];
   368→        return NextResponse.json(
   369→          {
   370→            success: true,
   371→            data: {
   372→              _id: d._id || batchId,
   373→              id: d.slug || batchId,
   374→              name: d.name || "Batch",
   375→              byName: d.byName || "",
   376→              description: typeof d.description === "string" ? d.description : "",
   377→              imageUrl: d.previewImage || d.previewImageUrl || "",
   378→              exams,
   379→              language: d.language || "",
   380→              className:
   381→                typeof d.class === "string"
   382→                  ? d.class
   383→                  : Array.isArray(d.class)
   384→                    ? (d.class as unknown[]).map((c) => String(c)).join(", ")
   385→                    : "",
   386→              amount: typeof d.feeTotal === "number" ? d.feeTotal : null,
   387→              startDate: d.startDate || "",
   388→              endDate: d.endDate || "",
   389→              // subject mapping — subjectId = SLUG (SubjectInfo/TopicInfo are
   390→              // slug-addressed), batchSubjectId & masterId for dpp/schedule apis
   391→              subjects: subjects.map((s) => ({
   392→                _id: s.slug || s.subjectId || s._id || "",
   393→                subject: s.subject || "",
   394→                subjectId: s.slug || s.subjectId || "",
   395→                batchSubjectId: s._id || "",
   396→                masterId: s.subjectId || "",
   397→              })),
   398→            },
   399→          },
   400→          { headers: CORS_HEADERS },
   401→        );
   402→      }
   403→
   404→      case "pw_sub_topics": {
   405→        const batchId = str(params.batchId).trim();
   406→        const subjectId = str(params.subjectId).trim();
   407→        const page = num(params.page, 1);
   408→        if (!batchId || !subjectId) {
   409→          return NextResponse.json({ success: true, data: { data: [] } }, { headers: CORS_HEADERS });
   410→        }
   411→        try {
   412→          const chapters = await lxChapters(batchId, subjectId, page);
   413→          const data = chapters.map((c, i) => ({
   414→            _id: c._id || "",
   415→            name: c.name || "Chapter",
   416→            typeId: subjectId,
   417→            notes: Number(c.notes || 0),
   418→            videos: Number(c.videos || c.lectureVideos || 0),
   419→            exercises: Number(c.exercises || 0),
   420→            order: Number.isFinite(c.displayOrder) ? Number(c.displayOrder) : i,
   421→          }));
   422→          return NextResponse.json({ success: true, data: { data } }, { headers: CORS_HEADERS });
   423→        } catch {
   424→          return NextResponse.json({ success: true, data: { data: [] } }, { headers: CORS_HEADERS });
   425→        }
   426→      }
   427→
   428→      case "pw_res_topics":
   429→        // upstream has no separate resource folders
   430→        return NextResponse.json({ success: true, data: { data: [] } }, { headers: CORS_HEADERS });
   431→
   432→      case "pw_sch_cntnt": {
   433→        const batchId = str(params.batchId).trim();
   434→        const subjectId = str(params.subjectId).trim();
   435→        const tagId = str(params.tagId).trim();
   436→        const contentType = str(params.contentType, "ALL");
   437→        const page = num(params.page, 1);
   438→        if (!batchId || !subjectId || !tagId) {
   439→          return NextResponse.json({ success: true, data: [] }, { headers: CORS_HEADERS });
   440→        }
   441→        const kind = resolveContentKind(contentType);
   442→        if (!kind) return NextResponse.json({ success: true, data: [] }, { headers: CORS_HEADERS });
   443→        try {
   444→          const items = await lxTopicContent(batchId, subjectId, tagId, upstreamContentType(contentType), page);
   445→          const nodes = contentNodesFromItems(batchId, subjectId, items, kind);
   446→          return NextResponse.json({ success: true, data: nodes }, { headers: CORS_HEADERS });
   447→        } catch {
   448→          return NextResponse.json({ success: true, data: [] }, { headers: CORS_HEADERS });
   449→        }
   450→      }
   451→
   452→      case "pw_dpp_lst": {
   453→        const batchId = str(params.batchId).trim();
   454→        const batchSubjectId = str(params.batchSubjectId).trim();
   455→        const chapterId = str(params.chapterId).trim();
   456→        const page = num(params.page, 1);
   457→        const limit = num(params.limit, 20);
   458→        if (!batchId || !batchSubjectId || !chapterId) {
   459→          return NextResponse.json({ success: true, data: { data: [] } }, { headers: CORS_HEADERS });
   460→        }
   461→        try {
   462→          const items = await lxDppList(batchId, batchSubjectId, chapterId, page, limit);
   463→          const data = items
   464→            .map((it) => {
   465→              const q = it.dppQuizDetails;
   466→              const t = q?.test;
   467→              if (!t?._id) return null;
   468→              return {
   469→                _id: t._id,
   470→                name: t.name || "DPP",
   471→                totalQuestions: Number(t.totalQuestions || 0),
   472→                totalMarks: Number(t.totalMarks || 0),
   473→                duration: Number(t.maxDuration || 0) * 60 || 1800,
   474→                testActivityStatus: "Not Started",
   475→                testStudentMapping: null,
   476→                dppQuizDetails: {
   477→                  contentId: q?.contentId || t._id,
   478→                  scheduleId: q?.scheduleId || "",
   479→                  test: {
   480→                    _id: t._id,
   481→                    name: t.name || "DPP",
   482→                    totalQuestions: Number(t.totalQuestions || 0),
   483→                    totalMarks: Number(t.totalMarks || 0),
   484→                  },
   485→                  testStudentMapping: { testActivityStatus: q?.tag || "Start" },
   486→                },
   487→              };
   488→            })
   489→            .filter(Boolean);
   490→          return NextResponse.json({ success: true, data: { data } }, { headers: CORS_HEADERS });
   491→        } catch {
   492→          return NextResponse.json({ success: true, data: { data: [] } }, { headers: CORS_HEADERS });
   493→        }
   494→      }
   495→
   496→      case "pw_dpp_start": {
   497→        // Auth-gated upstream: anonymous calls fail → surfaced honestly.
   498→        const testId = str(params.testId).trim();
   499→        const batchId = str(params.batchId).trim();
   500→        if (!testId || !batchId) {
   501→          return NextResponse.json(
   502→            { success: false, error: "testId and batchId are required" },
   503→            { headers: CORS_HEADERS },
   504→          );
   505→        }
   506→        try {
   507→          const data = await lxStartDpp({
   508→            testId,
   509→            batchId,
   510→            type: str(params.type, "Start"),
   511→            scheduleId: str(params.scheduleId),
   512→            batchScheduleId: str(params.batchScheduleId) || str(params.scheduleId),
   513→            subjectId: str(params.subjectId),
   514→          });
   515→          return NextResponse.json({ success: true, data }, { headers: CORS_HEADERS });
   516→        } catch (e) {
   517→          return NextResponse.json(
   518→            { success: false, error: e instanceof Error ? e.message : "Failed to start DPP" },
   519→            { headers: CORS_HEADERS },
   520→          );
   521→        }
   522→      }
   523→
   524→      case "pw_test_lst": {
   525→        const batchId = str(params.batchId).trim();
   526→        if (!batchId) return NextResponse.json({ success: true, data: [] }, { headers: CORS_HEADERS });
   527→        try {
   528→          const tests = await lxBatchTests(batchId);
   529→          const data = tests.map((t) => {
   530→            const o = t as Record<string, unknown>;
   531→            return {
   532→              _id: str(o._id),
   533→              name: str(o.name, "Test"),
   534→              totalQuestions: Number(o.totalQuestions || 0),
   535→              totalMarks: Number(o.totalMarks || 0),
   536→              duration: Number(o.maxDuration || 0) * 60 || 3600,
   537→              startTime: str(o.startTime),
   538→              testActivityStatus: "Not Started",
   539→              testStudentMapping: null,
   540→              tag2: str(o.tag2, "Start"),
   541→            };
   542→          });
   543→          return NextResponse.json({ success: true, data }, { headers: CORS_HEADERS });
   544→        } catch {
   545→          return NextResponse.json({ success: true, data: [] }, { headers: CORS_HEADERS });
   546→        }
   547→      }
   548→
   549→      case "pw_test_dtl": {
   550→        const testId = str(params.testId).trim();
   551→        const batchId = str(params.batchId).trim();
   552→        if (!testId || !batchId) {
   553→          return NextResponse.json({ success: true, data: null }, { headers: CORS_HEADERS });

   566→        const detail = await lxScheduleDetail(batchId, subjectId, scheduleId);

   568→

   570→          .flatMap((hw) => {

   630→      case "pw_tchr_dtl":

   634→        const childId = str(params.childId).trim();
   635→        const batchId = str(params.batchId).trim();
   636→        const subjectId = str(params.subjectId).trim();

   646→            { success: false, error: e instanceof Error ? e.message : "Stream resolution failed" },
   647→            { headers: CORS_HEADERS },
   648→          );
   649→        }
   650→      }
   651→
   652→      case "pw_test_map": {
   653→        // testId may be single or array; can live in params or payload.
   654→        const payloadParams = asParams(payload);
   655→        const raw: unknown =
   656→          params.testId ??
   657→          payloadParams.testId ??
   658→          (Array.isArray(payload) ? payload : undefined);

   670→      }
   671→
   672→      default:
   673→        return NextResponse.json(
   674→          { success: false, error: "Unknown action" },
   675→          { status: 400, headers: CORS_HEADERS },
   676→        );
   677→    }
   678→  } catch (error) {
   679→    return NextResponse.json(
   680→      { success: false, error: error instanceof Error ? error.message : String(error) },
   681→      { status: 500, headers: CORS_HEADERS },
   682→    );
   683→  }
   684→}
   685→
   686→/** Flatten test ids from scalars / arrays / objects ({testId}|{_id}|{id}). */
   687→function collectTestIds(source: unknown, out: string[], seen: Set<string>): void {
   688→  if (source === undefined || source === null) return;
   689→  if (Array.isArray(source)) {
   690→    source.forEach((item) => collectTestIds(item, out, seen));
   691→    return;
   692→  }
   693→  if (typeof source === "object") {
   694→    const o = source as Record<string, unknown>;
   695→    collectTestIds(o.testId ?? o._id ?? o.id, out, seen);
   696→    return;
   697→  }
   698→  const s = String(source).trim();
   699→  if (s && !seen.has(s)) {
   700→    seen.add(s);
   701→    out.push(s);
   702→  }
   703→}
