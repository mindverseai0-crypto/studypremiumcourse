   425→        function timeAgo(dateString) {
   426→            const date = new Date(dateString);
   427→            const now = new Date();
   428→            const seconds = Math.round((now - date) / 1000);
   429→            if(seconds < 60) return `${seconds}s ago`;
   430→            const minutes = Math.round(seconds / 60);
   431→            if(minutes < 60) return `${minutes}m ago`;
   432→            const hours = Math.round(minutes / 60);
   433→            if(hours < 24) return `${hours}h ago`;
   434→            const days = Math.round(hours / 24);
   435→            if(days < 7) return `${days}d ago`;
   436→            return date.toLocaleDateString('en-GB');
